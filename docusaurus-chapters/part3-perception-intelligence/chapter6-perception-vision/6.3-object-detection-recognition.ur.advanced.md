# آبجیکٹ ڈیٹیکشن اور ریکوگنیشن (اردو ایڈوانسڈ ورژن)

## باب کا جائزہ

روبوٹکس میں اعلیٰ معیار کی آبجیکٹ ڈیٹیکشن اور ریکوگنیشن جدید کمپیوٹیشنل طریقے استعمال کرتی ہے جن میں گہرے نیورل نیٹ ورکس، احتمالاتی ڈھانچے، اور آپٹیمائزیشن کی تکنیکس شامل ہیں۔ یہ صلاحیتیں خودمختار سسٹم کو پیچیدہ، متحرک ماحول میں اعلیٰ درستگی اور مضبوطی کے ساتھ اشیاء کی شناخت، جگہ کا تعین، اور سمجھ کی اجازت دیتی ہیں۔ یہ باب جدید آبجیکٹ ڈیٹیکشن اور ریکوگنیشن سسٹم کی ریاستی فن کی میتھوڈالوجیز، اعلیٰ معیار کی تعمیرات، اور روبوٹک ایپلی کیشنز میں نفاذ کے چیلنجز کا جائزہ لیتا ہے۔

### سیکھنے کے اہداف
- اعلیٰ معیار کی آبجیکٹ ڈیٹیکشن تعمیرات اور فریموں پر قابو پانا
- حقیقی وقت کے انفرس کے لیے آپٹیمائزیشن تکنیکس کو سمجھنا
- جدید تربیتی حکمت عمل اور نقصان کے فنکشنز کو نافذ کرنا
- بہتر ریکوگنیشن کے لیے ملٹی-موڈل فیوژن کا اطلاق
- کارکردگی کے کاروباروں کا تجزیہ اور روبوٹک ڈیپلائمنٹ کے لیے بہتر بنانا

## ریاضیاتی بنیادیں

### آبجیکٹ ڈیٹیکشن کے لیے احتمالاتی ڈھانچہ

احتمالاتی ڈھانچے میں، آبجیکٹ ڈیٹیکشن کو اشیاء کے وجود اور صفات کے بعد کے امکان کا تخمینہ کے طور پر فارمولیٹ کیا جاتا ہے::

P(آبجیکٹ, جگہ, کلاس | تصویر) = P(تصویر | آبجیکٹ, جگہ, کلاس) × P(آبجیکٹ, جگہ, کلاس) / P(تصویر)

ایک مقررہ تصویر کے لیے، یہ بسط کے شمار کو زیادہ سے زیادہ کرنے کا نتیجہ دیتا ہے۔

### احتمالاتی نقشے کے ساتھ جگہی مقام

جدید نقطہ نظر میں، جگہی مقام کو اس طرح کیا جاتا ہے کہ ہر جگہی پوزیشن اشیاء کے ہونے کا امکان ظاہر کرتا ہے::

P(آبجیکٹ_سینٹر | فیچر_میپ_جگہ) = σ(Conv(فیچر_میپ))

جہاں σ سگموڈ فعالیت ہے جو کنولوشنل فیچرز کو امکانات میں تبدیل کرتا ہے۔

### کثیر کام لینے کی تربیت کا فارمولیشن

آبجیکٹ ڈیٹیکشن ایک کثیر کام کا مسئلہ ہے جس میں مندرجہ ذیل کو جوڑا جاتا ہے::
1. کلاسیفکیشن: P(کلاس | آبجیکٹ_باکس)
2. جگہی وضاحت: P(باؤنڈنگ_بکس_پیرامیٹر | آبجیکٹ_باکس)
3. کمپیٹنس: P(کمپیٹنس | آبجیکٹ_باکس)

مشترکہ امکان کو عام طور پر اس طرح تقسیم کیا جاتا ہے::
P(کلاس، بکس، کمپیٹنس | تصویر) ∝ P(کلاس|·) × P(بکس|·) × P(کمپیٹنس|·)

class RegionProposalNetwork(nn.Module):
    def __init__(self, in_channels, anchor_sizes, anchor_ratios):
        super().__init__()
        self.anchor_num = len(anchor_sizes) * len(anchor_ratios)
        self.anchor_sizes = anchor_sizes
        self.anchor_ratios = anchor_ratios
        
        # اینچر جنریشن کنولوشنز
        self.rpn_conv = nn.Conv2d(in_channels, 512, 3, padding=1)
        self.objectness = nn.Conv2d(512, 2 * self.anchor_num, 1)  # آبجیکٹ/غیر-آبجیکٹ
        self.bbox_regression = nn.Conv2d(512, 4 * self.anchor_num, 1)  # باؤنڈنگ بکس ڈیلٹا
    
    def forward(self, features):
        x = F.relu(self.rpn_conv(features))
        
        # آبجیکٹ نیس اسکور جنریٹ کریں
        object_scores = self.objectness(x)
        # دوبارہ ترتیب دیں: [batch, 2*A, H, W] -> [batch, H, W, A, 2]
        batch_size, _, h, w = object_scores.shape
        object_scores = object_scores.view(batch_size, 2, self.anchor_num, h, w)
        object_scores = object_scores.permute(0, 3, 4, 2, 1).contiguous()
        object_scores = object_scores.view(batch_size, h * w * self.anchor_num, 2)
        
        # باؤنڈنگ بکس ریگریشن ڈیلٹا جنریٹ کریں
        bbox_deltas = self.bbox_regression(x)
        bbox_deltas = bbox_deltas.view(batch_size, 4, self.anchor_num, h, w)

    def forward(self, images, targets=None):
        # فیچر ایکسٹریکٹ کریں

        object_scores, bbox_deltas = self.rpn(features)
        
        # NMS لاگو کریں اور بہترین پروپوزل منتخب کریں
        proposals = self.proposal_generator(object_scores, bbox_deltas)

        bbox_deltas = self.bbox_regressor(pooled_features)
        
        if self.training:

            return loss

            detections = self.post_process(class_scores, bbox_deltas, proposals)

class YOLOv5Head(nn.Module):
    def __init__(self, num_classes, anchors, channels_list):
        super().__init__()
        self.num_classes = num_classes
        self.anchors = anchors
        self.nl = len(anchors)  # ڈیٹیکشن لیئرز کی تعداد
        self.na = len(anchors[0])  # ہر لیئر کے لیے اینچر کی تعداد
        self.no = num_classes + 5  # آؤٹ پٹ کی تعداد (کلاسز + bbox + کمپیٹنس)
        
        # ڈیٹیکشن لیئرز
        self.m = nn.ModuleList(nn.Conv2d(x, self.no * self.na, 1) for x in channels_list)
    
    def forward(self, x):
        z = []  # انفرس کے لیے آؤٹ پٹ
        outputs = []  # خام آؤٹ پٹ
        
        for i, layer in enumerate(self.m):
            x[i] = layer(x[i])  # ڈیٹیکشن کنولوشن لاگو کریں
            bs, _, ny, nx = x[i].shape  # بیچ سائز، اینچر کی تعداد، گرڈ کی اونچائی، گرڈ کی چوڑائی
            
            # دوبارہ ترتیب دیں: [batch, num_anchors, num_outputs, ny, nx] -> [batch, grid_h, grid_w, num_anchors, num_outputs]
            x[i] = x[i].view(bs, self.na, self.no, ny, nx).permute(0, 1, 3, 4, 2).contiguous()
            
            outputs.append(x[i])
            
            if not self.training:
                # امکانات اور آف سیٹ حاصل کرنے کے لیے سگموڈ لاگو کریں
                y = x[i].sigmoid()
                
                # گرڈ کوآرڈینیٹس حاصل کریں
                grid = self._make_grid(nx, ny).to(x[i].device)
                
                # بکس کوآرڈینیٹس کو ڈیکوڈ کریں
                # مرکزی کوآرڈینیٹس (x, y) گرڈ میں شامل کیے گئے آف سیٹ ہیں
                y[..., 0:2] = (y[..., 0:2] * 2.0 - 0.5 + grid) * self.stride[i]
                # چوڑائی اور اونچائی (w, h) اسکیلنگ فیکٹر ہیں
                y[..., 2:4] = (y[..., 2:4] * 2) ** 2 * self.anchor_grid[i]
                
                z.append(y.view(bs, -1, self.no))
        
        return outputs if self.training else torch.cat(z, 1)

### اینچر-فری ڈیٹیکٹرز

**FCOS (Fully Convolutional One-Stage)**::

```python
class FCOSHead(nn.Module):
    def __init__(self, in_channels, num_classes, num_convs=4):
        super().__init__()
        self.num_classes = num_classes
        self.cls_convs = nn.ModuleList()
        self.reg_convs = nn.ModuleList()
        
        # کلاسیفکیشن ٹاور
        for i in range(num_convs):
            in_ch = in_channels if i == 0 else 256
            self.cls_convs.append(nn.Conv2d(in_ch, 256, 3, padding=1))
            self.cls_convs.append(nn.GroupNorm(32, 256))
            self.cls_convs.append(nn.ReLU())
        
        # ریگریشن ٹاور
        for i in range(num_convs):
            in_ch = in_channels if i == 0 else 256
            self.reg_convs.append(nn.Conv2d(in_ch, 256, 3, padding=1))
            self.reg_convs.append(nn.GroupNorm(32, 256))
            self.reg_convs.append(nn.ReLU())
        
        # حتمی پریڈکشن لیئرز
        self.cls_pred = nn.Conv2d(256, num_classes, 3, padding=1)
        self.reg_pred = nn.Conv2d(256, 4, 3, padding=1)  # l, t, r, b فاصلے
        self.centerness_pred = nn.Conv2d(256, 1, 3, padding=1)
    
    def forward(self, x):
        # کلاسیفکیشن ٹاور
        cls_feat = x
        for layer in self.cls_convs:
            cls_feat = layer(cls_feat)
        cls_score = self.cls_pred(cls_feat)
        
        # ریگریشن ٹاور
        reg_feat = x
        for layer in self.reg_convs:
            reg_feat = layer(reg_feat)
        bbox_pred = self.reg_pred(reg_feat).exp()
        centerness = self.centerness_pred(reg_feat).sigmoid()
        
        return cls_score, bbox_pred, centerness

class FCOS(nn.Module):
    def __init__(self, backbone, num_classes):
        super().__init__()
        self.backbone = backbone
        self.fpn = FPN(backbone.out_channels)
        self.head = FCOSHead(256, num_classes)
    
    def forward(self, images):
        features = self.backbone(images)
        fpn_features = self.fpn(features)
        
        outputs = []
        for feat in fpn_features:
            cls_score, bbox_pred, centerness = self.head(feat)
            outputs.append((cls_score, bbox_pred, centerness))
        
        if self.training:
            # تربیت کے دوران نقصان کا حساب لگائیں
            targets = None  # عمل میں، ہدف فراہم کیے جائیں گے
            losses = self.compute_losses(outputs, targets)
            return losses
        else:
            # انفرس کے دوران ڈیکوڈ اور پوسٹ-پروسیسنگ
            detections = self.decode_outputs(outputs)
            return detections
```

## اعلیٰ تکنیکی تربیتی حکمت عمل

### نقصان کے فنکشنز

**جرنلائزڈ فوکل نقصان**::
کلاسیفکیشن اور باکس ریگریشن دونوں کے نقصان کو جوڑتا ہے::

```python
def generalized_focal_loss(pred, target, alpha=0.25, gamma=2.0):
    """
    آبجیکٹ ڈیٹیکشن کے لیے جرنلائزڈ فوکل نقصان
    """
    p = torch.sigmoid(pred)
    
    # کلاسیفکیشن فوکل نقصان
    ce_loss = F.cross_entropy(pred, target, reduction='none')
    p_t = p * target + (1 - p) * (1 - target)
    loss = ce_loss * ((1 - p_t) ** gamma)
    
    # الفا ویٹنگ شامل کریں
    alpha_t = alpha * target + (1 - alpha) * (1 - target)
    loss = alpha_t * loss
    
    return loss.mean()

def giou_loss(pred_boxes, target_boxes):
    """
    جنرلائزڈ IoU نقصان جو شکل اور اسکیل کا خیال رکھتا ہے
    """
    # عام IoU کا حساب لگائیں
    inter_x1 = torch.max(pred_boxes[..., 0], target_boxes[..., 0])
    inter_y1 = torch.max(pred_boxes[..., 1], target_boxes[..., 1])
    inter_x2 = torch.min(pred_boxes[..., 2], target_boxes[..., 2])
    inter_y2 = torch.min(pred_boxes[..., 3], target_boxes[..., 3])
    
    inter_area = torch.clamp(inter_x2 - inter_x1, min=0) * torch.clamp(inter_y2 - inter_y1, min=0)
    pred_area = (pred_boxes[..., 2] - pred_boxes[..., 0]) * (pred_boxes[..., 3] - pred_boxes[..., 1])
    target_area = (target_boxes[..., 2] - target_boxes[..., 0]) * (target_boxes[..., 3] - target_boxes[..., 1])
    
    union_area = pred_area + target_area - inter_area
    iou = inter_area / (union_area + 1e-7)
    
    # احاطہ کرنے والے باکس کا حساب لگائیں
    enclose_x1 = torch.min(pred_boxes[..., 0], target_boxes[..., 0])
    enclose_y1 = torch.min(pred_boxes[..., 1], target_boxes[..., 1])
    enclose_x2 = torch.max(pred_boxes[..., 2], target_boxes[..., 2])
    enclose_y2 = torch.max(pred_boxes[..., 3], target_boxes[..., 3])
    
    enclose_area = (enclose_x2 - enclose_x1) * (enclose_y2 - enclose_y1)
    
    # GIoU
    giou = iou - (enclose_area - union_area) / (enclose_area + 1e-7)
    
    # نقصان = 1 - GIoU
    return (1 - giou).mean()
```

### ڈیٹا آگمنٹیشن کی حکمت عمل

**موزیک آگمنٹیشن** (YOLOv4/v5 میں استعمال):: 
تربیت کے دوران 4 تصاویر کو 1 میں ملاتا ہے تاکہ تربیت کی کارکردگی اور آبجیکٹ ڈیٹیکشن کی کارکردگی میں بہتری لائی جا سکے۔

```python
def mosaic_augmentation(images, targets, new_size=(640, 640)):
    """
    موزیک ڈیٹا آگمنٹیشن - 4 تربیتی تصاویر کو 1 میں ملاتا ہے
    """
    # 4 بے ترتیب تصاویر حاصل کریں
    batch_size = len(images)
    if batch_size < 4:
        return images, targets
    
    # موزیک تصویر تیار کریں
    mosaic_img = torch.zeros(3, new_size[0], new_size[1])
    
    # مرکزی نقطہ کا حساب لگائیں
    center_x = random.randint(new_size[0] * 0.3, new_size[0] * 0.7)
    center_y = random.randint(new_size[1] * 0.3, new_size[1] * 0.7)
    
    # 4 تصاویر کو خاص تبدیلیوں کے ساتھ گوشوں میں رکھیں
    for idx, (img, target) in enumerate(zip(images[:4], targets[:4])):
        if idx == 0:  # ٹاپ-لیفٹ
            img_crop = img[:, :center_y, :center_x]
            mosaic_img[:, :center_y, :center_x] = img_crop
        elif idx == 1:  # ٹاپ-رائٹ
            img_crop = img[:, :center_y, center_x:]
            mosaic_img[:, :center_y, center_x:] = img_crop  
        elif idx == 2:  # باؤم-لیفٹ
            img_crop = img[:, center_y:, :center_x]
            mosaic_img[:, center_y:, :center_x] = img_crop
        else:  # باؤم-رائٹ
            img_crop = img[:, center_y:, center_x:]
            mosaic_img[:, center_y:, center_x:] = img_crop
        
        # ہدف کوآرڈینیٹس کو موزیک خاکے کے مطابق ایڈجسٹ کریں
        # یہ ہدف کی ایڈجسٹمنٹ کے لیے جاری رکھا جائے گا
        # کوآرڈینیٹس کی ایڈجسٹمنٹ کا کوڈ یہاں ہوگا
    
    return mosaic_img.unsqueeze(0), targets[:1]  # ایڈجسٹڈ ہدف لوٹائیں
```

**مکس اَپ آگمنٹیشن**::
تربیتی ڈیٹا کی تنوع کو بڑھانے کے لیے دو تصاویر اور ان کے ہدف کو لکیری طور پر جوڑتا ہے۔

```python
def mixup_augmentation(img1, target1, img2, target2, alpha=0.2):
    """
    مکس اَپ آگمنٹیشن - دو تصاویر اور ہدف کو لکیری ملانے کے ذریعے
    """
    lam = np.random.beta(alpha, alpha)
    
    # تصاویر کو ملاتا ہے
    mixed_img = lam * img1 + (1 - lam) * img2
    
    # ہدف کو ملاتا ہے
    # یہ زیادہ پیچیدہ ہے اور ہدف کے فارمیٹ پر منحصر ہے
    # باؤنڈنگ باکس کے لیے: دونوں چیزوں کے باکسز کو جوڑنا پڑ سکتا ہے
    mixed_targets = lam * target1 + (1 - lam) * target2
    
    return mixed_img, mixed_targets
```

### وژن-لیڈر فیوژن کے لیے 3D ڈیٹیکشن

class VisionLidarFusion(nn.Module):
    def __init__(self, vision_feature_dim, lidar_feature_dim, fusion_dim):
        super().__init__()
        self.vision_transform = nn.Linear(vision_feature_dim, fusion_dim)
        self.lidar_transform = nn.Linear(lidar_feature_dim, fusion_dim)
        self.fusion_layer = nn.Linear(2 * fusion_dim, fusion_dim)
        self.classifier = nn.Linear(fusion_dim, num_classes)
        
    def forward(self, vision_features, lidar_features):
        # خصوصیات کو مشترکہ جگہ میں تبدیل کریں
        vision_emb = F.relu(self.vision_transform(vision_features))
        lidar_emb = F.relu(self.lidar_transform(lidar_features))
        
        # ضم کریں اور فیوژن کریں
        combined = torch.cat([vision_emb, lidar_emb], dim=-1)
        fused = F.relu(self.fusion_layer(combined))
        
        # کلاسیفکیشن لوٹائیں
        return self.classifier(fused)

class RGBDFusion(nn.Module):
    def __init__(self, rgb_channels=3, depth_channels=1, out_channels=64):
        super().__init__()
        # فیچر ایکسٹریکٹرز
        self.rgb_extractor = nn.Sequential(
            nn.Conv2d(rgb_channels, out_channels//2, 3, padding=1),
            nn.ReLU(),
            nn.Conv2d(out_channels//2, out_channels//2, 3, padding=1),
            nn.ReLU()
        )
        
        self.depth_extractor = nn.Sequential(
            nn.Conv2d(depth_channels, out_channels//2, 3, padding=1),
            nn.ReLU(),
            nn.Conv2d(out_channels//2, out_channels//2, 3, padding=1),
            nn.ReLU()
        )
        
        # فیوژن لیئر
        self.fusion = nn.Conv2d(out_channels, out_channels, 3, padding=1)
        
    def forward(self, rgb, depth):
        rgb_feat = self.rgb_extractor(rgb)
        depth_feat = self.depth_extractor(depth)
        
        # فیچر کو جوڑیں
        combined = torch.cat([rgb_feat, depth_feat], dim=1)
        fused = F.relu(self.fusion(combined))
        
        return fused

class TeacherStudentModel(nn.Module):
    def __init__(self, teacher_model, student_model, temperature=3.0, alpha=0.7):
        super().__init__()
        self.teacher = teacher_model
        self.student = student_model
        self.temperature = temperature
        self.alpha = alpha  # سخت اور نرم ہدف کے درمیان توازن
        
    def forward(self, images, targets=None):
        with torch.no_grad():
            teacher_logits = self.teacher(images) / self.temperature
            
        student_logits = self.student(images) / self.temperature
        
        # نرم ہدف نقصان (ڈسٹیلیشن نقصان)
        soft_loss = F.kl_div(
            F.log_softmax(student_logits, dim=1),
            F.softmax(teacher_logits, dim=1),
            reduction='batchmean'
        ) * (self.temperature ** 2)
        
        if targets is not None:
            # سخت ہدف نقصان
            hard_loss = F.cross_entropy(student_logits * self.temperature, targets)
            
            # مشترکہ نقصان
            total_loss = self.alpha * soft_loss + (1 - self.alpha) * hard_loss
            return total_loss
        else:
            # انفرس
            return self.student(images)
```

class DepthwiseSeparableConv(nn.Module):
    """ڈیپتھ وائز سیپریبل کنولوشن - معیاری کنولوشن کا موثر متبادل"""
    def __init__(self, in_channels, out_channels, stride=1):
        super().__init__()
        # ڈیپتھ وائز کنولوشن - ہر ان پٹ چینل کے لیے ایک فلٹر
        self.depthwise = nn.Conv2d(
            in_channels, in_channels, kernel_size=3, stride=stride,
            padding=1, groups=in_channels, bias=False
        )
        self.bn1 = nn.BatchNorm2d(in_channels)
        
        # پوائنٹ وائز کنولوشن - 1x1 کنولوشن آؤٹ پٹ کو جوڑنے کے لیے
        self.pointwise = nn.Conv2d(
            in_channels, out_channels, kernel_size=1, stride=1,
            padding=0, bias=False
        )
        self.bn2 = nn.BatchNorm2d(out_channels)
    
    def forward(self, x):
        x = F.relu(self.bn1(self.depthwise(x)))
        x = F.relu(self.bn2(self.pointwise(x)))
        return x

class PrototypicalNetwork(nn.Module):
    def __init__(self, feature_dim):
        super().__init__()
        self.encoder = self.build_encoder()
        self.feature_dim = feature_dim
    
    def build_encoder(self):
        """فیچر ایکسٹریکٹر تیار کریں (جیسے، CNN بیک بون)"""
        return nn.Sequential(
            nn.Conv2d(3, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(64, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(64, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.AdaptiveAvgPool2d((1, 1)),
            nn.Flatten(),
            nn.Linear(64, self.feature_dim)
        )
    
    def compute_prototypes(self, support_features, support_labels):
        """
        سپورٹ سیٹ میں ہر کلاس کے لیے پرووٹوٹائپس کا حساب لگائیں
        """
        prototypes = []
        num_classes = support_labels.max().item() + 1
        
        for cls in range(num_classes):
            mask = support_labels == cls
            class_features = support_features[mask]
            prototype = class_features.mean(dim=0)  # اس کلاس کے تمام نمونوں کا اوسط
            prototypes.append(prototype)
        
        return torch.stack(prototypes)
    
    def forward(self, query_features, support_features, support_labels):
        """
        کویری فیچرز اور کلاس پرووٹوٹائپس کے درمیان فاصلہ کا حساب لگائیں
        """
        prototypes = self.compute_prototypes(support_features, support_labels)
        
        # مربع یوکلڈین فاصلے کا حساب لگائیں
        distances = torch.cdist(query_features, prototypes, p=2)
        # م probabilities میں تبدیل کریں سافٹ plus کا استعمال کرتے ہوئے (سو فٹ میکس کی طرح لیکن زیادہ مستحکم)
        logits = -distances
        
        return logits
```

### اعلیٰ معیار کے میٹرکس

**اوسط پریسیژن (AP) انٹرپولیشن کے ساتھ**::

```python
def compute_ap(recalls, precisions, use_07_metric=False):
    """
    2007 PASCAL VOC کے انٹرپولیشن کے ساتھ یا بدون اوسط پریسیژن کا حساب لگائیں
    """
    if use_07_metric:
        # PASCAL VOC 2007 طریقہ: 11 یاداشتی سطحوں پر انٹرپولیٹ
        ap = 0.0
        for t in np.arange(0.0, 1.1, 0.1):
            if np.sum(recalls >= t) == 0:
                p = 0.0
            else:
                p = np.max(precisions[recalls >= t])
            ap += p / 11.0
    else:
        # نیا طریقہ: تاب کے نیچے کا علاقہ
        # منحنی کے اختتام کو 0 پریسیژن کے ساتھ سینٹنل ویلیوز کو شامل کریں
        mrec = np.concatenate(([0.0], recalls, [1.0]))
        mpre = np.concatenate(([0.0], precisions, [0.0]))
        
        # کم ہوتی ہوئی پریسیژن ویلیوز کا حساب لگائیں
        for i in range(mpre.size - 1, 0, -1):
            mpre[i - 1] = np.maximum(mpre[i - 1], mpre[i])
        
        # وہ یاداشتی ویلیوز تلاش کریں جہاں اس کا تبدیلی ہوتی ہے
        i = np.where(mrec[1:] != mrec[:-1])[0]
        
        # جمع کریں (\Delta recall) * precision
        ap = np.sum((mrec[i + 1] - mrec[i]) * mpre[i + 1])
    
    return ap

def compute_map_per_class(detections, ground_truths, num_classes, iou_threshold=0.5):
    """
    ہر کلاس کے لیے اوسط اوسط پریسیژن کا حساب لگائیں
    """
    aps = []
    for class_id in range(num_classes):
        # اس کلاس کے لیے ڈیٹیکشنز اور گراؤنڈ ٹروتھ حاصل کریں
        class_detections = [d for d in detections if d['class'] == class_id]
        class_ground_truths = [gt for gt in ground_truths if gt['class'] == class_id]
        
        # ڈیٹیکشنز کو کمپیٹنس کے لحاظ سے ترتیب دیں (نازک)
        class_detections.sort(key=lambda x: x['confidence'], reverse=True)
        
        # TP اور FP سرنیاں حساب لگائیں
        nd = len(class_detections)
        tp = np.zeros(nd)
        fp = np.zeros(nd)
        
        # گراؤنڈ ٹروتھ میچس
        gt_matches = {}
        for i, gt in enumerate(class_ground_truths):
            gt_matches[i] = False
        
        for d_idx, detection in enumerate(class_detections):
            # اس ڈیٹیکشن کی امیج کے لیے گراؤنڈ ٹرو تلاش کریں
            gt_boxes_in_image = [gt for gt in class_ground_truths if gt['image_id'] == detection['image_id']]
            
            if not gt_boxes_in_image:
                fp[d_idx] = 1
                continue
            
            # سبھی GTs کے ساتھ IoUs کا حساب لگائیں اس امیج میں
            ious = []
            gt_indices = []
            for i, gt in enumerate(gt_boxes_in_image):
                if i in gt_matches and not gt_matches[i]:
                    iou_val = calculate_iou(detection['bbox'], gt['bbox'])
                    ious.append(iou_val)
                    gt_indices.append(i)
            
            if not ious or max(ious) < iou_threshold:
                fp[d_idx] = 1
            else:
                gt_match_idx = gt_indices[np.argmax(ious)]
                
                if gt_matches[gt_match_idx]:
                    fp[d_idx] = 1  # پہلے ہی ایک ڈیٹیکشن سے جڑا ہوا
                else:
                    tp[d_idx] = 1  # درست میچ
                    gt_matches[gt_match_idx] = True
        
        # تجمع TP اور FP کا حساب لگائیں
        cum_tp = np.cumsum(tp)
        cum_fp = np.cumsum(fp)
        
        # یاداشت اور پریسیژن کا حساب لگائیں
        recalls = cum_tp / len(class_ground_truths) if len(class_ground_truths) > 0 else np.zeros(nd)
        precisions = cum_tp / (cum_tp + cum_fp) if (cum_tp + cum_fp).sum() > 0 else np.zeros(nd)
        
        # AP کا حساب لگائیں
        ap = compute_ap(recalls, precisions)
        aps.append(ap)
    
    return np.mean(aps) if aps else 0.0
```

    """
    ٹینسر آرٹی انجن کا استعمال کرتے ہوئے بہتر کردہ انفرس چلائیں
    """
    # ایکسیکیوشن کونٹیکسٹ تیار کریں
    context = engine.create_execution_context()
    
    # CUDA میموری مختص کریں
    input_binding = 0  # پہلا بائنڈنگ ان پٹ ہے
    output_binding = 1  # دوسرا بائنڈنگ آؤٹ پٹ ہے
    
    input_size = trt.volume(engine.get_binding_shape(0)) * engine.max_batch_size * np.dtype(np.float32).itemsize  # بائٹس میں
    output_size = trt.volume(engine.get_binding_shape(1)) * engine.max_batch_size * np.dtype(np.float32).itemsize  # بائٹس میں
    
    # ڈیوائس میموری مختص کریں
    d_input = cuda.mem_alloc(1 * input_size)
    d_output = cuda.mem_alloc(1 * output_size)
    
    bindings = [int(d_input), int(d_output)]
    stream = cuda.Stream()
    
    # ان پٹ ڈیٹا کو ڈیوائس پر منتقل کریں
    cuda.memcpy_htod_async(d_input, input_data, stream)
    
    # ماڈل چلائیں
    context.execute_async(bindings=bindings, stream_handle=stream.handle)
    
    # پریڈکشنز کو ڈیوائس سے واپس لائیں
    outputs = np.empty(output_size//4, dtype=np.float32)
    cuda.memcpy_dtoh_async(outputs, d_output, stream)
    
    stream.synchronize()
    
    return outputs

```python
import threading
import queue
import time

class AsyncObjectDetector:
    def __init__(self, model, num_threads=4):
        self.model = model
        self.input_queue = queue.Queue(maxsize=10)
        self.output_queue = queue.Queue()
        self.num_threads = num_threads
        
        # ورکر تھریڈس شروع کریں
        self.workers = []
        for i in range(num_threads):
            worker = threading.Thread(target=self._worker, args=(i,))
            worker.daemon = True
            worker.start()
            self.workers.append(worker)
    
    def _worker(self, worker_id):
        """ورکر تھریڈ فنکشن"""
        while True:
            try:
                # قطار سے ان پٹ بیچ حاصل کریں
                image_batch = self.input_queue.get(timeout=1)
                
                # ڈیٹیکشن چلائیں
                with torch.no_grad():
                    detections = self.model(image_batch)
                
                # نتائج کو آؤٹ پٹ قطار میں ڈالیں
                self.output_queue.put((worker_id, detections))
                
                # کام ختم ہونے کا نشان لگائیں
                self.input_queue.task_done()
                
            except queue.Empty:
                continue
    
    def submit_batch(self, image_batch):
        """ڈیٹیکشن کے لیے بیچ جمع کرائیں"""
        self.input_queue.put(image_batch)
    
    def get_results(self):
        """دستیاب ڈیٹیکشن نتائج حاصل کریں"""
        results = []
        while not self.output_queue.empty():
            results.append(self.output_queue.get_nowait())
        return results
```

class DomainAdversarialNetwork(nn.Module):
    def __init__(self, feature_extractor, classifier, domain_discriminator):
        super().__init__()
        self.feature_extractor = feature_extractor
        self.classifier = classifier
        self.domain_discriminator = domain_discriminator
        self.grl = GradientReversalLayer()  # گریڈیئنٹ ریورس لیئر
    
    def forward(self, source_images, target_images):
        # دونوں ڈومینز کے لیے فیچرز نکالیں
        source_features = self.feature_extractor(source_images)
        target_features = self.feature_extractor(target_images)
        
        # سورس امیجز کلاسیفائز کریں
        source_predictions = self.classifier(source_features)
        
        # ڈومینز کے درمیان فرق (سورس بمقابلہ ٹارگٹ)
        source_domain_scores = self.domain_discriminator(self.grl(source_features))
        target_domain_scores = self.domain_discriminator(self.grl(target_features))
        
        return {
            'source_predictions': source_predictions,
            'source_domain_scores': source_domain_scores,
            'target_domain_scores': target_domain_scores
        }

class GradientReversalFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, input):
        return input.view_as(input)

    @staticmethod
    def backward(ctx, grad_output):
        grad_input = grad_output.neg()  # گریڈیئنٹ کو الٹیں
        return grad_input
```

        self.query_embed = nn.Embedding(num_queries, transformer.d_model)
        
        # پریڈکشن ہیڈس
        self.class_predictor = nn.Linear(transformer.d_model, num_classes + 1)  # +1 بیک گراؤنڈ کے لیے
        self.bbox_predictor = nn.Linear(transformer.d_model, 4)  # (x, y, w, h)
    
    def forward(self, images):
        # فیچرز نکالیں
        features = self.backbone(images)
        
        # جگہی ابعاد کو فلیٹ کریں
        batch_size, n_channels, h, w = features.shape
        features_flat = features.flatten(2).permute(2, 0, 1)  # [H*W, batch, channels]
        
        # جگہی انکوڈنگس شامل کریں
        pos_embed = self.build_pos_encodings(features)  # نفاذ نہیں دکھایا گیا
        
        # کویری انکوڈنگس
        query_embed = self.query_embed.weight.unsqueeze(1).repeat(1, batch_size, 1)
        
        # ٹرانسفارمر کے ذریعے گزریں
        hs = self.transformer(features_flat, query_embed, pos_embed=pos_embed)
        
        # پریڈکشنز بنائیں
        outputs_class = self.class_predictor(hs)
        outputs_coord = self.bbox_predictor(hs).sigmoid()
        
        return {'pred_logits': outputs_class[-1], 'pred_boxes': outputs_coord[-1]}

def build_position_encoding(hidden_dim):
    """DETR کے لیے جگہی انکوڈنگ بنائیں"""
    N_steps = hidden_dim // 2
    position = torch.arange(N_steps, dtype=torch.float32)  # [0, 1, 2, ..., N_steps-1]
    div_term = torch.exp(torch.arange(0, N_steps, 2, dtype=torch.float32) *
                         -(math.log(10000.0) / N_steps))
    
    pe = torch.zeros(1, N_steps * 2)
    pe[0, 0::2] = torch.sin(position * div_term)
    pe[0, 1::2] = torch.cos(position * div_term)
    
    return pe
```

### خود-نگرانی لرننگ

لیبل والے ڈیٹا کے بغیر نمائندگیاں سیکھنا::

```python
class SiameseNetwork(nn.Module):
    def __init__(self, feature_extractor):
        super().__init__()
        self.feature_extractor = feature_extractor
    
    def forward(self, img1, img2):
        feat1 = self.feature_extractor(img1)
        feat2 = self.feature_extractor(img2)
        return feat1, feat2

def contrastive_loss(feat1, feat2, labels, margin=1.0):
    """
    خود-نگرانی لرننگ کے لیے کنٹراسٹو نقصان
    لیبل: 1 اگر مماثل، 0 اگر غیر مماثل
    """
    distances = F.pairwise_distance(feat1, feat2)
    
    # مماثل جوڑ: فاصلہ کم کرنا چاہتے ہیں
    loss_similar = (labels) * torch.pow(distances, 2)
    
    # غیر مماثل جوڑ: فاصلہ > حد چاہتے ہیں
    zeros = torch.zeros_like(distances)
    loss_dissimilar = (1 - labels) * torch.pow(torch.max(margin - distances, zeros), 2)
    
    return torch.mean(loss_similar + loss_dissimilar)
```
