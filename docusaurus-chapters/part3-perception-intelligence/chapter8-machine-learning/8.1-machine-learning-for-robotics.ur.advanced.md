# میشین لرننگ برائے روبوٹکس (اردو ایڈوانسڈ ورژن)

## باب کا جائزہ

روبوٹکس میں میشین لرننگ جامع الگورتھم اور ریاضیاتی فریم ورکس کو احاطہ کرتا ہے جو خودکار نظام کو تجربے سے سیکھنے، ایڈجسٹ کرنے، اور اپنے طرز عمل کو بہتر بنانے کے قابل بناتا ہے۔ یہ میدان احصائی لرننگ تھیوری، کنٹرول سسٹم، آپٹیمائزیشن تھیوری، اور کمپیوٹیشنل جیومیٹری کو ایک ساتھ جوڑتا ہے تاکہ ذہیب روبوٹک سسٹم تخلیق کیے جا سکیں۔ ریاضیاتی بنیادوں میں احتمالات نظریہ، لکیر میت، ویریشنز کی حساب، اور معلوماتی نظریہ شامل ہیں۔

### ریاضیاتی فریم ورک

روبوٹ لرننگ کے لیے عام آپٹیمائزیشن کا مسئلہ اس طرح ظاہر کیا جا سکتا ہے:

min_θ E[Loss(f_θ(s), a)]

جہاں:
- θ ماڈل پیرامیٹر کی نمائندگی کرتا ہے
- s ریاست کے مشاہدات کی نمائندگی کرتا ہے
- a بہترین ایکشن کی نمائندگی کرتا ہے
- f_θ سیکھی گئی پالیسی کی نمائندگی کرتا ہے
- E ریاست-ایکشن تقسیم پر امکان کی نمائندگی کرتا ہے

### سسٹم آرکیٹیکچر

جدید روبوٹک ML سسٹم سلسلہ وار آرکیٹیکچر کو اپناتے ہیں:

```
ادراک لیئر → سمجھ → منصوبہ بندی → کنٹرول → انجام
    ↓              ↓              ↓         ↓         ↓
  سینسرز      ML ماڈلز    ML پالیسیز  PID/FF   ایکچو ایٹرز
    ↑              ↑              ↑         ↑         ↑
  خام ڈیٹا    فیچرز    ایکشنز    کمانڈز  جسمانی
```

## جدید ترین لرننگ پیراڈائیمز

### ساخت شدہ آؤٹ پٹ کے ساتھ نگرانی شدہ لرننگ

روبوٹکس میں، آؤٹ پٹ کے خلا عام طور پر سادہ سکیلر ویلیوز کے بجائے ساخت شدہ ہوتے ہیں:

**ساخت شدہ پریڈکشن:**
```python
import torch
import torch.nn as nn
from torchcrf import CRF

class StructuredRobotControl(nn.Module):
    def __init__(self, input_dim, num_actions, num_states):
        super().__init__()
        self.feature_extractor = nn.Linear(input_dim, 256)
        self.hidden_layers = nn.Sequential(
            nn.Linear(256, 512),
            nn.ReLU(),
            nn.Linear(512, 512),
            nn.ReLU()
        )
        
        # ہر ایکشن-ریاست جوڑ کے لیے اسکور
        self.action_scorer = nn.Linear(512, num_actions * num_states)
        self.crf = CRF(num_states, batch_first=True)
        
    def forward(self, x, states=None):
        features = self.feature_extractor(x)
        hidden = self.hidden_layers(features)
        
        # ایکشن-ریاست اسکورنگ کے لیے ری شیپ
        scores = self.action_scorer(hidden).view(-1, num_actions, num_states)
        
        if states is not None:
            return self.crf(scores, states)
        else:
            return self.crf.decode(scores)
```

### ری انفورسمنٹ لرننگ فریم ورکس

#### ڈیپ Q-نیٹ ورک (DQN) برائے مسلسل کنٹرول

مسلسل ایکشن سپیس کے لیے، ہم DQN کو وسعت دیتے ہیں:

```python
import torch
import torch.nn as nn
import numpy as np

class DuelingDQN(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(DuelingDQN, self).__init__()
        
        self.feature_extractor = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )
        
        # فائدہ سٹریم
        self.advantage_stream = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim)
        )
        
        # قیمت سٹریم
        self.value_stream = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
    
    def forward(self, state):
        features = self.feature_extractor(state)
        advantages = self.advantage_stream(features)
        values = self.value_stream(features)
        
        # Q(s,a) = V(s) + A(s,a) - mean(A(s,a'))
        q_values = values + (advantages - advantages.mean(dim=1, keepdim=True))
        return q_values

class ContinuousDQN(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(ContinuousDQN, self).__init__()
        
        # ایکشن منتخب کرنے کے لیے ایکٹر نیٹ ورک
        self.actor = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Tanh()  # ایکشنز عام طور پر [-1, 1] میں ہوتے ہیں
        )
        
        # Q-ویلیو کے اندازے کے لیے کریٹک نیٹ ورک
        self.critic = nn.Sequential(
            nn.Linear(state_dim + action_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
    
    def forward(self, state, action=None):
        if action is None:
            action = self.actor(state)
        
        state_action = torch.cat([state, action], dim=1)
        q_value = self.critic(state_action)
        return q_value, action
```

```python
class PPOActorCritic(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(PPOActorCritic, self).__init__()
        
        # مشترکہ فیچر ایکسٹریکٹر
        self.feature_extractor = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU()
        )
        
        # ایکٹر نیٹ ورک (مسلسل ایکشنز کے لیے گاؤسین پالیسی)
        self.actor_mean = nn.Linear(hidden_dim, action_dim)
        self.actor_std = nn.Parameter(torch.ones(action_dim) * 0.1)
        
        # کریٹک نیٹ ورک (ویلیو فنکشن)
        self.critic = nn.Linear(hidden_dim, 1)
    
    def forward(self, state):
        features = self.feature_extractor(state)
        
        # ایکٹر: گاؤسین تقسیم کے لیے مین اور اسٹی ڈیوی ایشن
        action_mean = self.actor_mean(features)
        action_std = torch.clamp(self.actor_std, min=1e-3, max=1.0)
        
        # کریٹک: ریاست کی ویلیو
        state_value = self.critic(features)
        
        return action_mean, action_std, state_value
    
    def get_action(self, state):
        action_mean, action_std, state_value = self.forward(state)
        dist = torch.distributions.Normal(action_mean, action_std)
        action = dist.sample()
        log_prob = dist.log_prob(action).sum(dim=1, keepdim=True)
        return action, log_prob, state_value

        with torch.no_grad():
            next_action, next_log_prob, _ = self.actor.get_action(next_state)
            next_q1 = self.critic1_target(next_state, next_action)
            next_q2 = self.critic2_target(next_state, next_action)
            next_q = torch.min(next_q1, next_q2) - self.alpha * next_log_prob
            target_q = reward + (1 - done) * gamma * next_q
        
        # Q-نیٹ ورک لو losses
        current_q1 = self.critic1(state, action)
        current_q2 = self.critic2(state, action)
        q1_loss = F.mse_loss(current_q1, target_q)
        q2_loss = F.mse_loss(current_q2, target_q)
        
        # پالیسی اپ ڈیٹ
        pi, log_pi, _ = self.actor.get_action(state)
        q1_pi = self.critic1(state, pi)
        q2_pi = self.critic2(state, pi)
        min_q_pi = torch.min(q1_pi, q2_pi)
        policy_loss = ((self.alpha * log_pi) - min_q_pi).mean()
        
        return q1_loss, q2_loss, policy_loss
```

### بےزین نقطہ نظر

روبوٹکس میں عدم یقینی کے کوائف کے لیے:

```python
class BayesianNeuralNetwork(nn.Module):
    def __init__(self, input_dim, output_dim, hidden_dim=256):
        super().__init__()
        
        # ہر لیئر کے وزن کے لیے مین اور ویریئنس
        self.fc1_mean = nn.Linear(input_dim, hidden_dim)
        self.fc1_rho = nn.Parameter(torch.randn(hidden_dim, input_dim) * 1e-3)
        
        self.fc2_mean = nn.Linear(hidden_dim, hidden_dim)
        self.fc2_rho = nn.Parameter(torch.randn(hidden_dim, hidden_dim) * 1e-3)
        
        self.fc_out_mean = nn.Linear(hidden_dim, output_dim)
        self.fc_out_rho = nn.Parameter(torch.randn(output_dim, hidden_dim) * 1e-3)
    
    def forward(self, x, sample=True):
        if sample:
            # پوسٹیریئر سے وزن نمونہ لیں
            w1 = self.fc1_mean.weight + self.fc1_rho.exp() * torch.randn_like(self.fc1_rho)
            w2 = self.fc2_mean.weight + self.fc2_rho.exp() * torch.randn_like(self.fc2_rho)
            w_out = self.fc_out_mean.weight + self.fc_out_rho.exp() * torch.randn_like(self.fc_out_rho)
            
            # نمونہ لیے گئے وزن کے ساتھ فارورڈ پاس
            x = torch.matmul(x, w1.t()) + self.fc1_mean.bias
            x = F.relu(x)
            x = torch.matmul(x, w2.t()) + self.fc2_mean.bias
            x = F.relu(x)
            x = torch.matmul(x, w_out.t()) + self.fc_out_mean.bias
        else:
            # مین وزن استعمال کریں
            x = self.fc1_mean(x)
            x = F.relu(x)
            x = self.fc2_mean(x)
            x = F.relu(x)
            x = self.fc_out_mean(x)
        
        return x

class ThompsonSamplingAgent:
    def __init__(self, state_dim, action_dim, num_models=10):
        self.models = [BayesianNeuralNetwork(state_dim, action_dim) for _ in range(num_models)]
        self.optimizers = [torch.optim.Adam(model.parameters()) for model in self.models]
    
    def select_action(self, state):
        # مختلف ماڈلز سے نمونہ لیں (تھامسن سیمپلنگ)
        model_idx = np.random.randint(len(self.models))
        action = self.models[model_idx](state, sample=True)
        return action
```

```python
class MultimodalFusion(nn.Module):
    def __init__(self, vision_dim, lidar_dim, imu_dim, output_dim):
        super().__init__()
        
        # ہر ماڈل کے لیے الگ الگ اینکوڈرز
        self.vision_encoder = VisionEncoder(vision_dim)
        self.lidar_encoder = PointNet(lidar_dim)
        self.imu_encoder = nn.Sequential(
            nn.Linear(imu_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128)
        )
        
        # متعدد ماڈلز کے لیے کراس- توجہ
        self.cross_attention = nn.MultiheadAttention(
            embed_dim=512, num_heads=8
        )
        
        # حتمی فیوژن لیئر
        self.fusion = nn.Sequential(
            nn.Linear(512 * 3, 1024),  # 512 * 3 for concatenated features
            nn.ReLU(),
            nn.Linear(1024, output_dim)
        )
    
    def forward(self, vision_input, lidar_input, imu_input):
        # ہر ماڈل کو انکوڈ کریں
        vision_features = self.vision_encoder(vision_input)
        lidar_features = self.lidar_encoder(lidar_input)
        imu_features = self.imu_encoder(imu_input)
        
        # توجہ میکنزم کے لیے اسٹیک
        modalities = torch.stack([vision_features, lidar_features, imu_features], dim=1)
        
        # ماڈلز کے ذریعے خود- توجہ
        attended_features, _ = self.cross_attention(
            modalities, modalities, modalities
        )
        
        # فلیٹن اور فیوژن
        fused = attended_features.view(-1, 512 * 3)
        output = self.fusion(fused)
        
        return output
```

### 3D آبجیکٹ ڈیٹیکشن اور منظر کی سمجھ

روبوٹکس کے لیے جدید ترین 3D ادراک:

```python
class VoxelNet(nn.Module):
    def __init__(self, num_classes=10, voxel_size=0.1):
        super().__init__()
        self.voxel_size = voxel_size
        
        # ووکسل فیچر انکوڈنگ
        self.voxel_encoder = nn.Sequential(
            nn.Linear(35, 32),  # 35 = ووکسل کے لیے فیچرز (پوائنٹ کاؤنٹ، میان، ویریئنس، وغیرہ کو شامل کرتے ہوئے)
            nn.ReLU(),
            nn.Linear(32, 64),
            nn.ReLU()
        )
        
        # 3D CNN بیک بون
        self.backbone = nn.Sequential(
            nn.Conv3d(64, 64, 3, padding=1),
            nn.BatchNorm3d(64),
            nn.ReLU(),
            nn.Conv3d(64, 128, 3, padding=1),
            nn.BatchNorm3d(128),
            nn.ReLU(),
            nn.Conv3d(128, 256, 3, padding=1),
            nn.BatchNorm3d(256),
            nn.ReLU()
        )
        
        # ڈیٹیکشن ہیڈز
        self.classification_head = nn.Conv3d(256, num_classes, 1)
        self.regression_head = nn.Conv3d(256, 7, 1)  # 7 = dx, dy, dz, l, w, h, yaw
    
    def voxelization(self, points):
        """پوائنٹ کلاؤڈ کو ووکسل گرڈ میں تبدیل کریں"""
        # کوائف کو ووکسل گرڈ میں کوائف کریں
        voxel_coords = torch.floor(points / self.voxel_size).long()
        
        # پوائنٹس کو ووکسل کے لحاظ سے گروپ کریں
        voxel_dict = {}
        for i, coord in enumerate(voxel_coords):
            key = tuple(coord.tolist())
            if key not in voxel_dict:
                voxel_dict[key] = []
            voxel_dict[key].append(points[i])
        
        # ووکسل فیچرز تخلیق کریں
        voxel_features = []
        voxel_coords_list = []
        
        for coord, pts in voxel_dict.items():
            pts = torch.stack(pts)
            # فیچرز: [mean_xyz, variance_xyz, point_count, ...]
            mean_xyz = pts.mean(dim=0)
            variance_xyz = pts.var(dim=0)
            count = torch.tensor(len(pts), dtype=torch.float32)
            
            voxel_feature = torch.cat([mean_xyz, variance_xyz, count.unsqueeze(0)])
            voxel_features.append(voxel_feature)
            voxel_coords_list.append(torch.tensor(coord))
        
        return torch.stack(voxel_features), torch.stack(voxel_coords_list)
    
    def forward(self, points):
        # ووکسلائزیشن
        voxel_features, voxel_coords = self.voxelization(points)
        
        # ووکسل فیچرز انکوڈ کریں
        encoded_features = self.voxel_encoder(voxel_features)
        
        # 3D گرڈ میں ری شیپ
        # یہ سادہ کرنا ہے - اصل نفاذ مختلف طریقے سے سپارس ووکسلز کو سنبھالے گا
        voxel_grid = self.create_voxel_grid(encoded_features, voxel_coords)
        
        # 3D CNN کے ذریعے پروسیس
        features = self.backbone(voxel_grid)
        
        # پیشن گوئیاں تیار کریں
        class_scores = self.classification_head(features)
        bbox_deltas = self.regression_head(features)
        
        return class_scores, bbox_deltas

class SceneGraphNetwork(nn.Module):
    def __init__(self, node_dim=256, edge_dim=128):
        super().__init__()
        self.node_dim = node_dim
        self.edge_dim = edge_dim
        
        # آبجیکٹ نوڈ انکوڈر
        self.node_encoder = nn.Sequential(
            nn.Linear(512, node_dim),  # آبجیکٹ فیچرز
            nn.ReLU(),
            nn.Linear(node_dim, node_dim)
        )
        
        # ریلیشن شپ ایج انکوڈر
        self.edge_encoder = nn.Sequential(
            nn.Linear(512, edge_dim),  # ریلیشن شپ فیچرز
            nn.ReLU(),
            nn.Linear(edge_dim, edge_dim)
        )
        
        # گراف نیورل نیٹ ورک لیئرز
        self.gnn_layers = nn.ModuleList([
            GNNLayer(node_dim, edge_dim) for _ in range(4)
        ])
        
        # آؤٹ پٹ ہیڈز
        self.relation_predictor = nn.Linear(edge_dim, num_relations)
        self.object_predictor = nn.Linear(node_dim, num_objects)
    
    def forward(self, objects, relationships):
        # نوڈز اور ایجز انکوڈ کریں
        node_features = self.node_encoder(objects)
        edge_features = self.edge_encoder(relationships)
        
        # GNN لیئرز کے ذریعے پاس کریں
        for gnn_layer in self.gnn_layers:
            node_features, edge_features = gnn_layer(node_features, edge_features, relationships)
        
        # پیشن گوئیاں تیار کریں
        object_predictions = self.object_predictor(node_features)
        relation_predictions = self.relation_predictor(edge_features)
        
        return object_predictions, relation_predictions
```

## جدید ترین کنٹرول سسٹم

### ماڈل پریڈکٹو کنٹرول (MPC) لرننگ کے ساتھ

سیکھے ہوئے ماڈلز کو MPC کے ساتھ جوڑنا:

```python
class LearnedMPC:
    def __init__(self, dynamics_model, horizon=10, num_simulations=1000):
        self.dynamics_model = dynamics_model  # سیکھا ہوا ڈائنا مکس ماڈل
        self.horizon = horizon
        self.num_simulations = num_simulations
        self.cost_fn = self.default_cost_function
    
    def default_cost_function(self, state, action, goal):
        """نیویگیشن کے لیے ڈیفالٹ قیمت فنکشن"""
        state_cost = torch.norm(state[:2] - goal[:2], dim=1)  # ہدف تک کا فاصلہ
        action_cost = torch.norm(action, dim=1) * 0.01  # کنٹرول کوشش کو کم کریں
        collision_cost = self.collision_penalty(state) * 10.0
        return state_cost + action_cost + collision_cost
    
    def collision_penalty(self, state):
        """ماحول کی نمائندگی (گرڈ میپ، پوائنٹ کلاؤڈ، وغیرہ) کے ذریعے سادہ کولیژن پینلٹی"""
        return torch.zeros(state.shape[0])
    
    def optimize(self, current_state, goal_state):
        """سیکھے ہوئے ماڈل کا استعمال کرکے کنٹرول سیکوئنس کو بہتر بنائیں"""
        best_sequence = None
        best_cost = float('inf')
        
        for _ in range(self.num_simulations):
            # بے ترتیب ایکشن سیکوئنس نمونہ لیں
            action_sequence = torch.randn(self.horizon, current_state.shape[-1], device=current_state.device)
            action_sequence = torch.clamp(action_sequence, -1, 1)  # مناسب حد میں کلیمپ کریں
            
            # ٹریجکٹری سیمولیٹ کریں
            state = current_state.unsqueeze(0)
            total_cost = 0
            
            for t in range(self.horizon):
                # سیکھے ہوئے ڈائنا مکس کا استعمال کرکے اگلی حالت کی پیشن گوئی کریں
                next_state = self.dynamics_model(state, action_sequence[t].unsqueeze(0))
                
                # قیمت کمپیوٹ کریں
                cost = self.cost_fn(next_state, action_sequence[t], goal_state)
                total_cost += cost.item()
                
                state = next_state
            
            # اگر یہ بہتر ہے تو بہترین سیکوئنس کو اپ ڈیٹ کریں
            if total_cost < best_cost:
                best_cost = total_cost
                best_sequence = action_sequence.clone()
        
        return best_sequence[0]  # سیکوئنس میں پہلا ایکشن واپس کریں

class DifferentiableMPC(nn.Module):
    def __init__(self, state_dim, action_dim, horizon=10):
        super().__init__()
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.horizon = horizon
        
        # سیکھا ہوا ڈائنا مکس ماڈل
        self.dynamics = nn.Sequential(
            nn.Linear(state_dim + action_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 256),
            nn.ReLU(),
            nn.Linear(256, state_dim)
        )
        
        # قیمت فنکشن پیرامیٹرز
        self.Q = nn.Parameter(torch.eye(state_dim))  # ریاست قیمت میٹرکس
        self.R = nn.Parameter(torch.eye(action_dim))  # ایکشن قیمت میٹرکس
        
    def forward(self, initial_state, goal_state):
        # ایکشن سیکوئنس کو شروع کریں
        actions = nn.Parameter(torch.randn(self.horizon, self.action_dim, requires_grad=True))
        
        # گریڈیئنٹ ڈیسینٹ کا استعمال کرکے بہتر بنائیں
        optimizer = torch.optim.Adam([actions], lr=0.01)
        
        for _ in range(50):  # آپٹیمائزیشن دہراؤ
            optimizer.zero_grad()
            
            # ٹریجکٹری سیمولیٹ کریں
            state = initial_state
            total_cost = 0
            
            for t in range(self.horizon):
                # اگلی حالت کی پیشن گوئی کریں
                state_action = torch.cat([state, actions[t]], dim=0)
                next_state = state + self.dynamics(state_action)  # ریزیڈوئل ڈائنا مکس
                
                # قیمت کمپیوٹ کریں
                state_cost = (next_state - goal_state).T @ self.Q @ (next_state - goal_state)
                action_cost = actions[t].T @ self.R @ actions[t]
                total_cost = total_cost + state_cost + action_cost
                
                state = next_state
            
            total_cost.backward()
            optimizer.step()
        
        return actions[0]  # بہترین پہلا ایکشن واپس کریں
```

```python
class MAMLRobotController(nn.Module):
    """تیزی سے ایڈاپٹ کرنے کے لیے ماڈل-اگنواسٹک میٹا-لرننگ"""
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super().__init__()
        self.state_dim = state_dim
        self.action_dim = action_dim
        
        # مرکزی پالیسی نیٹ ورک
        self.policy = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Tanh()
        )
    
    def forward(self, state, params=None):
        if params is None:
            return self.policy(state)
        
        # حسب ضرورت پیرامیٹر لاگو کریں (میٹا-لرننگ اپ ڈیٹس کے لیے)
        x = F.linear(state, params[0], params[1])
        x = F.relu(x)
        x = F.linear(x, params[2], params[3])
        x = F.relu(x)
        x = F.linear(x, params[4], params[5])
        x = torch.tanh(x)
        return x
    
    def get_flat_params(self):
        """فلیٹ پیرامیٹر حاصل کریں"""
        return torch.cat([p.flatten() for p in self.parameters()])
    
    def set_flat_params(self, flat_params):
        """فلیٹ ٹینسر سے پیرامیٹر سیٹ کریں"""
        param_idx = 0
        for param in self.parameters():
            num_params = param.numel()
            param.data = flat_params[param_idx:param_idx + num_params].view(param.shape)
            param_idx += num_params

class MetaLearningSystem:
    def __init__(self, robot_controller, learning_rate=0.001, meta_lr=0.01):
        self.controller = robot_controller
        self.optimizer = torch.optim.Adam(robot_controller.parameters(), lr=learning_rate)
        self.meta_optimizer = torch.optim.Adam(robot_controller.parameters(), lr=meta_lr)
    
    def meta_train_step(self, tasks):
        """میٹا-ٹریننگ کا ایک قدم انجام دیں"""
        meta_loss = 0
        
        for task in tasks:
            # ابتدائی پیرامیٹر حاصل کریں
            original_params = self.controller.get_flat_params()
            
            # مخصوص کام کے لیے ایڈاپٹ (انر لوپ)
            task_loss = 0
            for episode in task['adaptation_episodes']:
                state, action, reward = episode
                pred_action = self.controller(state)
                task_loss += F.mse_loss(pred_action, action)
            
            # گریڈیئنٹس کمپیوٹ کریں
            task_grads = torch.autograd.grad(task_loss, self.controller.parameters())
            
            # جائزہ کے لیے پیرامیٹر اپ ڈیٹ کریں
            adapted_params = []
            for param, grad in zip(self.controller.parameters(), task_grads):
                adapted_params.append(param - 0.01 * grad)  # انر لوپ لرننگ ریٹ
            
            # کام-مخصوص ٹیسٹ ڈیٹا پر جائزہ لیں
            test_loss = 0
            for episode in task['test_episodes']:
                state, action, reward = episode
                pred_action = self.controller(state, adapted_params)
                test_loss += F.mse_loss(pred_action, action)
            
            meta_loss += test_loss
        
        # میٹا-پیرامیٹر اپ ڈیٹ کریں
        self.meta_optimizer.zero_grad()
        meta_loss.backward()
        self.meta_optimizer.step()
```

```python
class SafeRLAgent:
    def __init__(self, policy_network, safety_critic, safety_threshold=0.8):
        self.policy = policy_network
        self.safety_critic = safety_critic  # حفاظتی امکان کا تخمینہ لگاتا ہے
        self.safety_threshold = safety_threshold
    
    def safe_action(self, state):
        """محدودت کے تحت محفوظ ایکشن منتخب کریں"""
        # پالیسی سے خام ایکشن حاصل کریں
        raw_action = self.policy(state)
        
        # حفاظت کا جائزہ لیں
        safety_prob = self.safety_critic(state, raw_action)
        
        if safety_prob >= self.safety_threshold:
            return raw_action
        else:
            # پابندی شدہ آپٹیمائزیشن کا استعمال کرکے محفوظ متبادل کمپیوٹ کریں
            return self.compute_safe_fallback(state, raw_action)
    
    def compute_safe_fallback(self, state, unsafe_action):
        """محفوظ متبادل ایکشن کمپیوٹ کریں"""
        # یہ استعمال کر سکتا ہے: روایتی کنٹرول، موشن پلاننگ، یا محتاط پالیسی
        with torch.no_grad():
            # مثال: غیر محفوظ ایکشن کو محفوظ سیٹ میں پروجیکٹ کریں
            safe_action = self.project_to_safe_set(state, unsafe_action)
        return safe_action
    
    def project_to_safe_set(self, state, action):
        """ایکشن کو محفوظ ایکشن سپیس میں پروجیکٹ کریں"""
        # نفاذ حفاظت کی تصریحات پر منحصر ہے
        # یہ جگہ کا مقصد ہے domain-specific محفوظ پروجیکشن کے لیے
        return torch.clamp(action, -0.5, 0.5)  # محتاط ایکشن باؤنڈز
```

```python
def evaluate_robotic_ml_system(model, environment, num_episodes=100):
    """روبوٹک ML سسٹم کا جامع جائزہ"""
    metrics = {
        'کامیابی کی شرح': [],
        'کام مکمل کرنے کا وقت': [],
        'متواتر انعام': [],
        'حفاظت کی خلاف ورزیاں': [],
        'کارآمدی کا تناسب': [],
        'جنرلائزیشن اسکور': [],
        'نمونہ کارآمدی': []
    }
    
    total_steps = 0
    for episode in range(num_episodes):
        state = environment.reset()
        done = False
        episode_reward = 0
        steps = 0
        safety_violations = 0
        
        while not done:
            action = model.get_action(state)
            next_state, reward, done, info = environment.step(action)
            
            # حفاظت کی پابندیوں کی جانچ کریں
            if not info.get('safe', True):
                safety_violations += 1
            
            state = next_state
            episode_reward += reward
            steps += 1
            total_steps += 1
        
        # ایپی سوڈ میٹرکس کمپیوٹ کریں
        metrics['کامیابی کی شرح'].append(1 if info.get('success', False) else 0)
        metrics['کام مکمل کرنے کا وقت'].append(steps)
        metrics['متواتر انعام'].append(episode_reward)
        metrics['حفاظت کی خلاف ورزیاں'].append(safety_violations)
    
    # ایگریگیٹ میٹرکس
    aggregated = {}
    for key, values in metrics.items():
        if values:
            aggregated[key] = sum(values) / len(values)
        else:
            aggregated[key] = 0
    
    # اضافی مشتق میٹرکس
    aggregated['کام مکمل کرنے کا اوسط اسٹیپس'] = np.mean(metrics['کام مکمل کرنے کا وقت'])
    aggregated['حفاظت کی شرح'] = 1 - (sum(metrics['حفاظت کی خلاف ورزیاں']) / total_steps)
    
    return aggregated

def compute_generalization_metrics(train_env, test_envs, model):
    """مختلف ماحولوں میں جنرلائزیشن میٹرکس کمپیوٹ کریں"""
    train_performance = evaluate_robotic_ml_system(model, train_env)
    test_performances = [evaluate_robotic_ml_system(model, env) for env in test_envs]
    
    # جنرلائزیشن گیپ
    gaps = {}
    for metric in train_performance.keys():
        if metric in test_performances[0]:
            train_val = train_performance[metric]
            test_vals = [perf[metric] for perf in test_performances]
            gaps[f'{metric}_generalization_gap'] = train_val - np.mean(test_vals)
    
    return gaps
```
