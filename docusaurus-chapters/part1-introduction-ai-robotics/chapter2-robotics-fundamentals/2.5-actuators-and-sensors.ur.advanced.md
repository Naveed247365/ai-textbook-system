# ایکچوایٹر اور سینسر - اعلیٰ

ایکچوایٹر اور سینسر روبوٹک کنٹرول سسٹم اور حقیقی دنیا کے درمیان اہم فزیکل انٹرفیس کی نمائندگی کرتے ہیں، جس میں جدید الیکٹرو میکینیکل ڈیوائسز اور جدید سگنل پروسیسنگ کی تکنیکیں شامل ہیں۔ یہ اجزاء فزیکل دنیا کے ساتھ بات چیت کے قابل بنانے کے لیے محتاط طریقے سے ڈیزائن کردہ ٹرانسڈکشن میکنزم کو فعال کرتے ہیں، جس کے لیے بہترین کارکردگی کے لیے فزکس، میٹریل سائنس، اور سگنل پروسیسنگ کی گہری سمجھ کی ضرورت ہوتی ہے۔

## اعلیٰ ایکچوایٹر ٹیکنالوجیز

### جدید الیکٹرک موٹر

جدید روبوٹک ایپلی کیشنز کو اعلیٰ کارکردگی کے ایکچوایٹر کی ضرورت ہوتی ہے جس میں درست کنٹرول، زیادہ پاور ڈینسٹی، اور جدید فیڈ بیک انٹیگریشن شامل ہے۔ برش لیس DC (BLDC) موٹر اور پرمننٹ میگنیٹ سنکرونس موٹر (PMSM) اعلیٰ کارکردگی والی ایپلی کیشنز میں غالب ہیں کیونکہ ان کی کارآمدی اور کنٹرول کی اہلیت۔

ان موٹر میں الیکٹرومیگنیٹک ٹورک مندرجہ ذیل کے مطابق ہے:
**τ = k_t × I_q**

جہاں k_t ٹورک کنٹسٹنٹ ہے اور I_q dq-reference فریم میں کواریچر ایکسیس کرنٹ ہے۔

```python
class AdvancedPMSM:
    def __init__(self, pole_pairs=4, flux_linkage=0.1, inductances=(0.001, 0.001), resistance=0.5):
        self.pp = pole_pairs          # قطب جوڑوں کی تعداد
        self.λ = flux_linkage         # پرمننٹ میگنیٹ فلکس لنکیج
        self.Ld, self.Lq = inductances # d-axis اور q-axis انڈکٹنس
        self.R = resistance           # سٹیٹر ریزسٹنس
        self.θ = 0.0                # روٹر الیکٹرکل زاویہ
        self.ω = 0.0                # زاویہ ویلوسٹی

    def dq_transformation(self, phase_currents, rotor_angle):
        """
        پارک ٹرانسفارمیشن 3-فیز کرنٹس کو dq ریفرنس فریم میں تبدیل کرنے کے لیے
        """
        Ia, Ib, Ic = phase_currents
        θ = rotor_angle

        # پارک ٹرانسفارمیشن میٹرکس
        Id = (2/3) * (Ia * np.cos(θ) + Ib * np.cos(θ - 2*np.pi/3) + Ic * np.cos(θ + 2*np.pi/3))
        Iq = (2/3) * (-Ia * np.sin(θ) - Ib * np.sin(θ - 2*np.pi/3) - Ic * np.sin(θ + 2*np.pi/3))

        return Id, Iq

    def torque_model(self, Id, Iq):
        """
        میگنیٹک سیلیئنسی کو مدنظر رکھتے ہوئے الیکٹرومیگنیٹک ٹورک کا حساب
        τ = 3/2 * pp * [λ * Iq + (Ld - Lq) * Id * Iq]
        """
        torque = 1.5 * self.pp * (self.λ * Iq + (self.Ld - self.Lq) * Id * Iq)
        return torque

    def voltage_model(self, Id, Iq, ω_r):
        """
        کرنٹ کنٹرول کے لیے ضروری وولٹیج کا حساب
        vd = R*Id - ω_e*Lq*Iq
        vq = R*Iq + ω_e*Ld*Id + ω_e*λ
        """
        ω_e = self.pp * ω_r  # الیکٹرکل زاویہ ویلوسٹی

        vd = self.R * Id - ω_e * self.Lq * Iq
        vq = self.R * Iq + ω_e * self.Ld * Id + ω_e * self.λ

        return vd, vq

    def field_oriented_control(self, torque_ref, flux_ref=0.0):
        """
        PMSM کے لیے فیلڈ اورینٹڈ کنٹرول (ویکٹر کنٹرول)
        """
        # سر فیس PMSM کے لیے، زیادہ ٹورک فی ایمپئر کے لیے Id_ref = 0
        Id_ref = flux_ref / self.λ if flux_ref != 0.0 else 0.0
        Iq_ref = torque_ref / (1.5 * self.pp * self.λ)

        return Id_ref, Iq_ref
```

### سیریز الیسٹک ایکچوایٹر (SEA)

سیریز الیسٹک ایکچوایٹر موٹر اور لوڈ کے درمیان ایک سپرنگ کو سیریز میں جوڑتے ہیں، جو ذاتی طور پر مطیع اور درست زور کنٹرول فراہم کرتا ہے۔ یہ محفوظ انسان-روبوٹ بات چیت کے لیے اہم ہے۔

**τ_load = k_spring × (θ_motor - θ_load)**

جہاں k_spring سپرنگ کنٹسٹنٹ ہے۔

```python
class SeriesElasticActuator:
    def __init__(self, motor, spring_constant=1000.0, gear_ratio=100.0):
        self.motor = motor
        self.k = spring_constant  # N⋅m/rad
        self.gear_ratio = gear_ratio
        self.motor_pos = 0.0
        self.load_pos = 0.0
        self.spring_deflection = 0.0

    def compute_force(self):
        """
        سپرنگ ڈیفلیکشن کی بنیاد پر زور کا حساب
        """
        self.spring_deflection = self.motor_pos - self.load_pos/self.gear_ratio
        force = self.k * self.spring_deflection
        return force

    def update(self, motor_voltage, load_torque, dt):
        """
        SEA ڈائنامکس کو اپ ڈیٹ کریں
        """
        # موٹر ڈائنامکس کو اپ ڈیٹ کریں
        motor_torque = self.motor.update_state(motor_voltage, load_torque/self.gear_ratio, dt)
        self.motor_pos += motor_torque[1] * dt  # فرض کریں موٹر ویلوسٹی لوٹاتا ہے

        # زور کا حساب لگائیں اور لوڈ کو اپ ڈیٹ کریں
        force = self.compute_force()
        load_torque = force * self.gear_ratio
        self.load_pos += (load_torque / 10.0) * dt  # فرض کریں لوڈ انیشیا

        return force, self.load_pos
```

### ویری ایبل سٹفنس ایکچوایٹر (VSA)

ویری ایبل سٹفنس ایکچوایٹر حقیقی وقت میں میکینیکل امپیڈنس کو ایڈجسٹ کرنے کی اجازت دیتے ہیں، مختلف ٹاسکس کے لیے بہترین کارکردگی کو فعال کرتے ہیں۔

```python
class VariableStiffnessActuator:
    def __init__(self, primary_motor, stiffness_motor, base_stiffness=100.0):
        self.primary_motor = primary_motor    # پوزیشن کنٹرول کے لیے
        self.stiffness_motor = stiffness_motor  # سٹفنس کنٹرول کے لیے
        self.k_base = base_stiffness
        self.stiffness_ratio = 1.0  # بیس سٹفنس کا ملٹی پلائر

    def update_stiffness(self, stiffness_command):
        """
        ایکچوایٹر کا سٹفنس اپ ڈیٹ کریں
        """
        self.stiffness_ratio = max(0.1, min(10.0, stiffness_command))
        current_stiffness = self.k_base * self.stiffness_ratio
        return current_stiffness

    def control(self, position_ref, stiffness_ref, dt):
        """
        ایک ہی وقت میں پوزیشن اور سٹفنس دونوں کو کنٹرول کریں
        """
        # سٹفنس کو اپ ڈیٹ کریں
        current_stiffness = self.update_stiffness(stiffness_ref)

        # موجودہ سٹفنس پیرامیٹر کے ساتھ پوزیشن کنٹرول
        pos_control = self.primary_motor.command_position(position_ref, dt)

        return {
            'position': pos_control,
            'stiffness': current_stiffness,
            'stiffness_ratio': self.stiffness_ratio
        }
```

## جدید سینسر ٹیکنالوجیز

### ملٹی ایکسیس زور/ٹورک سینسر

چھ ایکسیس زور/ٹورک سینسر زور (Fx, Fy, Fz) اور ٹورک (Mx, My, Mz) کو ایک ہی وقت میں پیمائش کرتے ہیں، جو جٹیل بات چیت کنٹرول کو فعال کرتا ہے۔

```python
class SixAxisForceTorqueSensor:
    def __init__(self, calibration_matrix=None):
        if calibration_matrix is None:
            # شناخت میٹرکس ڈیفالٹ کے طور پر (اصل سینسر کے لیے کیلیبریٹ کیا جائے گا)
            self.C = np.eye(6)
        else:
            self.C = calibration_matrix

        self.bias = np.zeros(6)  # سینسر بائس
        self.raw_readings = np.zeros(6)
        self.compensated_readings = np.zeros(6)

    def measure(self, raw_voltages):
        """
        رو سینسر ریڈنگ کو زور/ٹورک ویلیوز میں تبدیل کریں
        """
        self.raw_readings = np.array(raw_voltages)

        # بائس کمپنیشن کا اطلاق
        biased_readings = self.raw_readings - self.bias

        # کیلیبریشن میٹرکس کا اطلاق
        self.compensated_readings = self.C @ biased_readings

        return {
            'forces': self.compensated_readings[:3],   # Fx, Fy, Fz
            'torques': self.compensated_readings[3:],  # Mx, My, Mz
            'magnitude': np.linalg.norm(self.compensated_readings)
        }

    def bias_calibration(self, num_samples=100):
        """
        بغیر لوڈ کی حالت میں سینسر بائس کیلیبریٹ کریں
        """
        readings_sum = np.zeros(6)
        for _ in range(num_samples):
            # حقیقی نفاذ میں، اصل سینسر پڑھیں
            readings_sum += np.random.normal(0, 0.01, 6)  # شبیہ سازی شدہ نوائز

        self.bias = readings_sum / num_samples
        return self.bias
```

### جدید ویژن سسٹم

جدید روبوٹک ویژن سسٹم متعدد سینسر، جدید کیلیبریشن، اور حقیقی وقت پروسیسنگ کو اکٹھا کرتے ہیں۔

```python
class StereoVisionSystem:
    def __init__(self, left_camera_params, right_camera_params):
        self.left_cam = self._init_camera(left_camera_params)
        self.right_cam = self._init_camera(right_camera_params)
        self.extrinsic_matrix = self._compute_extrinsics()

    def _init_camera(self, params):
        """
        انٹرنسک پیرامیٹر کے ساتھ کیمرہ شروع کریں
        """
        K = np.array([
            [params['fx'], 0, params['cx']],
            [0, params['fy'], params['cy']],
            [0, 0, 1]
        ])
        return {
            'K': K,
            'distortion': np.array(params['distortion_coeffs']),
            'resolution': params['resolution']
        }

    def _compute_extrinsics(self):
        """
        بائیں اور دائیں کیمرے کے درمیان ٹرانسفارمیشن کمپیوٹ کریں
        """
        # ٹرانسلیشن ویکٹر (کیمرے کے درمیان بیس لائن)
        T = np.array([-0.1, 0, 0])  # 10cm بیس لائن

        # ریوٹیشن میٹرکس (سٹیریو سیٹ اپ کے لیے عام طور پر شناخت)
        R = np.eye(3)

        # ایکسٹرنسکس میٹرکس
        extrinsics = np.hstack([R, T.reshape(3, 1)])
        return extrinsics

    def rectify_images(self, left_img, right_img):
        """
        ڈسپیرٹی کمپیوٹیشن کے لیے سٹیریو امیجز کو ریکٹیفائی کریں
        """
        # عمل میں، یہ cv2.stereoRectify اور cv2.initUndistortRectifyMap استعمال کرے گا
        # شبیہ سازی کے لیے، امیجز لوٹائیں
        return left_img, right_img

    def compute_disparity(self, left_img, right_img):
        """
        سیمی گلوبل بلاک میچنگ کا استعمال کرتے ہوئے ڈسپیرٹی میپ کمپیوٹ کریں
        """
        # سادہ نفاذ - حقیقی سسٹم بہترین الگورتھم استعمال کرے گا
        height, width = left_img.shape[:2]
        disparity_map = np.zeros((height, width))

        # سادہ بلاک میچنگ (بہترین نہیں)
        block_size = 15
        max_disparity = 64

        for y in range(block_size//2, height - block_size//2):
            for x in range(block_size//2 + max_disparity, width - block_size//2):
                best_match = float('inf')
                best_d = 0

                left_block = left_img[y:y+block_size, x:x+block_size]

                for d in range(max_disparity):
                    if x - d >= block_size//2:
                        right_block = right_img[y:y+block_size, x-d:x-d+block_size]
                        ssd = np.sum((left_block.astype(float) - right_block.astype(float))**2)

                        if ssd < best_match:
                            best_match = ssd
                            best_d = d

                disparity_map[y, x] = best_d

        return disparity_map

    def depth_from_disparity(self, disparity_map, baseline=0.1, focal_length=500):
        """
        ڈسپیرٹی کو ڈیپتھ میں تبدیل کریں
        depth = (baseline * focal_length) / disparity
        """
        # تقسیم کو صفر سے بچیں
        depth_map = np.zeros_like(disparity_map, dtype=np.float32)
        valid_disparity = disparity_map > 0

        depth_map[valid_disparity] = (baseline * focal_length) / disparity_map[valid_disparity]

        return depth_map
```

### ٹیکٹائل سینسر ایرے

جدید ٹیکٹائل سینسر سطح پر فورس سینسنگ کو تقسیم کرتے ہیں۔

```python
class TactileSensorArray:
    def __init__(self, rows=16, cols=16, sensing_range=(0, 100)):
        self.rows = rows
        self.cols = cols
        self.range = sensing_range
        self.pressure_map = np.zeros((rows, cols))
        self.temperature_map = np.zeros((rows, cols))
        self.taxels = rows * cols  # ٹیکٹائل عناصر

    def sense(self, contact_force_distribution):
        """
        ٹیکٹائل سینسر ریڈنگ کو اپ ڈیٹ کریں
        """
        # ان پٹ کو سینسر ایرے کے مطابق تبدیل کریں
        if contact_force_distribution.size == self.taxels:
            self.pressure_map = contact_force_distribution.reshape((self.rows, self.cols))
        else:
            # ضرورت کے مطابق انٹرپولیٹ یا ڈاؤن سیمپل کریں
            pass

        # رینج حدود لاگو کریں
        self.pressure_map = np.clip(self.pressure_map, self.range[0], self.range[1])

        return self.pressure_map.copy()

    def extract_features(self):
        """
        ٹیکٹائل ڈیٹا سے مطلب نکالیں
        """
        features = {}

        # رابطہ کا پتہ لگانا
        contact_mask = self.pressure_map > 0.1 * self.range[1]  # 10% تھریشولڈ
        features['contact_exists'] = np.any(contact_mask)
        features['contact_area'] = np.sum(contact_mask)

        # سینٹر آف پریشر
        if features['contact_exists']:
            y_indices, x_indices = np.mgrid[0:self.rows, 0:self.cols]
            total_force = np.sum(self.pressure_map)

            if total_force > 0:
                cop_x = np.sum(x_indices * self.pressure_map) / total_force
                cop_y = np.sum(y_indices * self.pressure_map) / total_force
                features['center_of_pressure'] = (cop_x, cop_y)

        # پریشر تقسیم میٹرکس
        features['mean_pressure'] = np.mean(self.pressure_map)
        features['max_pressure'] = np.max(self.pressure_map)
        features['pressure_std'] = np.std(self.pressure_map)

        # شکل اور جہت (سادہ مومنٹ-بیسڈ نقطہ نظر)
        if features['contact_area'] > 1:  # شکل کے لیے کم از کم 2 پوائنٹس کی ضرورت ہے
            mu20 = np.sum((x_indices - features['center_of_pressure'][0])**2 * self.pressure_map) / total_force
            mu02 = np.sum((y_indices - features['center_of_pressure'][1])**2 * self.pressure_map) / total_force
            mu11 = np.sum((x_indices - features['center_of_pressure'][0]) * (y_indices - features['center_of_pressure'][1]) * self.pressure_map) / total_force

            features['orientation'] = 0.5 * np.arctan2(2*mu11, mu20-mu02)
            features['eccentricity'] = np.sqrt(1 - (mu20-mu02)**2 / (mu20+mu02)**2)

        return features
```

## سینسر فیوژن اور اسٹیٹ ایسٹیمیشن

### سینسر فیوژن کے لیے ایکسٹینڈڈ کیلمن فلٹر

اعلیٰ کارکردگی والے روبوٹس کو متعدد سینسر ریڈنگ کو بہترین طریقے سے ملانے کے لیے جدید سینسر فیوژن کی ضرورت ہوتی ہے۔

```python
class ExtendedKalmanFilter:
    def __init__(self, state_dim, measurement_dim):
        self.n = state_dim
        self.m = measurement_dim
        self.x = np.zeros(state_dim)  # اسٹیٹ ویکٹر
        self.P = np.eye(state_dim) * 0.1  # کوواریئنس میٹرکس
        self.Q = np.eye(state_dim) * 0.1  # پروسیس نوائز
        self.R = np.eye(measurement_dim) * 0.1  # پیمائش نوائز

    def predict(self, control_input, f_func, F_func, dt):
        """
        غیر لکیری سسٹم ماڈل کا استعمال کرتے ہوئے پریڈکشن اسٹیپ
        f_func: اسٹیٹ ٹرانزیشن فنکشن x_k = f(x_{k-1}, u_k)
        F_func: اسٹیٹ ٹرانزیشن فنکشن کا جیکوبین
        """
        # اسٹیٹ کی پریڈکشن
        self.x = f_func(self.x, control_input, dt)

        # کوواریئنس کی پریڈکشن
        F = F_func(self.x, control_input, dt)
        self.P = F @ self.P @ F.T + self.Q

    def update(self, measurement, h_func, H_func):
        """
        غیر لکیری پیمائش ماڈل کا استعمال کرتے ہوئے اپ ڈیٹ اسٹیپ
        h_func: پیمائش فنکشن z_k = h(x_k)
        H_func: پیمائش فنکشن کا جیکوبین
        """
        # انوویشن کمپیوٹ کریں
        expected_measurement = h_func(self.x)
        innovation = measurement - expected_measurement

        # جیکوبین کمپیوٹ کریں
        H = H_func(self.x)

        # انوویشن کوواریئنس کمپیوٹ کریں
        S = H @ self.P @ H.T + self.R

        # کیلمن گین کمپیوٹ کریں
        K = self.P @ H.T @ np.linalg.inv(S)

        # اسٹیٹ اور کوواریئنس کو اپ ڈیٹ کریں
        self.x = self.x + K @ innovation
        self.P = (np.eye(self.n) - K @ H) @ self.P

    def get_state(self):
        """موجودہ اسٹیٹ ایسٹیمیٹ لوٹائیں"""
        return self.x.copy()
```

### غیر گاؤسین سسٹم کے لیے پارٹیکل فلٹر

غیر گاؤسین نوائز یا متعدد موڈل ڈسٹری بیوشن والے سسٹم کے لیے، پارٹیکل فلٹر بہتر کارکردگی فراہم کرتا ہے۔

```python
class ParticleFilter:
    def __init__(self, state_dim, num_particles=1000):
        self.state_dim = state_dim
        self.num_particles = num_particles
        self.particles = np.random.randn(num_particles, state_dim) * 0.1
        self.weights = np.ones(num_particles) / num_particles

    def predict(self, control_input, process_noise_std, dt):
        """
        سسٹم ڈائنامکس کا استعمال کرتے ہوئے پارٹیکل اسٹیٹ کی پریڈکشن
        """
        for i in range(self.num_particles):
            # نوائز کے ساتھ سسٹم ڈائنامکس لاگو کریں
            self.particles[i] += self.system_dynamics(self.particles[i], control_input, dt)
            self.particles[i] += np.random.normal(0, process_noise_std, self.state_dim)

    def system_dynamics(self, state, control, dt):
        """
        مثال سسٹم ڈائنامکس (یہ آپ کے سسٹم کے مطابق ہوگا)
        """
        # مثال کے طور پر سادہ لکیری سسٹم
        A = np.eye(self.state_dim) + dt * np.random.randn(self.state_dim, self.state_dim) * 0.1
        return A @ state * dt + control * dt * 0.1

    def update(self, measurement, measurement_std):
        """
        پیمائش کی لائیکلی ہوڈ کی بنیاد پر پارٹیکل ویٹس کو اپ ڈیٹ کریں
        """
        for i in range(self.num_particles):
            # اس پارٹیکل کے لیے متوقع پیمائش کا حساب لگائیں
            expected_measurement = self.particle_measurement_model(self.particles[i])

            # اسٹیٹ پیمائش کے لیے اصل پیمائش کی لائیکلی ہوڈ کا حساب لگائیں
            likelihood = self.gaussian_likelihood(measurement, expected_measurement, measurement_std)
            self.weights[i] *= likelihood

        # ویٹس کو نارملائز کریں
        self.weights = self.weights / np.sum(self.weights)

        # مؤثر نمونہ سائز بہت کم ہے تو ری سیمپل کریں
        if self.effective_sample_size() < self.num_particles / 2:
            self.resample()

    def particle_measurement_model(self, state):
        """
        ایک پارٹیکل کے لیے پیمائش ماڈل (مثال: پہلے اسٹیٹ متغیر کا مشاہدہ)
        """
        return state[0]  # مثال: پہلا اسٹیٹ متغیر مشاہدہ کریں

    def gaussian_likelihood(self, measurement, expected, std):
        """
        گاؤسین لائیکلی ہوڈ کا حساب لگائیں
        """
        diff = measurement - expected
        likelihood = np.exp(-0.5 * (diff / std)**2) / (std * np.sqrt(2 * np.pi))
        return likelihood

    def effective_sample_size(self):
        """
        مؤثر نمونہ سائز کا حساب لگائیں
        """
        return 1.0 / np.sum(self.weights**2)

    def resample(self):
        """
        ویٹس کی بنیاد پر پارٹیکل کو ری سیمپل کریں
        """
        # سسٹمیٹک ری سیمپلنگ
        indices = self.systematic_resample()
        self.particles = self.particles[indices]
        self.weights = np.ones(self.num_particles) / self.num_particles

    def systematic_resample(self):
        """
        سسٹمیٹک ری سیمپلنگ الگورتھم
        """
        indices = np.zeros(self.num_particles, dtype=int)
        cumulative_sum = np.cumsum(self.weights)

        U = np.linspace(0, 1, self.num_particles + 1)
        U = U[:-1] + np.random.uniform(0, 1/self.num_particles)

        i, j = 0, 0
        while i < self.num_particles and j < self.num_particles:
            while cumulative_sum[j] < U[i]:
                j += 1
            indices[i] = j
            i += 1

        return indices

    def estimate(self):
        """
        پارٹیکل کے وزن شدہ اوسط کے طور پر اسٹیٹ ایسٹیمیٹ کا حساب لگائیں
        """
        return np.average(self.particles, axis=0, weights=self.weights)
```

```python
class ImpedanceController:
    def __init__(self, desired_mass=1.0, desired_damping=2.0, desired_stiffness=100.0):
        self.M_d = desired_mass      # مطلوبہ ماس
        self.B_d = desired_damping   # مطلوبہ ڈیمپنگ
        self.K_d = desired_stiffness # مطلوبہ سٹفنس

        self.x_d = np.zeros(3)      # مطلوبہ پوزیشن
        self.x_d_dot = np.zeros(3)  # مطلوبہ ویلوسٹی
        self.x_d_ddot = np.zeros(3) # مطلوبہ ایکسلریشن

        self.x = np.zeros(3)        # موجودہ پوزیشن
        self.x_dot = np.zeros(3)    # موجودہ ویلوسٹی

    def compute_impedance_force(self, position_error, velocity_error):
        """
        امپیڈنس کنٹرول زور کمپیوٹ کریں
        F = M_d * x_d_ddot + B_d * (x_d_dot - x_dot) + K_d * (x_d - x)
        """
        acceleration_term = self.M_d * self.x_d_ddot
        damping_term = self.B_d * velocity_error
        stiffness_term = self.K_d * position_error

        impedance_force = acceleration_term + damping_term + stiffness_term
        return impedance_force

    def control_with_environment(self, desired_pos, current_pos, current_vel,
                               environment_force, dt):
        """
        ماحولیاتی بات چیت کو مدنظر رکھتے ہوئے امپیڈنس کنٹرول انجام دیں
        """
        # غلطیوں کا حساب لگائیں
        pos_error = desired_pos - current_pos
        vel_error = self.x_d_dot - current_vel

        # مطلوبہ امپیڈنس زور کمپیوٹ کریں
        desired_impedance_force = self.compute_impedance_force(pos_error, vel_error)

        # ماحولیاتی زور کمپنیشن لاگو کریں
        total_force = desired_impedance_force - environment_force

        # نتیجہ خیز ایکسلریشن کمپیوٹ کریں
        acceleration = total_force / self.M_d

        # ویلوسٹی اور پوزیشن کو اپ ڈیٹ کریں
        new_velocity = current_vel + acceleration * dt
        new_position = current_pos + new_velocity * dt

        return new_position, new_velocity, total_force
```

### سینسر عدم یقینی کے ساتھ ایڈاپٹو کنٹرول

```python
class AdaptiveSensorController:
    def __init__(self, initial_params, sensor_noise_model):
        self.params = initial_params.copy()
        self.sensor_model = sensor_noise_model
        self.param_covariance = np.eye(len(initial_params)) * 1.0
        self.forgetting_factor = 0.99  # ٹائم ویرینگ پیرامیٹر کے لیے

    def update_with_sensor_uncertainty(self, measurement, regressor, sensor_variance):
        """
        سینسر عدم یقینی کو مدنظر رکھتے ہوئے پیرامیٹر اپ ڈیٹ کریں
        """
        # پریڈکشن غلطی
        prediction = regressor.T @ self.params
        error = measurement - prediction

        # عدم یقینی کے وزن والی گین
        P_phi = self.param_covariance @ regressor
        innovation_variance = regressor.T @ P_phi + sensor_variance

        # سکیلر تقسیم کو محتاط طریقے سے ہینڈل کریں
        if innovation_variance > 1e-8:
            gain = P_phi / innovation_variance
        else:
            gain = P_phi * 1e8  # چھوٹی تغییر کے لیے بڑی گین

        # پیرامیٹر اپ ڈیٹ کریں
        self.params = self.params + gain * error

        # فورگیٹنگ فیکٹر کے ساتھ کوواریئنس اپ ڈیٹ کریں
        self.param_covariance = (self.forgetting_factor *
                               (self.param_covariance - np.outer(gain, P_phi)))

        # یقینی بنائیں کہ کوواریئنس مثبت واضح رہے
        eigenvals = np.linalg.eigvals(self.param_covariance)
        if np.any(eigenvals <= 0):
            # مثبت واضح رکھنے کے لیے چھوٹی مثبت واضح میٹرکس شامل کریں
            self.param_covariance += np.eye(len(eigenvals)) * 0.001

        return self.params, self.param_covariance
```

## ہائی بینڈ وڈتھ سینسر-ایکچوایٹر سسٹم

### حقیقی وقت کنٹرول آرکیٹیکچر

اعلیٰ کارکردگی والے روبوٹک سسٹم محتاط طریقے سے ڈیزائن کردہ حقیقی وقت آرکیٹیکچر کی ضرورت رکھتے ہیں۔

```python
import threading
import time
import queue
from collections import deque

class RealTimeSensorActuatorSystem:
    def __init__(self, control_frequency=1000, sensor_frequency=2000, actuator_frequency=1000):
        self.control_freq = control_frequency    # Hz
        self.sensor_freq = sensor_frequency      # Hz
        self.actuator_freq = actuator_frequency  # Hz

        self.control_dt = 1.0 / control_frequency
        self.sensor_dt = 1.0 / sensor_frequency
        self.actuator_dt = 1.0 / actuator_frequency

        # ملٹی ریٹ آپریشن کے لیے ڈیٹا قیوز
        self.sensor_queue = queue.Queue(maxsize=10)
        self.control_queue = queue.Queue(maxsize=5)
        self.actuator_queue = queue.Queue(maxsize=5)

        # ٹائم کی اعداد و شمار
        self.timing_stats = {
            'sensor': {'min': float('inf'), 'max': 0, 'avg': 0},
            'control': {'min': float('inf'), 'max': 0, 'avg': 0},
            'actuator': {'min': float('inf'), 'max': 0, 'avg': 0}
        }

        # تھریڈز
        self.sensor_thread = None
        self.control_thread = None
        self.actuator_thread = None
        self.running = False

    def sensor_thread_func(self):
        """ہائی فریکوئنسی سینسر سیمپلنگ تھریڈ"""
        next_sample_time = time.time()

        while self.running:
            start_time = time.time()

            # سینسر ریڈنگ کی شبیہ سازی
            sensor_data = self.read_sensors()

            # قیو میں شامل کریں (اگر بھری ہو تو غیر مسدود)
            try:
                self.sensor_queue.put_nowait(sensor_data)
            except queue.Full:
                pass  # سب سے پرانا ڈراپ کریں اگر قیو بھری ہو

            # ٹائم کو برقرار رکھیں
            next_sample_time += self.sensor_dt
            sleep_time = next_sample_time - time.time()
            if sleep_time > 0:
                time.sleep(sleep_time)
            else:
                # ٹائم کی خلاف ورزی کا ریکارڈ کریں
                pass

    def control_thread_func(self):
        """کنٹرول کمپیوٹیشن تھریڈ"""
        next_control_time = time.time()

        while self.running:
            start_time = time.time()

            # تازہ ترین سینسر ڈیٹا حاصل کریں
            try:
                sensor_data = self.sensor_queue.get_nowait()
                while not self.sensor_queue.empty():
                    sensor_data = self.sensor_queue.get_nowait()  # سب سے تازہ حاصل کریں
            except queue.Empty:
                continue  # نیا ڈیٹا نہ ہونے پر چھوڑ دیں

            # کنٹرول کمانڈ کمپیوٹ کریں
            control_command = self.compute_control(sensor_data)

            # ایکچوایٹر قیو میں بھیجیں
            try:
                self.control_queue.put_nowait(control_command)
            except queue.Full:
                pass

            # ٹائم کو برقرار رکھیں
            next_control_time += self.control_dt
            sleep_time = next_control_time - time.time()
            if sleep_time > 0:
                time.sleep(sleep_time)

    def actuator_thread_func(self):
        """ایکچوایٹر کمانڈ تھریڈ"""
        next_command_time = time.time()

        while self.running:
            start_time = time.time()

            # کنٹرول کمانڈ حاصل کریں
            try:
                control_command = self.control_queue.get_nowait()
            except queue.Empty:
                continue  # کمانڈ دستیاب نہ ہونے پر چھوڑ دیں

            # ایکچوایٹر پر لاگو کریں
            self.send_to_actuators(control_command)

            # ٹائم کو برقرار رکھیں
            next_command_time += self.actuator_dt
            sleep_time = next_command_time - time.time()
            if sleep_time > 0:
                time.sleep(sleep_time)

    def read_sensors(self):
        """سینسر ڈیٹا پڑھیں (مخصوص سسٹم کے ذریعہ نافذ کیا جائے گا)"""
        # مثال: رینڈم سینسر ڈیٹا لوٹائیں
        return np.random.randn(6)  # 6DOF سینسر ڈیٹا

    def compute_control(self, sensor_data):
        """سینسر ڈیٹا سے کنٹرول کمانڈ کمپیوٹ کریں"""
        # مثال: سادہ PD کنٹرول
        return sensor_data * 0.1

    def send_to_actuators(self, command):
        """کمانڈ ایکچوایٹر کو بھیجیں"""
        # مثال: ایکچوایٹر کمانڈ کی شبیہ سازی
        pass

    def start(self):
        """حقیقی وقت سسٹم شروع کریں"""
        self.running = True

        # تھریڈز شروع کریں
        self.sensor_thread = threading.Thread(target=self.sensor_thread_func, daemon=True)
        self.control_thread = threading.Thread(target=self.control_thread_func, daemon=True)
        self.actuator_thread = threading.Thread(target=self.actuator_thread_func, daemon=True)

        self.sensor_thread.start()
        self.control_thread.start()
        self.actuator_thread.start()

    def stop(self):
        """حقیقی وقت سسٹم بند کریں"""
        self.running = False

        # تھریڈز کو ختم ہونے کا انتظار کریں
        if self.sensor_thread:
            self.sensor_thread.join(timeout=1.0)
        if self.control_thread:
            self.control_thread.join(timeout=1.0)
        if self.actuator_thread:
            self.actuator_thread.join(timeout=1.0)
```

```python
class OnlineCalibrationSystem:
    def __init__(self, sensor_system):
        self.sensor_system = sensor_system
        self.calibration_data = deque(maxlen=1000)
        self.calibration_parameters = {}
        self.calibration_active = False

    def collect_calibration_data(self, known_reference, sensor_readings):
        """
        جانے والے حوالہ ویلیوز کے ساتھ کیلیبریشن ڈیٹا جمع کریں
        """
        calibration_sample = {
            'reference': known_reference,
            'readings': sensor_readings,
            'timestamp': time.time()
        }
        self.calibration_data.append(calibration_sample)

    def estimate_calibration_parameters(self):
        """
        جمع کردہ ڈیٹا کا استعمال کرتے ہوئے کیلیبریشن پیرامیٹر کا تخمینہ لگائیں
        """
        if len(self.calibration_data) < 10:  # کم از کم نمونے کی ضرورت ہے
            return False

        # ڈیٹا نکالیں
        references = np.array([sample['reference'] for sample in self.calibration_data])
        readings = np.array([sample['readings'] for sample in self.calibration_data])

        # لکیری کیلیبریشن: حوالہ = A * پڑھنا + b
        # حل: A = (R^T * R)^(-1) * R^T * حوالہ
        try:
            A = np.linalg.lstsq(readings, references, rcond=None)[0]
            b = np.mean(references - np.dot(readings, A), axis=0)

            self.calibration_parameters = {'A': A, 'b': b}
            return True
        except np.linalg.LinAlgError:
            return False

    def apply_calibration(self, raw_readings):
        """
        رو سینسر ریڈنگ پر کیلیبریشن لاگو کریں
        """
        if not self.calibration_parameters:
            return raw_readings  # کوئی کیلیبریشن دستیاب نہیں

        A = self.calibration_parameters['A']
        b = self.calibration_parameters['b']

        calibrated_readings = np.dot(raw_readings, A) + b
        return calibrated_readings

    def validation_metrics(self):
        """
        کیلیبریشن کوالٹی کے لیے توثیق کے معیار کا حساب لگائیں
        """
        if not self.calibration_parameters or len(self.calibration_data) < 10:
            return {}

        references = np.array([sample['reference'] for sample in self.calibration_data])
        readings = np.array([sample['readings'] for sample in self.calibration_data])

        # جمع کردہ ریڈنگ پر کیلیبریشن لاگو کریں
        calibrated = np.array([self.apply_calibration(r) for r in readings])

        # معیار کا حساب لگائیں
        error = references - calibrated
        rmse = np.sqrt(np.mean(error**2))
        max_error = np.max(np.abs(error))
        mean_error = np.mean(np.abs(error))

        return {
            'rmse': rmse,
            'max_error': max_error,
            'mean_error': mean_error,
            'calibration_samples': len(self.calibration_data)
        }
```
