## سینسنگ کے ریاضیاتی بنیادیات

### سینسر ماڈل اور عدم یقینی کی نمائندگی

سینسرز ان پروسیسز کو ماڈل کرتے ہیں جو جسمانی مقداروں کو قابل پیمائش سگنلز میں تبدیل کرتے ہیں، جس میں ضروری طور پر عدم یقینی شامل ہوتی ہے:

**y = h(x, w) + v**

جہاں:
- y: سینسر پیمائش
- x: ماحول کی حقیقی حالت
- w: ماحولیاتی پیرامیٹر
- v: پیمائش نوائز
- h: سینسر مشاہدہ ماڈل


### سینسر کارکردگی کے لیے کریمیر-راو کا کم از کم باؤنڈ

کریمیر-راو کا کم از کم باؤنڈ (CRLB) پیرامیٹر کے تخمینہ کی درستگی کی نظریاتی حد فراہم کرتا ہے:

**Cov(θ̂) ≥ I⁻¹(θ)**

جہاں I(θ) فشر انفارمیشن میٹرکس ہے۔

```python
class CramérRaoAnalysis:
    def __init__(self, sensor_parameters):
        self.params = sensor_parameters

    def fisher_information_matrix(self, state, measurement):
        """
        سینسر کنفیگریشن کے لیے فشر انفارمیشن میٹرکس کا حساب لگائیں
        """
        # لاگ لائکلی ہوڈ گریڈیئن کا حساب لگائیں
        def log_likelihood_gradient(params):
            # نفاذ مخصوص سینسر ماڈل پر منحصر ہے
            return np.zeros_like(params)

        # FIM کی عددی تقریب
        n_params = len(self.params)
        fim = np.zeros((n_params, n_params))

        epsilon = 1e-8
        for i in range(n_params):
            for j in range(n_params):
                # دوسری ماخوذہ تقریب
                params_plus_i = self.params.copy()
                params_plus_i[i] += epsilon
                grad_plus_i = log_likelihood_gradient(params_plus_i)

                params_minus_i = self.params.copy()
                params_minus_i[i] -= epsilon
                grad_minus_i = log_likelihood_gradient(params_minus_i)

                fim[i, j] = (grad_plus_i[j] - grad_minus_i[j]) / (2 * epsilon)

        return fim

    def theoretical_precision_limit(self, fim):
        """
        FIM سے نظریاتی درستی کی حد کا حساب لگائیں
        """
        try:
            covariance_bound = np.linalg.inv(fim)
            precision_limits = np.sqrt(np.diag(covariance_bound))
            return precision_limits
        except np.linalg.LinAlgError:
            # FIM مفرد ہے - پیرامیٹر قابل شناخت نہیں ہیں
            return np.full(len(self.params), np.inf)
```

### کووانٹم سینسرز

کووانٹم سینسرز کووانٹم مکینیکل خصوصیات کا استعمال انتہائی درست پیمائش کے لیے کرتے ہیں:

```python
class QuantumSensor:
    def __init__(self, quantum_state_dimension=2, coherence_time=1e-3):
        self.d = quantum_state_dimension  # کیوبٹ طول
        self.T2 = coherence_time          # کوہیرنس ٹائم
        self.state = np.zeros(self.d, dtype=complex)
        self.state[0] = 1.0  # زمینی حالت میں شروع کریں

    def prepare_state(self, parameters):
        """
        پیمائش کے لیے پیرامیٹر کے مطابق کووانٹم سینسر کی حالت تیار کریں
        """
        # پیرامیٹر-پر مبنی یونٹری ٹرانسفارمیشن لاگو کریں
        # یہ ایک کیوبٹ کے لیے ایک سادہ مثال ہے
        theta = parameters[0]  # پیمائش کے لیے پیرامیٹر
        U = np.array([
            [np.cos(theta/2), -np.sin(theta/2)],
            [np.sin(theta/2), np.cos(theta/2)]
        ])
        self.state = U @ self.state
        return self.state

    def interact_with_environment(self, field_strength, interaction_time):
        """
        پیمائش شدہ میدان کے ساتھ کووانٹم سینسر کی بات چیت
        """
        # ہیملٹونین ایволوشن: |ψ⟩ = exp(-iHt) |ψ₀⟩
        # مقناطیسی میدان کی پیمائش کے لیے: H = γB·σ (γ = گائیرو میگنیٹک ریشیو)
        gamma = 2.675e8  # پروٹون گائیرو میگنیٹک ریشیو (rad/s/T)
        H = gamma * field_strength * np.array([[1, 0], [0, -1]])  # σ_z سادگی کے لیے
        U_evolution = scipy.linalg.expm(-1j * H * interaction_time)
        self.state = U_evolution @ self.state
        return self.state

    def measure(self, observable=np.array([[1, 0], [0, -1]])):
        """
        کووانٹم پیمائش کریں
        """
        # ایکسپیکٹیشن ویلیو کا حساب لگائیں
        expectation = np.vdot(self.state, observable @ self.state)
        return np.real(expectation)

    def sensitivity(self):
        """
        کووانٹم سینسر حساسیت کا حساب لگائیں (ہائزینبرگ حد)
        """
        # ہائزینبرگ حد: Δθ ∝ 1/N (N = ذرات کی تعداد)
        # معیاری کووانٹم حد: Δθ ∝ 1/√N
        return 1.0 / np.sqrt(self.d)  # سادہ اسکیلنگ
```

### ایونٹ-بیسڈ ویژن سینسرز

ایونٹ-بیسڈ کیمرے غیر مسلسل، پتلا وقتی معلومات فراہم کرتے ہیں:

```python
class EventBasedCamera:
    def __init__(self, resolution=(640, 480), contrast_threshold=0.1):
        self.width, self.height = resolution
        self.threshold = contrast_threshold
        self.last_frame = np.zeros(resolution)
        self.event_queue = []

    def process_events(self, current_frame, timestamp):
        """
        فریم کو ایونٹس بنانے کے لیے پروسیس کریں
        """
        # شدت کی تبدیلیوں کا حساب لگائیں
        diff = current_frame - self.last_frame
        events = np.where(np.abs(diff) > self.threshold)

        # ایونٹ سٹریم بنائیں
        for y, x in zip(events[0], events[1]):
            polarity = 1 if diff[y, x] > 0 else -1
            event = {
                'x': x,
                'y': y,
                'timestamp': timestamp,
                'polarity': polarity
            }
            self.event_queue.append(event)

        self.last_frame = current_frame.copy()
        return self.event_queue

    def reconstruct_frame(self, event_window):
        """
        ایونٹ سٹریم سے فریم کی تعمیر کریں
        """
        frame = np.zeros((self.height, self.width))

        for event in event_window:
            x, y = int(event['x']), int(event['y'])
            if 0 <= x < self.width and 0 <= y < self.height:
                frame[y, x] += event['polarity'] * self.threshold

        return frame

    def temporal_resolution(self):
        """
        موثر وقتی ریزولوشن کا حساب لگائیں
        """
        # پکسل-لیول پروسیسنگ ٹائم کی طرف سے محدود
        return 1e-6  # 1 مائیکرو سیکنڈ فی ایونٹ (عام)
```

### سینسر نوائز کم کرنے کے لیے اڈاپٹو فلٹر

```python
class AdaptiveFilter:
    def __init__(self, filter_order=10, step_size=0.1):
        self.order = filter_order
        self.mu = step_size
        self.weights = np.zeros(filter_order)
        self.buffer = np.zeros(filter_order)

    def lms_filter(self, input_signal, desired_signal):
        """
        لیسٹ مین سکوئر اڈاپٹو فلٹر
        """
        output = np.zeros_like(input_signal)
        error = np.zeros_like(input_signal)

        for i in range(len(input_signal)):
            # فلٹر کا اطلاق
            self.buffer = np.roll(self.buffer, 1)
            self.buffer[0] = input_signal[i]

            y = np.dot(self.weights, self.buffer)
            output[i] = y

            # غلطی کا حساب لگائیں
            error[i] = desired_signal[i] - output[i]

            # وزن اپ ڈیٹ کریں
            self.weights += self.mu * error[i] * self.buffer

        return output, error

    def rls_filter(self, input_signal, desired_signal, forgetting_factor=0.98):
        """
        ریکریسیو لیسٹ سکوئر اڈاپٹو فلٹر
        """
        P = np.eye(self.order) / 0.1  # معکوس تعلق میٹرکس
        output = np.zeros_like(input_signal)
        error = np.zeros_like(input_signal)

        for i in range(len(input_signal)):
            # بفر اپ ڈیٹ کریں
            self.buffer = np.roll(self.buffer, 1)
            self.buffer[0] = input_signal[i]

            # گین کا حساب لگائیں
            k = P @ self.buffer / (forgetting_factor + self.buffer.T @ P @ self.buffer)

            # آؤٹ پٹ اور غلطی کا حساب لگائیں
            output[i] = np.dot(self.weights, self.buffer)
            error[i] = desired_signal[i] - output[i]

            # وزن اور معکوس تعلق اپ ڈیٹ کریں
            self.weights += k * error[i]
            P = (P - k[:, np.newaxis] * self.buffer * k.T) / forgetting_factor

        return output, error
```

```python
class CompressedSensing:
    def __init__(self, compression_ratio=0.5):
        self.M = int(compression_ratio * 1000)  # پیمائش میٹرکس قطاروں
        self.N = 1000  # اصل سگنل کی لمبائی
        self.sampling_matrix = np.random.randn(self.M, self.N)

    def compress_signal(self, signal):
        """
        بے ترتیب پروجیکشن کا استعمال کر کے سگنل کمپریس کریں
        """
        if len(signal) != self.N:
            raise ValueError(f"Signal length {len(signal)} doesn't match expected {self.N}")

        compressed = self.sampling_matrix @ signal
        return compressed

    def reconstruct_signal(self, compressed_signal, sparsity=50):
        """
        کمپریسڈ پیمائش سے پتلا سگنل دوبارہ تعمیر کریں
        """
        # حل کریں: min ||x||_1 subject to y = Ax
        # بیسز پرچی یا دیگر پتلا بازیابی الگورتھم کا استعمال کریں
        from scipy.optimize import minimize

        def objective(x):
            return np.sum(np.abs(x))  # L1 norm

        def constraint(x):
            return self.sampling_matrix @ x - compressed_signal

        # ابتدائی اندازہ
        x0 = np.random.randn(self.N) * 0.01

        # اصلاح کا مسئلہ حل کریں
        result = minimize(objective, x0, method='SLSQP',
                         constraints={'type': 'eq', 'fun': constraint})

        return result.x
```

### ملٹی-ٹارگٹ ٹریکنگ اور ڈیٹا اسوسی ایشن

```python
class MultiTargetTracker:
    def __init__(self, num_targets=10):
        self.num_targets = num_targets
        self.trackers = [KalmanFilter(state_dim=4, measurement_dim=2) for _ in range(num_targets)]
        self.target_states = [None] * num_targets
        self.existence_probabilities = np.zeros(num_targets)
        self.association_matrix = None

    def joint_probabilistic_data_association(self, measurements):
        """
        JPDA برائے ملٹی-ٹارگٹ ٹریکنگ
        """
        # ہر پیمائش-ٹریک جوڑی کے لیے اسوسی ایشن امکان کا حساب لگائیں
        num_measurements = len(measurements)
        num_tracks = len(self.target_states)

        # اسوسی ایشن امکان میٹرکس
        P_assoc = np.zeros((num_measurements, num_tracks))

        for i, meas in enumerate(measurements):
            for j, track in enumerate(self.target_states):
                if track is not None:
                    # ٹریک سے منسلک ہونے کا امکان کا حساب لگائیں
                    innovation = meas - self.trackers[j].H @ track
                    S = self.trackers[j].H @ self.trackers[j].P @ self.trackers[j].H.T + self.trackers[j].R
                    likelihood = scipy.stats.multivariate_normal.pdf(innovation, cov=S)
                    P_assoc[i, j] = likelihood * self.existence_probabilities[j]

        # اسوسی ایشن امکان کو نارملائز کریں
```

### تقسیم شدہ سینسر نیٹ ورکس

```python
class DistributedSensorNetwork:
    def __init__(self, sensor_nodes, communication_topology):
        self.nodes = sensor_nodes
        self.topology = communication_topology
        self.consensus_matrix = self.compute_consensus_matrix()
        self.fusion_algorithm = 'average_consensus'

    def consensus_based_fusion(self, local_estimates):
        """
        کنسسوس الگورتھم کا استعمال کر کے تقسیم شدہ فیوژن
        """
        # ہر نوڈ اپنے اندازے کو ہمسائیوں کی بنیاد پر اپ ڈیٹ کرتا ہے
        new_estimates = np.copy(local_estimates)

        for i, node in enumerate(self.nodes):
            neighbors = np.where(self.topology[i, :] > 0)[0]
            neighbor_estimates = local_estimates[neighbors]

            # کنسسوس اپ ڈیٹ: x_i(k+1) = x_i(k) + Σ(x_j(k) - x_i(k))
            consensus_term = np.mean(neighbor_estimates) - local_estimates[i]
            new_estimates[i] = local_estimates[i] + 0.1 * consensus_term  # قدم کا سائز = 0.1

        return new_estimates

    def gossip_based_fusion(self, initial_values):
        """
        گوسپ-بیسڈ تقسیم شدہ فیوژن
        """
        values = np.copy(initial_values)
        n_nodes = len(values)

        # بے ترتیب نوڈ جوڑی گوسپ
        for iteration in range(1000):
            # بے ترتیب طور پر دو منسلک نوڈس منتخب کریں
            i, j = np.random.choice(n_nodes, 2, replace=False)
            if self.topology[i, j] > 0:  # نوڈس منسلک ہیں
                # ان کی ویلیو اوسط کریں
                avg = (values[i] + values[j]) / 2
                values[i] = avg
                values[j] = avg

        return values
```

### سینسر کارکردگی کے معیار

```python
class SensorPerformanceEvaluator:
    def __init__(self):
        self.metrics = {
            'accuracy': [],
            'precision': [],
            'recall': [],
            'f1_score': [],
            'rmse': [],
            'mae': [],
            'timing': []
        }

    def evaluate_detection_performance(self, ground_truth, predictions, iou_threshold=0.5):
        """
        اشیاء کی شناخت کی کارکردگی کا جائزہ لیں
        """
        # صحیح مثبت، غلط مثبت، غلط منفی کا حساب لگائیں
        tp = 0
        fp = 0
        fn = 0

        matched_gt = set()

        for pred in predictions:
            best_iou = 0
            best_gt_idx = -1

            for gt_idx, gt in enumerate(ground_truth):
                if gt_idx in matched_gt:
                    continue

                iou = self.calculate_iou(pred['bbox'], gt['bbox'])
                if iou > best_iou and iou > iou_threshold:
                    best_iou = iou
                    best_gt_idx = gt_idx

            if best_gt_idx != -1:
                tp += 1
                matched_gt.add(best_gt_idx)
            else:
                fp += 1

        fn = len(ground_truth) - len(matched_gt)

        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        f1_score = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

        return {
            'precision': precision,
            'recall': recall,
            'f1_score': f1_score,
            'tp': tp,
            'fp': fp,
            'fn': fn
        }

    def calculate_iou(self, bbox1, bbox2):
        """
        یونین پر انٹرسیکشن کا حساب لگائیں
        """
        x1, y1, x2, y2 = bbox1
        x3, y3, x4, y4 = bbox2

        # انٹرسیکشن ایریا کا حساب لگائیں
        xi1, yi1 = max(x1, x3), max(y1, y3)
        xi2, yi2 = min(x2, x4), min(y2, y4)

        if xi2 <= xi1 or yi2 <= yi1:
            return 0

        intersection = (xi2 - xi1) * (yi2 - yi1)

        # یونین ایریا کا حساب لگائیں
        area1 = (x2 - x1) * (y2 - y1)
        area2 = (x4 - x3) * (y4 - y3)
        union = area1 + area2 - intersection

        return intersection / union

    def timing_analysis(self, sensor_data):
        """
        سینسر ٹائمنگ کارکردگی کا تجزیہ کریں
        """
        timestamps = sensor_data.get('timestamps', [])
        if len(timestamps) < 2:
            return {}

        intervals = np.diff(timestamps)
        frequency = 1.0 / np.mean(intervals)

        return {
            'average_frequency': frequency,
            'timing_jitter': np.std(intervals),
            'min_interval': np.min(intervals),
            'max_interval': np.max(intervals)
        }
```
