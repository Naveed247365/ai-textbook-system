## Mathematical Foundations and Optimization

### Geometric Computer Vision

Geometric computer vision forms the mathematical backbone of robotic vision systems, dealing with the geometry of multiple views and 3D reconstruction.

#### Epipolar Geometry and Fundamental Matrix

The epipolar geometry describes the geometric relationship between two images of the same scene taken from different viewpoints:

**x₂^T F x₁ = 0**

Where x₁ and x₂ are corresponding points in two images, and F is the fundamental matrix.

```python
class EpipolarGeometry:
    def __init__(self):
        self.fundamental_matrix = None
        self.essential_matrix = None
        self.camera_matrices = []

    def compute_fundamental_matrix(self, points1, points2):
        """
        Compute fundamental matrix using 8-point algorithm
        """
        # Normalize points for numerical stability
        points1_norm, T1 = self.normalize_points(points1)
        points2_norm, T2 = self.normalize_points(points2)

        # Build constraint matrix
        A = np.zeros((len(points1_norm), 9))
        for i in range(len(points1_norm)):
            x1, y1 = points1_norm[i]
            x2, y2 = points2_norm[i]
            A[i] = [x2*x1, x2*y1, x2, y2*x1, y2*y1, y2, x1, y1, 1]

        # Solve for fundamental matrix using SVD
        U, S, Vt = np.linalg.svd(A)
        F = Vt[-1].reshape(3, 3)

        # Enforce rank 2 constraint
        U_F, S_F, Vt_F = np.linalg.svd(F)
        S_F[2] = 0  # Set smallest singular value to 0
        F = U_F @ np.diag(S_F) @ Vt_F

        # Denormalize
        F = T2.T @ F @ T1

        # Normalize
        F = F / F[2, 2]

        self.fundamental_matrix = F
        return F

    def normalize_points(self, points):
        """
        Normalize points for numerical stability
        """
        centroid = np.mean(points, axis=0)
        centered = points - centroid
        mean_dist = np.mean(np.sqrt(np.sum(centered**2, axis=1)))
        scale = np.sqrt(2) / mean_dist

        T = np.array([
            [scale, 0, -scale * centroid[0]],
            [0, scale, -scale * centroid[1]],
            [0, 0, 1]
        ])

        normalized_points = np.hstack([points, np.ones((len(points), 1))])
        normalized_points = (T @ normalized_points.T).T
        normalized_points = normalized_points[:, :2] / normalized_points[:, 2:3]

        return normalized_points, T

    def compute_essential_matrix(self, points1, points2, K1, K2):
        """
        Compute essential matrix from fundamental matrix
        E = K2^T * F * K1
        """
        F = self.compute_fundamental_matrix(points1, points2)
        E = K2.T @ F @ K1
        self.essential_matrix = E
        return E

    def triangulate_points(self, points1, points2, P1, P2):
        """
        Triangulate 3D points from 2D correspondences
        """
        points_3d = []
        for pt1, pt2 in zip(points1, points2):
            # Build matrix for triangulation
            A = np.array([
                pt1[0] * P1[2, :] - P1[0, :],
                pt1[1] * P1[2, :] - P1[1, :],
                pt2[0] * P2[2, :] - P2[0, :],
                pt2[1] * P2[2, :] - P2[1, :]
            ])

            # Solve using SVD
            U, S, Vt = np.linalg.svd(A)
            X = Vt[-1]
            X = X / X[3]  # Normalize by scale factor
            points_3d.append(X[:3])

        return np.array(points_3d)
```

**x₂ = H x₁**

### Convex Optimization for Vision Problems

Many computer vision problems can be formulated as convex optimization problems:

```python
import cvxpy as cp

class VisionOptimization:
    def __init__(self):
        pass

    def robust_regression(self, X, y, epsilon=1e-6):
        """
        Robust regression using Huber loss
        """
        n, d = X.shape
        beta = cp.Variable(d)
        residuals = X @ beta - y

        # Huber loss for robust regression
        huber_loss = cp.sum(cp.huber(residuals, M=1.0))

        # Formulate and solve optimization problem
        problem = cp.Problem(cp.Minimize(huber_loss))
        problem.solve()

        return beta.value

    def total_variation_denoising(self, image, lambda_reg=0.1):
        """
        Total variation denoising
        """
        m, n = image.shape
        X = cp.Variable((m, n))

        # Data fidelity term
        data_term = cp.sum_squares(X - image)

        # Total variation regularization
        tv_term = 0
        for i in range(m-1):
            for j in range(n-1):
                tv_term += cp.norm(X[i, j] - X[i+1, j], 2) + cp.norm(X[i, j] - X[i, j+1], 2)

        # Formulate optimization problem
        objective = cp.Minimize(data_term + lambda_reg * tv_term)
        constraints = [X >= 0, X <= 255]  # Pixel value bounds

        problem = cp.Problem(objective, constraints)
        problem.solve()

        return X.value

    def dictionary_learning(self, data, k, lambda_reg=0.1):
        """
        Dictionary learning for sparse representation
        """
        n, d = data.shape  # n samples, d dimensions
        D = cp.Variable((d, k))  # Dictionary
        X = cp.Variable((k, n))  # Sparse coefficients

        # Reconstruction error
        recon_error = cp.sum_squares(data.T - D @ X)

        # Sparsity promoting term
        sparsity = cp.norm1(X)

        # Dictionary constraints (unit norm columns)
        constraints = [cp.norm(D[:, i], 2) <= 1 for i in range(k)]

        # Formulate optimization problem
        objective = cp.Minimize(recon_error + lambda_reg * sparsity)
        problem = cp.Problem(objective, constraints)
        problem.solve()

        return D.value, X.value
```

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class SuperPointNet(nn.Module):
    """
    SuperPoint: Self-Supervised Interest Point Detection and Description
    """
    def __init__(self, descriptor_dim=256):
        super(SuperPointNet, self).__init__()
        self.descriptor_dim = descriptor_dim

        # Shared encoder
        self.encoder = nn.Sequential(
            nn.Conv2d(1, 64, 3, stride=2, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(64, 64, 3, stride=2, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(64, 128, 3, stride=1, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(128, 128, 3, stride=1, padding=1),
            nn.ReLU(inplace=True)
        )

        # Detector head
        self.detector_head = nn.Sequential(
            nn.Conv2d(128, 256, 1),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 65, 1),  # 65 = 64 descriptors + 1 for detector
        )

        # Descriptor head
        self.descriptor_head = nn.Sequential(
            nn.Conv2d(128, 256, 1),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, descriptor_dim, 1),
        )

    def forward(self, x):
        features = self.encoder(x)

        # Detection
        detection_output = self.detector_head(features)
        # Split into detector and descriptor parts
        detector_output = detection_output[:, :-1, :, :]  # 64 channels for descriptors
        descriptor_output = self.descriptor_head(features)

        # Apply softmax to detector output to get interest point probabilities
        Hc, Wc = detector_output.shape[-2:]
        probs = F.softmax(detector_output.view(-1, 64, Hc*Wc), dim=-1).view(-1, 64, Hc, Wc)

        # L2 normalize descriptors
        descriptors = F.normalize(descriptor_output, p=2, dim=1)

        return probs, descriptors

class DeepFeatureMatcher:
    def __init__(self, model_path=None):
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.model = SuperPointNet().to(self.device)
        if model_path:
            self.model.load_state_dict(torch.load(model_path))
        self.model.eval()

    def extract_features(self, image):
        """
        Extract deep features from image
        """
        # Preprocess image
        if len(image.shape) == 3:
            gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        else:
            gray = image

        # Normalize to [0, 1]
        gray = gray.astype(np.float32) / 255.0

        # Convert to tensor and add batch dimension
        tensor = torch.from_numpy(gray).unsqueeze(0).unsqueeze(0).to(self.device)

        with torch.no_grad():
            probs, descriptors = self.model(tensor)

        # Process outputs
        interest_points = self.extract_interest_points(probs.cpu().numpy()[0])
        descriptors = descriptors.cpu().numpy()[0]

        return interest_points, descriptors

    def extract_interest_points(self, prob_maps, threshold=0.015):
        """
        Extract interest points from probability maps
        """
        H, W = prob_maps.shape[1], prob_maps.shape[2]
        interest_points = []

        for i in range(H):
            for j in range(W):
                for k in range(64):  # 64 different probability maps
                    if prob_maps[k, i, j] > threshold:
                        # Convert grid coordinates to image coordinates
                        x = j * 8 + 4  # Assuming 8x upsampling
                        y = i * 8 + 4
                        interest_points.append([x, y, prob_maps[k, i, j]])

        return np.array(interest_points)

    def match_features(self, desc1, desc2, ratio_threshold=0.8):
        """
        Match features using nearest neighbor search
        """
        # Reshape descriptors
        desc1_flat = desc1.reshape(desc1.shape[0], -1).T  # Shape: (num_features, dim)
        desc2_flat = desc2.reshape(desc2.shape[0], -1).T

        # Compute cosine similarity
        similarities = desc1_flat @ desc2_flat.T
        norms1 = np.linalg.norm(desc1_flat, axis=1, keepdims=True)
        norms2 = np.linalg.norm(desc2_flat, axis=1, keepdims=True)
        similarities = similarities / (norms1 @ norms2.T)

        # Find best matches
        matches = []
        for i in range(similarities.shape[0]):
            sim_row = similarities[i]
            best_idx = np.argmax(sim_row)
            best_sim = sim_row[best_idx]

            # Find second best match
            second_best_idx = np.argsort(sim_row)[-2]
            second_best_sim = sim_row[second_best_idx]

            # Apply ratio test
            if best_sim > 0 and second_best_sim / best_sim < ratio_threshold:
                matches.append((i, best_idx, best_sim))

        return matches
```

### Multi-Scale Feature Detection

```python
class MultiScaleFeatureDetector:
    def __init__(self, num_octaves=4, num_scales=3, sigma=1.6):
        self.num_octaves = num_octaves
        self.num_scales = num_scales
        self.sigma = sigma
        self.scale_factor = 2**(1.0/num_scales)

    def build_gaussian_pyramid(self, image):
        """
        Build Gaussian pyramid for multi-scale analysis
        """
        pyramid = []
        current = image.copy().astype(np.float32)

        for octave in range(self.num_octaves):
            octave_images = []
            sigma_prev = 0.5  # Starting sigma
            for scale in range(self.num_scales + 3):  # +3 for DoG computation
                # Calculate sigma for this level
                k = self.scale_factor ** scale
                sigma_cur = self.sigma * k

                # Calculate additional sigma needed
                sigma_add = np.sqrt(sigma_cur**2 - sigma_prev**2) if sigma_cur > sigma_prev else 0

                # Apply Gaussian blur
                if sigma_add > 0:
                    blurred = cv2.GaussianBlur(current, (0, 0), sigma_add)
                else:
                    blurred = current.copy()

                octave_images.append(blurred)
                sigma_prev = sigma_cur

            pyramid.append(octave_images)
            # Downsample for next octave
            current = octave_images[-1][::2, ::2]

        return pyramid

    def build_dog_pyramid(self, gaussian_pyramid):
        """
        Build Difference of Gaussians pyramid
        """
        dog_pyramid = []
        for octave_images in gaussian_pyramid:
            dog_octave = []
            for i in range(len(octave_images) - 1):
                dog = octave_images[i+1] - octave_images[i]
                dog_octave.append(dog)
            dog_pyramid.append(dog_octave)

        return dog_pyramid

    def detect_keypoints_dog(self, dog_pyramid):
        """
        Detect keypoints using DoG extrema
        """
        keypoints = []
        for octave_idx, dog_octave in enumerate(dog_pyramid):
            for scale_idx in range(1, len(dog_octave) - 1):  # Exclude borders
                # Current scale image
                curr = dog_octave[scale_idx]
                prev = dog_octave[scale_idx - 1]
                next_scale = dog_octave[scale_idx + 1]

                # Check 3x3x3 neighborhood for extrema
                h, w = curr.shape
                for i in range(1, h-1):
                    for j in range(1, w-1):
                        # Get 3x3 patches from all three scales
                        patch_prev = prev[i-1:i+2, j-1:j+2]
                        patch_curr = curr[i-1:i+2, j-1:j+2]
                        patch_next = next_scale[i-1:i+2, j-1:j+2]

                        center_value = curr[i, j]

                        # Check if center is local maximum or minimum
                        all_values = np.concatenate([
                            patch_prev.flatten(),
                            patch_curr.flatten(),
                            patch_next.flatten()
                        ])

                        # Remove center value from comparison
                        all_others = all_values[all_values != center_value]

                        is_maxima = np.all(center_value >= all_others)
                        is_minima = np.all(center_value <= all_others)

                        if is_maxima or is_minima:
                            # Subpixel localization
                            subpix_coords = self.subpixel_localization(
                                dog_pyramid, octave_idx, scale_idx, i, j
                            )
                            if subpix_coords is not None:
                                x, y, scale = subpix_coords
                                # Convert to original image coordinates
                                x_orig = x * (2**octave_idx)
                                y_orig = y * (2**octave_idx)
                                keypoints.append((x_orig, y_orig, scale))

        return keypoints

    def subpixel_localization(self, dog_pyramid, octave, scale, i, j):
        """
        Refine keypoint location to subpixel accuracy
        """
        # This would implement Taylor expansion-based subpixel refinement
        # For brevity, returning the integer coordinates
```

```python
class SSD(nn.Module):
    def __init__(self, num_classes=80, backbone='vgg16'):
        super(SSD, self).__init__()
        self.num_classes = num_classes
        self.backbone = backbone

        # Feature extraction layers (simplified)
        if backbone == 'vgg16':
            self.features = self._make_vgg_layers()
        else:
            raise NotImplementedError(f"Backbone {backbone} not implemented")

        # Additional feature layers for multi-scale detection
        self.extra_layers = self._make_extra_layers()

        # Location and confidence prediction heads
        self.loc_heads = nn.ModuleList()
        self.conf_heads = nn.ModuleList()

        # Number of predictions per feature map location
        num_defaults = [4, 6, 6, 6, 4, 4]  # Different aspect ratios for each scale

        # Feature map channels for each scale
        feature_channels = [512, 1024, 512, 256, 256, 256]

        for i, (num_default, channels) in enumerate(zip(num_defaults, feature_channels)):
            loc_layers = []
            conf_layers = []

            # Location prediction (4 values: dx, dy, dw, dh)
            loc_layers += [nn.Conv2d(channels, num_default * 4, kernel_size=3, padding=1)]
            # Confidence prediction (num_classes values)
            conf_layers += [nn.Conv2d(channels, num_default * num_classes, kernel_size=3, padding=1)]

            self.loc_heads.append(nn.Sequential(*loc_layers))
            self.conf_heads.append(nn.Sequential(*conf_layers))

    def _make_vgg_layers(self):
        """
        Create VGG-style feature extraction layers
        """
        layers = []
        cfg = [64, 64, 'M', 128, 128, 'M', 256, 256, 256, 'M', 512, 512, 512, 'M', 512, 512, 512]

        in_channels = 3
        for v in cfg:
            if v == 'M':
                layers += [nn.MaxPool2d(kernel_size=2, stride=2)]
            else:
                conv2d = nn.Conv2d(in_channels, v, kernel_size=3, padding=1)
                layers += [conv2d, nn.ReLU(inplace=True)]
                in_channels = v

        # Add additional layers for higher-level features
        layers += [
            nn.MaxPool2d(kernel_size=3, stride=1, padding=1),
            nn.Conv2d(512, 1024, kernel_size=3, padding=6, dilation=6),
            nn.ReLU(inplace=True),
            nn.Conv2d(1024, 1024, kernel_size=1),
            nn.ReLU(inplace=True)
        ]

        return nn.Sequential(*layers)

    def _make_extra_layers(self):
        """
        Create additional layers for multi-scale feature extraction
        """
        layers = []
        in_channels = 1024

        # Additional feature extraction layers
        for v in [256, 512, 128, 256, 128, 256, 128, 256]:
            if v == 256:
                layers += [
                    nn.Conv2d(in_channels, v, kernel_size=1, stride=1),
                    nn.ReLU(inplace=True),
                    nn.Conv2d(v, v*2, kernel_size=3, stride=2, padding=1),
                    nn.ReLU(inplace=True)
                ]
                in_channels = v*2
            else:
                layers += [
                    nn.Conv2d(in_channels, v, kernel_size=1, stride=1),
                    nn.ReLU(inplace=True),
                    nn.Conv2d(v, v*2, kernel_size=3, stride=1),
                    nn.ReLU(inplace=True)
                ]
                in_channels = v*2

        return nn.Sequential(*layers)

    def forward(self, x):
        """
        Forward pass of SSD
        """
        # Extract features through backbone
        features = []

        # Get intermediate features from VGG backbone
        for i, layer in enumerate(self.features):
            x = layer(x)
            if i in [22, 34]:  # After specific layers in VGG
                features.append(x)

        # Continue with extra layers
        for layer in self.extra_layers:
            x = layer(x)
            features.append(x)

        # Apply prediction heads
        loc_preds = []
        conf_preds = []

        for (x, l, c) in zip(features, self.loc_heads, self.conf_heads):
            loc_pred = l(x)
            conf_pred = c(x)

            # Reshape predictions
            batch_size = x.size(0)
            h, w = x.size()[2:4]
            num_priors = loc_pred.size(1) // 4

            loc_pred = loc_pred.permute(0, 2, 3, 1).contiguous()
```

### Vision Transformers for Object Detection

```python
import torch.nn.functional as F

class VisionTransformer(nn.Module):
    def __init__(self, image_size=224, patch_size=16, num_layers=12, num_heads=12,
                 hidden_dim=768, mlp_ratio=4, num_classes=1000):
        super(VisionTransformer, self).__init__()

        self.image_size = image_size
        self.patch_size = patch_size
        self.num_patches = (image_size // patch_size) ** 2
        self.hidden_dim = hidden_dim

        # Patch embedding
        self.patch_embed = nn.Conv2d(3, hidden_dim, kernel_size=patch_size,
                                    stride=patch_size, bias=False)

        # Class token and positional embedding
        self.cls_token = nn.Parameter(torch.zeros(1, 1, hidden_dim))
        self.pos_embed = nn.Parameter(torch.zeros(1, self.num_patches + 1, hidden_dim))

        # Transformer layers
        self.layers = nn.ModuleList([
            TransformerBlock(hidden_dim, num_heads, mlp_ratio)
            for _ in range(num_layers)
        ])

        # Final normalization
        self.norm = nn.LayerNorm(hidden_dim)

        # Classification head
        self.head = nn.Linear(hidden_dim, num_classes) if num_classes > 0 else nn.Identity()

        # Initialize weights
        self.apply(self._init_weights)

    def _init_weights(self, m):
        if isinstance(m, nn.Linear):
            torch.nn.init.trunc_normal_(m.weight, std=0.02)
            if m.bias is not None:
                nn.init.constant_(m.bias, 0)
        elif isinstance(m, nn.LayerNorm):
            nn.init.constant_(m.bias, 0)
            nn.init.constant_(m.weight, 1.0)

    def forward(self, x):
        B, C, H, W = x.shape

        # Patch embedding
        x = self.patch_embed(x)  # (B, hidden_dim, grid_h, grid_w)
        x = x.flatten(2).transpose(1, 2)  # (B, num_patches, hidden_dim)

        # Add class token
        cls_tokens = self.cls_token.expand(B, -1, -1)
        x = torch.cat([cls_tokens, x], dim=1)  # (B, num_patches + 1, hidden_dim)

        # Add positional embedding
        x = x + self.pos_embed

        # Apply transformer layers
        for layer in self.layers:
            x = layer(x)

        # Final normalization
        x = self.norm(x)

        # Classification output (using class token)
        cls_output = self.head(x[:, 0])

        return cls_output

class TransformerBlock(nn.Module):
    def __init__(self, hidden_dim, num_heads, mlp_ratio=4):
        super(TransformerBlock, self).__init__()

        self.norm1 = nn.LayerNorm(hidden_dim)
        self.attn = nn.MultiheadAttention(hidden_dim, num_heads, batch_first=True)
        self.norm2 = nn.LayerNorm(hidden_dim)

        mlp_hidden_dim = int(hidden_dim * mlp_ratio)
        self.mlp = nn.Sequential(
            nn.Linear(hidden_dim, mlp_hidden_dim),
            nn.GELU(),
            nn.Linear(mlp_hidden_dim, hidden_dim)
        )

    def forward(self, x):
        # Self-attention
        attn_out, _ = self.attn(self.norm1(x), self.norm1(x), self.norm1(x))
        x = x + attn_out

        # MLP
        mlp_out = self.mlp(self.norm2(x))
        x = x + mlp_out

        return x
```

```python
class StructureFromMotion:
    def __init__(self):

        self.extrinsics = []

    def initialize_camera_intrinsics(self, image_size, focal_length, principal_point=None):
        """
        Initialize camera intrinsic parameters
        """

            cx, cy = image_size[0] / 2, image_size[1] / 2
        else:
            cx, cy = principal_point

        self.camera_intrinsics = np.array([
            [focal_length, 0, cx],
            [0, focal_length, cy],
            [0, 0, 1]
        ])

    def find_feature_matches(self, img1, img2, detector='sift'):
        """
        Find feature matches between two images
        """
        if detector == 'sift':
            sift = cv2.SIFT_create()
            kp1, desc1 = sift.detectAndCompute(img1, None)
            kp2, desc2 = sift.detectAndCompute(img2, None)
        elif detector == 'orb':
            orb = cv2.ORB_create()
            kp1, desc1 = orb.detectAndCompute(img1, None)
            kp2, desc2 = orb.detectAndCompute(img2, None)
        else:
            raise ValueError(f"Unknown detector: {detector}")

        if desc1 is None or desc2 is None:
            return [], [], []

        # Match features
        bf = cv2.BFMatcher()
        matches = bf.knnMatch(desc1, desc2, k=2)

        # Apply Lowe's ratio test
        good_matches = []
        for match_pair in matches:
            if len(match_pair) == 2:
                m, n = match_pair
                if m.distance < 0.75 * n.distance:
                    good_matches.append(m)

        # Extract matching points
        pts1 = np.float32([kp1[m.queryIdx].pt for m in good_matches]).reshape(-1, 2)
        pts2 = np.float32([kp2[m.trainIdx].pt for m in good_matches]).reshape(-1, 2)

        return pts1, pts2, good_matches

    def estimate_fundamental_matrix(self, pts1, pts2):
        """
        Estimate fundamental matrix using RANSAC
        """
        F, mask = cv2.findFundamentalMat(pts1, pts2, cv2.RANSAC, 4, 0.999)
        F = F / F[2, 2]  # Normalize
        return F, mask

    def estimate_essential_matrix(self, pts1, pts2):
        """
        Estimate essential matrix from fundamental matrix
        """
        F, _ = self.estimate_fundamental_matrix(pts1, pts2)
        E = self.camera_intrinsics.T @ F @ self.camera_intrinsics
        return E

    def recover_pose(self, E, pts1, pts2):
        """
        Recover relative pose from essential matrix
        """
        points, R, t, mask = cv2.recoverPose(E, pts1, pts2, self.camera_intrinsics)

        # Store the recovered pose
        pose = np.eye(4)
        pose[:3, :3] = R
        pose[:3, 3] = t.flatten()

        return R, t, pose, mask

    def triangulate_points(self, pts1, pts2, P1, P2):
        """
        Triangulate 3D points from stereo correspondences
        """
        points_3d = cv2.triangulatePoints(P1, P2, pts1.T, pts2.T)
        points_3d = points_3d / points_3d[3, :]  # Normalize by scale
        points_3d = points_3d[:3, :].T  # Transpose to (N, 3)

        return points_3d

    def bundle_adjustment(self, points_3d, points_2d_list, camera_poses,
                         camera_intrinsics, max_iterations=20):
        """
        Bundle adjustment optimization
        """
        from scipy.optimize import least_squares

        def reprojection_error(params, points_3d, points_2d_list,
                              camera_intrinsics, n_cameras):
            """
            Reprojection error function for bundle adjustment
            """
            # Extract camera parameters
            camera_params = params[:n_cameras * 6].reshape(n_cameras, 6)
            points_3d_flat = params[n_cameras * 6:].reshape(-1, 3)

            errors = []
            point_idx = 0

            for cam_idx, points_2d in enumerate(points_2d_list):
                # Convert Rodrigues vector to rotation matrix
                rvec = camera_params[cam_idx, :3]
                tvec = camera_params[cam_idx, 3:]

                R, _ = cv2.Rodrigues(rvec)
                P = np.hstack([R, tvec.reshape(3, 1)])
                P = camera_intrinsics @ P

                # Project 3D points to 2D
                projected_points = cv2.projectPoints(
                    points_3d_flat[point_idx:point_idx+len(points_2d)],
                    rvec, tvec, camera_intrinsics[:2, :2], None
                )[0].squeeze()

                # Calculate reprojection error
                errors.extend((projected_points - points_2d).flatten())
                point_idx += len(points_2d)

            return np.array(errors)

        # Initial parameters
        n_cameras = len(camera_poses)
        n_points = len(points_3d)

        # Convert camera poses to parameter format [rvec, tvec] for each camera
        initial_params = []
        for pose in camera_poses:
            R = pose[:3, :3]
            t = pose[:3, 3]
            rvec, _ = cv2.Rodrigues(R)
            initial_params.extend(rvec.flatten())
            initial_params.extend(t)

        # Add 3D points
        initial_params.extend(points_3d.flatten())
        initial_params = np.array(initial_params)

        # Optimize
        result = least_squares(
            reprojection_error,
            initial_params,
            args=(points_3d, points_2d_list, camera_intrinsics, n_cameras),
            method='lm',
            ftol=1e-10
        )

        # Extract optimized parameters
        optimized_camera_params = result.x[:n_cameras * 6].reshape(n_cameras, 6)
        optimized_points_3d = result.x[n_cameras * 6:].reshape(n_points, 3)

        return optimized_points_3d, optimized_camera_params
```

```python
import threading
import queue

class MultiThreadedVisionPipeline:
    def __init__(self, num_threads=4):
        self.num_threads = num_threads
        self.input_queue = queue.Queue(maxsize=10)
        self.output_queue = queue.Queue(maxsize=10)
        self.result_queue = queue.Queue(maxsize=10)
        self.threads = []
        self.running = False

    def start_pipeline(self):
        """
        Start the multi-threaded vision pipeline
        """
        self.running = True

        # Start worker threads
        for i in range(self.num_threads):
            thread = threading.Thread(target=self._worker, args=(i,))
            thread.daemon = True
            thread.start()
            self.threads.append(thread)

        # Start processing thread
        processing_thread = threading.Thread(target=self._processing_loop)
        processing_thread.daemon = True
        processing_thread.start()
        self.threads.append(processing_thread)

    def stop_pipeline(self):
        """
        Stop the multi-threaded pipeline
        """
        self.running = False

        # Wait for threads to finish
        for thread in self.threads:
            thread.join(timeout=1.0)

    def _worker(self, worker_id):
        """
        Worker thread function
        """
        while self.running:
            try:
                # Get work item from queue
                work_item = self.input_queue.get(timeout=0.1)

                # Process the work item
                result = self._process_work_item(work_item)

                # Put result in output queue
                self.output_queue.put((work_item['id'], result))

                # Mark task as done
                self.input_queue.task_done()

            except queue.Empty:
                continue
            except Exception as e:
                print(f"Worker {worker_id} error: {e}")

                # Collect results
```

```python
            classes.add(gt['class'])
        for pred in predictions:
            classes.add(pred['class'])

        class_metrics = {}

        for class_id in classes:
            # Get ground truth and predictions for this class
            gt_class = [gt for gt in ground_truth if gt['class'] == class_id]
            pred_class = [pred for pred in predictions if pred['class'] == class_id]

            # Calculate metrics for this class
            tp, fp, fn = self._calculate_detection_counts(gt_class, pred_class, iou_threshold)

            precision = tp / (tp + fp) if (tp + fp) > 0 else 0
            recall = tp / (tp + fn) if (tp + fn) > 0 else 0
            f1_score = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

            class_metrics[class_id] = {
                'precision': precision,
                'recall': recall,
                'f1_score': f1_score,
                'tp': tp,
                'fp': fp,
                'fn': fn
            }

        # Calculate mean Average Precision (mAP)
        mAP = np.mean([metrics['precision'] for metrics in class_metrics.values()])

        return {
            'class_metrics': class_metrics,
            'mAP': mAP,
            'overall_precision': np.mean([m['precision'] for m in class_metrics.values()]),
            'overall_recall': np.mean([m['recall'] for m in class_metrics.values()]),
            'overall_f1': np.mean([m['f1_score'] for m in class_metrics.values()])
        }

    def _calculate_detection_counts(self, gt_objects, pred_objects, iou_threshold):
        """
        Calculate True Positives, False Positives, False Negatives
        """
        # Mark which ground truth objects have been matched
        gt_matched = [False] * len(gt_objects)

        tp = 0  # True Positives
        fp = 0  # False Positives

        # For each prediction, find best matching ground truth
        for pred in pred_objects:
            best_iou = 0
            best_gt_idx = -1

            # Find best matching ground truth object
            for gt_idx, gt in enumerate(gt_objects):
                if gt_matched[gt_idx]:
                    continue  # Already matched

                iou = self._calculate_iou(pred['bbox'], gt['bbox'])
                if iou > best_iou:
                    best_iou = iou
                    best_gt_idx = gt_idx

            if best_iou >= iou_threshold:
                # True positive - correct detection
                tp += 1
                gt_matched[best_gt_idx] = True
            else:
                # False positive - incorrect detection
                fp += 1

        # False negatives - ground truth objects that weren't detected
        fn = len(gt_objects) - sum(gt_matched)

        return tp, fp, fn

    def _calculate_iou(self, bbox1, bbox2):
        """
        Calculate Intersection over Union
        """
        x1, y1, x2, y2 = bbox1
        x3, y3, x4, y4 = bbox2

        # Calculate intersection
        xi1, yi1 = max(x1, x3), max(y1, y3)
        xi2, yi2 = min(x2, x4), min(y2, y4)

        if xi2 <= xi1 or yi2 <= yi1:
            return 0

        intersection = (xi2 - xi1) * (yi2 - yi1)

        # Calculate union
        area1 = (x2 - x1) * (y2 - y1)
        area2 = (x4 - x3) * (y4 - y3)
        union = area1 + area2 - intersection

        return intersection / union if union > 0 else 0
```
