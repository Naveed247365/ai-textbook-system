- y: sensor measurement

### Cramér-Rao Lower Bound for Sensor Performance

The Cramér-Rao Lower Bound (CRLB) provides the theoretical limit on parameter estimation accuracy:

**Cov(θ̂) ≥ I⁻¹(θ)**

Where I(θ) is the Fisher Information Matrix.

```python
class CramérRaoAnalysis:
    def __init__(self, sensor_parameters):
        self.params = sensor_parameters

    def fisher_information_matrix(self, state, measurement):
        """
        Calculate Fisher Information Matrix for sensor configuration
        """
        # Calculate log-likelihood gradient
        def log_likelihood_gradient(params):
            # Implementation depends on specific sensor model
            return np.zeros_like(params)

        # Numerical approximation of FIM
        n_params = len(self.params)
        fim = np.zeros((n_params, n_params))

        epsilon = 1e-8
        for i in range(n_params):
            for j in range(n_params):
                # Second derivative approximation
                params_plus_i = self.params.copy()
                params_plus_i[i] += epsilon
                grad_plus_i = log_likelihood_gradient(params_plus_i)

                params_minus_i = self.params.copy()
                params_minus_i[i] -= epsilon
                grad_minus_i = log_likelihood_gradient(params_minus_i)

                fim[i, j] = (grad_plus_i[j] - grad_minus_i[j]) / (2 * epsilon)

        return fim

    def theoretical_precision_limit(self, fim):
        """
        Calculate theoretical precision limit from FIM
        """
        try:
            covariance_bound = np.linalg.inv(fim)
            precision_limits = np.sqrt(np.diag(covariance_bound))
            return precision_limits
        except np.linalg.LinAlgError:
            # FIM is singular - parameters not identifiable
            return np.full(len(self.params), np.inf)
```

### Quantum Sensors

Quantum sensors exploit quantum mechanical properties for ultra-precise measurements:

```python
class QuantumSensor:
    def __init__(self, quantum_state_dimension=2, coherence_time=1e-3):
        self.d = quantum_state_dimension  # Qubit dimension
        self.T2 = coherence_time          # Coherence time
        self.state = np.zeros(self.d, dtype=complex)
        self.state[0] = 1.0  # Initialize in ground state

    def prepare_state(self, parameters):
        """
        Prepare quantum sensor state based on parameters to measure
        """
        # Apply parameter-dependent unitary transformation
        # This is a simplified example for a single qubit
        theta = parameters[0]  # Parameter to measure
        U = np.array([
            [np.cos(theta/2), -np.sin(theta/2)],
            [np.sin(theta/2), np.cos(theta/2)]
        ])
        self.state = U @ self.state
        return self.state

    def interact_with_environment(self, field_strength, interaction_time):
        """
        Quantum sensor interaction with measured field
        """
        # Hamiltonian evolution: |ψ⟩ = exp(-iHt) |ψ₀⟩
        # For magnetic field sensing: H = γB·σ (γ = gyromagnetic ratio)
        gamma = 2.675e8  # Proton gyromagnetic ratio (rad/s/T)
        H = gamma * field_strength * np.array([[1, 0], [0, -1]])  # σ_z for simplicity
        U_evolution = scipy.linalg.expm(-1j * H * interaction_time)
        self.state = U_evolution @ self.state
        return self.state

    def measure(self, observable=np.array([[1, 0], [0, -1]])):
        """
        Perform quantum measurement
        """
        # Calculate expectation value
        expectation = np.vdot(self.state, observable @ self.state)
        return np.real(expectation)

    def sensitivity(self):
        """
        Calculate quantum sensor sensitivity (Heisenberg limit)
        """
        # Heisenberg limit: Δθ ∝ 1/N (N = number of particles)
        # Standard quantum limit: Δθ ∝ 1/√N
        return 1.0 / np.sqrt(self.d)  # Simplified scaling
```

### Event-Based Vision Sensors

Event-based cameras provide asynchronous, sparse temporal information:

```python
class EventBasedCamera:
    def __init__(self, resolution=(640, 480), contrast_threshold=0.1):
        self.width, self.height = resolution
        self.threshold = contrast_threshold
        self.last_frame = np.zeros(resolution)
        self.event_queue = []

    def process_events(self, current_frame, timestamp):
        """
        Process frame to generate events
        """
        # Calculate intensity changes
        diff = current_frame - self.last_frame
        events = np.where(np.abs(diff) > self.threshold)

        # Generate event stream
        for y, x in zip(events[0], events[1]):
            polarity = 1 if diff[y, x] > 0 else -1
            event = {
                'x': x,
                'y': y,
                'timestamp': timestamp,
                'polarity': polarity
            }
            self.event_queue.append(event)

        self.last_frame = current_frame.copy()
        return self.event_queue

    def reconstruct_frame(self, event_window):
        """
        Reconstruct frame from event stream
        """
        frame = np.zeros((self.height, self.width))

        for event in event_window:
            x, y = int(event['x']), int(event['y'])
            if 0 <= x < self.width and 0 <= y < self.height:
                frame[y, x] += event['polarity'] * self.threshold

        return frame

    def temporal_resolution(self):
        """
        Calculate effective temporal resolution
        """
        # Limited by pixel-level processing time
        return 1e-6  # 1 microsecond per event (typical)
```

### Adaptive Filtering for Sensor Noise Reduction

```python
class AdaptiveFilter:
    def __init__(self, filter_order=10, step_size=0.1):
        self.order = filter_order
        self.mu = step_size
        self.weights = np.zeros(filter_order)
        self.buffer = np.zeros(filter_order)

    def lms_filter(self, input_signal, desired_signal):
        """
        Least Mean Squares adaptive filter
        """
        output = np.zeros_like(input_signal)
        error = np.zeros_like(input_signal)

        for i in range(len(input_signal)):
            # Apply filter
            self.buffer = np.roll(self.buffer, 1)
            self.buffer[0] = input_signal[i]

            y = np.dot(self.weights, self.buffer)
            output[i] = y

            # Calculate error
            error[i] = desired_signal[i] - output[i]

            # Update weights
            self.weights += self.mu * error[i] * self.buffer

        return output, error

    def rls_filter(self, input_signal, desired_signal, forgetting_factor=0.98):
        """
        Recursive Least Squares adaptive filter
        """
        P = np.eye(self.order) / 0.1  # Inverse correlation matrix
        output = np.zeros_like(input_signal)
        error = np.zeros_like(input_signal)

        for i in range(len(input_signal)):
            # Update buffer
            self.buffer = np.roll(self.buffer, 1)
            self.buffer[0] = input_signal[i]

            # Calculate gain
            k = P @ self.buffer / (forgetting_factor + self.buffer.T @ P @ self.buffer)

            # Calculate output and error
            output[i] = np.dot(self.weights, self.buffer)
            error[i] = desired_signal[i] - output[i]

            # Update weights and inverse correlation
            self.weights += k * error[i]
            P = (P - k[:, np.newaxis] * self.buffer * k.T) / forgetting_factor

        return output, error
```

```python
class CompressedSensing:
    def __init__(self, compression_ratio=0.5):
        self.M = int(compression_ratio * 1000)  # Measurement matrix rows
        self.N = 1000  # Original signal length
        self.sampling_matrix = np.random.randn(self.M, self.N)

    def compress_signal(self, signal):
        """
        Compress signal using random projection
        """
        if len(signal) != self.N:
            raise ValueError(f"Signal length {len(signal)} doesn't match expected {self.N}")

        compressed = self.sampling_matrix @ signal
        return compressed

    def reconstruct_signal(self, compressed_signal, sparsity=50):
        """
        Reconstruct sparse signal from compressed measurements
        """
        # Solve: min ||x||_1 subject to y = Ax
        # Using basis pursuit or other sparse recovery algorithms
        from scipy.optimize import minimize

        def objective(x):
            return np.sum(np.abs(x))  # L1 norm

        def constraint(x):
            return self.sampling_matrix @ x - compressed_signal

        # Initial guess
        x0 = np.random.randn(self.N) * 0.01

        result = minimize(objective, x0, method='SLSQP',
                         constraints={'type': 'eq', 'fun': constraint})
```

```python
class DistributedSensorNetwork:
    def __init__(self, sensor_nodes, communication_topology):
        self.nodes = sensor_nodes
        self.topology = communication_topology
        self.consensus_matrix = self.compute_consensus_matrix()
        self.fusion_algorithm = 'average_consensus'

    def consensus_based_fusion(self, local_estimates):
        """
        Distributed fusion using consensus algorithms
        """
        # Each node updates its estimate based on neighbors
        new_estimates = np.copy(local_estimates)

        for i, node in enumerate(self.nodes):
            neighbors = np.where(self.topology[i, :] > 0)[0]
            neighbor_estimates = local_estimates[neighbors]

            # Consensus update: x_i(k+1) = x_i(k) + Σ(x_j(k) - x_i(k))
            consensus_term = np.mean(neighbor_estimates) - local_estimates[i]
            new_estimates[i] = local_estimates[i] + 0.1 * consensus_term  # Step size = 0.1

        return new_estimates

    def gossip_based_fusion(self, initial_values):
        """
        Gossip-based distributed fusion
        """
        values = np.copy(initial_values)
        n_nodes = len(values)

        # Random node pair gossip
        for iteration in range(1000):
            # Randomly select two connected nodes
            i, j = np.random.choice(n_nodes, 2, replace=False)
            if self.topology[i, j] > 0:  # Nodes are connected
                # Average their values
                avg = (values[i] + values[j]) / 2
                values[i] = avg
                values[j] = avg

        return values
```

```python
class SensorPerformanceEvaluator:
    def __init__(self):
        self.metrics = {
            'accuracy': [],
            'precision': [],
            'recall': [],
            'f1_score': [],
            'rmse': [],
            'mae': [],
            'timing': []
        }

    def evaluate_detection_performance(self, ground_truth, predictions, iou_threshold=0.5):
        """
        Evaluate object detection performance
        """
        # Calculate True Positives, False Positives, False Negatives
        tp = 0
        fp = 0
        fn = 0

        matched_gt = set()

        for pred in predictions:
            best_iou = 0
            best_gt_idx = -1

            for gt_idx, gt in enumerate(ground_truth):
                if gt_idx in matched_gt:
                    continue

                iou = self.calculate_iou(pred['bbox'], gt['bbox'])
                if iou > best_iou and iou > iou_threshold:
                    best_iou = iou
                    best_gt_idx = gt_idx

            if best_gt_idx != -1:
                tp += 1
                matched_gt.add(best_gt_idx)
            else:
                fp += 1

        fn = len(ground_truth) - len(matched_gt)

        precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        f1_score = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0

        return {
            'precision': precision,
            'recall': recall,
            'f1_score': f1_score,
            'tp': tp,
            'fp': fp,
            'fn': fn
        }

    def calculate_iou(self, bbox1, bbox2):
        """
        Calculate Intersection over Union
        """
        x1, y1, x2, y2 = bbox1
        x3, y3, x4, y4 = bbox2

        # Calculate intersection area
        xi1, yi1 = max(x1, x3), max(y1, y3)
        xi2, yi2 = min(x2, x4), min(y2, y4)

        if xi2 <= xi1 or yi2 <= yi1:
            return 0

        intersection = (xi2 - xi1) * (yi2 - yi1)

        # Calculate union area
        area1 = (x2 - x1) * (y2 - y1)
        area2 = (x4 - x3) * (y4 - y3)
        union = area1 + area2 - intersection

        return intersection / union

    def timing_analysis(self, sensor_data):
        """
        Analyze sensor timing performance
        """
        timestamps = sensor_data.get('timestamps', [])
        if len(timestamps) < 2:
            return {}

        intervals = np.diff(timestamps)
        frequency = 1.0 / np.mean(intervals)

        return {
            'average_frequency': frequency,
            'timing_jitter': np.std(intervals),
            'min_interval': np.min(intervals),
            'max_interval': np.max(intervals)
        }
```
