# روبوٹکس کے لئے AI ایجنٹس (اردو ایڈوانسڈ ورژن)

## باب کا جائزہ

روبوٹکس کے لیے AI ایجنٹس اعلیٰ مستوی کے ریاضیاتی فریم ورکس پر مشتمل ہیں جو آرٹیفیشل انٹیلی جینس، کنٹرول تھیوری، گیم تھیوری، اور معلوماتی تھیوری کو ایک جامع آرکیٹیکچر میں ضم کرتے ہیں۔ یہ ایجنٹ اسٹوکاسٹک آپٹیمائزیشن، فنکشنل آنالیسز، اور ڈیفرینشل جیومیٹری کے اصولوں پر کام کرتے ہیں تاکہ غیر یقینی، متحرک، اور میٹا ورلڈ ماحول میں فیصلہ سازی کی جا سکے۔ نظریاتی بنیادیں قطبیت نظریات، معلوماتی ڈیفرینشل جیومیٹری، اور توسیعی تھیوری کے ذریعے فراہم کی جاتی ہیں۔

### ریاضیاتی فریم ورک

AI روبوٹکس ایجنٹس کو اسٹوکاسٹک کنٹرول تھیوری کے ذریعے ظاہر کیا جا سکتا ہے:

min_π E[∫_0^T L(x_t, u_t, t) dt + Φ(x_T)]

جہاں::
- x_t: سسٹم حالت وقت t پر
- u_t: کنٹرول ان پٹ وقت t پر
- π: پالیسی (ایکشن منتخب کرنے کا قاعدہ)
- L: لگرانجیان (لاگت سیکھنے والے فنکشن)
- Φ: ٹرمنل لاگت
- T: ہوریزن کا وقت

**کنٹرول سسٹم**: ẋ = f(x, u) + g(x, u)w(t)

Where w(t) is process noise (typically Brownian motion).

### سٹوکاسٹک ڈیفرینشل مساوات (SDEs)

کنٹرول سسٹم کو اسٹوکاسٹک ڈیفرینشل مساوات کے ذریعے نمونہ دیا جا سکتا ہے:

dX_t = f(X_t, u_t)dt + g(X_t, u_t)dW_t

جہاں W_t وینر عمل (布朗运动) ہے۔

## اعلیٰ ترین ایجنٹ آرکیٹیکچر

### ملٹی-لیئر ایجنٹس

اسٹریٹجک، ٹیکٹیکل، اور سیمیولیشن لیئرز کے ساتھ::

**ہیورسٹکس ڈیسیجن میکنزمز**::
```
Level 1: ٹاسک لیئر (کیا کرنا ہے)
Level 2: بیہیویئر لیئر (کیسے کرنا ہے)
Level 3: کنٹرول لیئر (ایکٹو ایٹرز کو کیسے چلانا ہے)
```

#### ہمارکیکل ایجنٹس (HLAIs)

HLAI (Hierarchical Latent Action Inference) ایجنٹس کو اعلیٰ سطح کے ایکشنز استعمال کرتے ہیں::

**ریاضیاتی ماڈل**::
p(τ | θ) = ∫ p(τ | z, θ) p(z | θ) dz

جہاں τ ٹریجکٹری ہے، z لیٹنٹ سبٹاسکس ہیں، اور θ ہائیرآرکیکل پالیسی پیرامیٹرز ہیں۔

```python
class HierarchicalAgent:
    def __init__(self, state_dim, action_dim, subtask_dim, goal_dim):
        # ہدف مسخ کا تخمینہ
        self.goal_reacher = GoalReacher(state_dim, goal_dim, action_dim)
        
        # سب ٹاسک سلیکٹر
        self.subtask_selector = SubtaskSelector(state_dim, subtask_dim)
        
        # سب ٹاسک کی پالیسی
        self.subtask_policies = [
            PolicyNetwork(state_dim, action_dim) for _ in range(subtask_dim)
        ]
        
        # ترقی کا ماپنہ
        self.progress_estimator = ProgressEstimator(state_dim + goal_dim, 1)
    
    def plan(self, current_state, goal_state):
        """ہائیرآرکیکل منصوبہ بندی کریں"""
        # ہدف تک پیش رفت کا تخمینہ لگائیں
        progress = self.progress_estimator(torch.cat([current_state, goal_state]))
        
        # اگر ہدف قریب ہے تو براہ راست ایکشن کریں
        if progress > 0.8:
            return self.goal_reacher(current_state, goal_state)
        
        # ورنہ، سب ٹاسکس کا انتخاب کریں
        subtasks = self.subtask_selector(current_state, goal_state)
        
        # ہر سب ٹاسک کے لیے عمل کریں
        full_plan = []
        for subtask in subtasks:
            subtask_policy = self.subtask_policies[subtask]
            subtask_plan = subtask_policy(current_state)
            full_plan.extend(subtask_plan)
        
        return full_plan

class OptionCritic(nn.Module):
    def __init__(self, state_dim, action_dim, n_options, hidden_dim=256):
        super(OptionCritic, self).__init__()
        
        # ہر آپشن کے لیے Q-ویلیو فنکشن
        self.q_network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_options)  # ہر آپشن کے لیے Q-ویلیو
        )
        
        # انٹرا-آپشن پالیسی: π(a|s,o)
        self.intra_option_policy = nn.ModuleList([
            nn.Sequential(
                nn.Linear(state_dim + n_options, hidden_dim),  # + آپشن ایمبیڈنگ
                nn.ReLU(),
                nn.Linear(hidden_dim, hidden_dim),
                nn.ReLU(),
                nn.Linear(hidden_dim, action_dim),
                nn.Softmax(dim=-1)
            ) for _ in range(n_options)
        ])
        
        # آپشن ٹرمینیشن فنکشن: β(s,o)
        self.termination_network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_options),
            nn.Sigmoid()  # ہر آپشن کے لیے ٹرمینیشن امکان
        )
    
    def forward(self, state, option_embedding=None):
        q_values = self.q_network(state)
        
        if option_embedding is not None:
            # انٹرا-آپشن پالیسی کے لیے حالت + آپشن ضم کریں
            state_option = torch.cat([state, option_embedding], dim=-1)
            action_probs = self.intra_option_policies[option](
                state_option
            )
        else:
            action_probs = None
        
        termination_probs = self.termination_network(state)
        
        return q_values, action_probs, termination_probs

def update_option_critic(agent, state, option, action, reward, next_state, next_option, done, gamma):
    """Option-Critic الگورتھم کے لیے اپ ڈیٹ قواعد"""
    current_q, _, _ = agent(state, option_onehot(option))
    next_q, _, next_termination = agent(next_state, option_onehot(next_option))
    
    # انٹرا- آپشن پالیسی نقصان
    action_probs, _ = agent.intra_option_policy(
        torch.cat([state, option_onehot(option)], dim=-1)
    )
    log_prob = torch.log(action_probs[action] + 1e-8)
    
    # فوائد کا حساب لگائیں
    with torch.no_grad():
        next_q_max = torch.max(next_q)
        target_q = reward + gamma * (
            (1 - next_termination[next_option]) * next_q_max + 
            next_termination[next_option] * torch.max(next_q)
        )
    
    advantage = target_q - current_q[option]
    
    # پالیسی گریڈیئنٹ
    policy_loss = -log_prob * advantage
    
    # Q-ویلیو نقصان
    q_loss = F.mse_loss(current_q[option], target_q)
    
    # ٹرمینیشن نقصان
    termination_loss = termination_probs[option] * advantage
    
    return policy_loss + q_loss + termination_loss
```

### بیسیان فیوژن فریم ورک

متعدد ماڈلز کو ہمدیگر کے ساتھ ضم کرنا::

**کنڈیشنل ڈسٹری بیوشن**::
p(x | z₁, z₂, ..., z_n) ∝ p(z₁, z₂, ..., z_n | x) p(x)

Where zᵢ are measurements from different sensors.

**کو واریئنس نقل کا احاطہ کرنا**::
For correlated sensor noises, proper covariance modeling is crucial:

Σ_total = Σ₁ + Σ₂ + Σ₁₂

Where Σ₁₂ represents cross-covariance between sensors.

```python
class BayesianFusionAgent:
    def __init__(self, sensor_specs):
        """
        sensor_specs: [{name, noise_covariance, model_function}, ...]
        """
        self.sensors = sensor_specs
        self.state_dim = None  # Will be determined from first measurement
        self.belief_state = None  # [mean, covariance]
        
    def initialize_belief(self, initial_mean, initial_covariance):
        """Initialize the agent's belief state"""
        self.belief_mean = initial_mean
        self.belief_cov = initial_covariance
    
    def fuse_measurements(self, measurements):
        """
        measurements: {sensor_name: (value, covariance), ...}
        """
        # Start with prior belief
        posterior_mean = self.belief_mean.clone()
        posterior_cov_inv = torch.inverse(self.belief_cov)
        
        # For each measurement, update the belief
        for sensor_name, (measurement, sensor_cov) in measurements.items():
            H = self.get_observation_matrix(sensor_name)  # Observation model
            
            # Compute Kalman gain equivalent
            innovation_cov = H @ self.belief_cov @ H.T + sensor_cov
            K = self.belief_cov @ H.T @ torch.inverse(innovation_cov)
            
            # Update equations
            innovation = measurement - H @ self.belief_mean
            posterior_mean = self.belief_mean + K @ innovation
            posterior_cov = (torch.eye(self.state_dim) - K @ H) @ self.belief_cov
        
        self.belief_mean = posterior_mean
        self.belief_cov = posterior_cov
        
        return self.belief_mean, self.belief_cov
```

### فیلٹر کے انویئنٹس کے لیے گوسیان مکسچر ماڈل

```python
class GaussianMixtureFusion:
    def __init__(self, n_components=5, state_dim=6):
        self.n_components = n_components
        self.state_dim = state_dim
        
        # ہر جزو کے لیے امکانات، مین، اور کوواریئنس
        self.weights = torch.ones(n_components) / n_components
        self.means = torch.randn(n_components, state_dim) * 0.1
        self.covariances = torch.stack([
            torch.eye(state_dim) for _ in range(n_components)
        ])
    
    def predict(self):
        """Predict next state distribution"""
        for i in range(self.n_components):
            # Propagate each Gaussian component through dynamics
            self.means[i] = self.dynamics(self.means[i])
            
            # Linearize dynamics and propagate covariance
            F = self.compute_jacobian_dynamics(self.means[i])
            self.covariances[i] = (F @ self.covariances[i] @ F.T + 
                                 self.process_noise_cov)
    
    def update(self, observation, sensor_noise):
        """Update mixture components based on observation"""
        for i in range(self.n_components):
            # Observation model
            H = self.observation_jacobian(self.means[i])
            
            # Kalman update for each component
            innovation_cov = H @ self.covariances[i] @ H.T + sensor_noise
            K = self.covariances[i] @ H.T @ torch.inverse(innovation_cov)
            
            # Update mean and covariance
            innovation = observation - self.observation_model(self.means[i])
            self.means[i] = self.means[i] + K @ innovation
            I = torch.eye(self.state_dim)
            self.covariances[i] = (I - K @ H) @ self.covariances[i]
            
            # Update component weight based on likelihood
            likelihood = self.compute_observation_likelihood(
                observation, self.means[i], self.covariances[i], sensor_noise
            )
            self.weights[i] *= likelihood
        
        # Normalize weights
        self.weights /= torch.sum(self.weights)
```

## ری انفورسمنٹ لرننگ فریم ورکس

### Deep Deterministic Policy Gradient (DDPG)

DDPG ڈسکریٹ ایکشن سپیس کے ساتھ کام کرتا ہے::

**Mathematical Foundation**::
The policy gradient theorem for continuous actions:

∇_θ J(π_θ) = E_{τ~π_θ}[Σ_t ∇_θ log π_θ(a_t|s_t) Q^{π_θ}(s_t, a_t)]

Where τ is a trajectory and Q is the action-value function.

```python
class DDPGAgent:
    def __init__(self, state_dim, action_dim, hidden_dim=256, lr_actor=1e-4, lr_critic=1e-3):
        # Actor network (deterministic policy)
        self.actor = ActorNetwork(state_dim, action_dim, hidden_dim)
        self.actor_target = ActorNetwork(state_dim, action_dim, hidden_dim)
        self.actor_optimizer = torch.optim.Adam(self.actor.parameters(), lr=lr_actor)
        
        # Critic network (Q-function)
        self.critic = CriticNetwork(state_dim, action_dim, hidden_dim)
        self.critic_target = CriticNetwork(state_dim, action_dim, hidden_dim)
        self.critic_optimizer = torch.optim.Adam(self.critic.parameters(), lr=lr_critic)
        
        # Initialize target networks with same weights as main networks
        self.soft_update(self.actor_target, self.actor, 1.0)
        self.soft_update(self.critic_target, self.critic, 1.0)
        
        # Hyperparameters
        self.gamma = 0.99  # Discount factor
        self.tau = 0.005   # Soft update factor
        self.noise_std = 0.1  # Exploration noise standard deviation
    
    def select_action(self, state, add_noise=True):
        """Select action according to current policy"""
        with torch.no_grad():
            action = self.actor(state)
            
            if add_noise:
                # Add Ornstein-Uhlenbeck process noise for exploration
                noise = torch.randn_like(action) * self.noise_std
                action += noise
            
            # Clamp action to valid range
            action = torch.clamp(action, -1, 1)
        
        return action
    
    def update(self, replay_buffer, batch_size=100):
        """Update both actor and critic networks"""
        if len(replay_buffer) < batch_size:
            return  # Not enough samples to update
        
        # Sample batch from replay buffer
        states, actions, rewards, next_states, dones = replay_buffer.sample(batch_size)
        
        # Compute target Q-values using target networks
        with torch.no_grad():
            next_actions = self.actor_target(next_states)
            next_q_values = self.critic_target(next_states, next_actions)
            target_q_values = rewards + (1 - dones) * self.gamma * next_q_values
        
        # Critic update
        current_q_values = self.critic(states, actions)
        critic_loss = F.mse_loss(current_q_values, target_q_values)
        
        self.critic_optimizer.zero_grad()
        critic_loss.backward()
        self.critic_optimizer.step()
        
        # Actor update (maximize Q-value of current policy's actions)
        predicted_actions = self.actor(states)
        actor_loss = -self.critic(states, predicted_actions).mean()
        
        self.actor_optimizer.zero_grad()
        actor_loss.backward()
        self.actor_optimizer.step()
        
        # Soft update target networks
        self.soft_update(self.actor_target, self.actor, self.tau)
        self.soft_update(self.critic_target, self.critic, self.tau)
    
    def soft_update(self, target_network, source_network, tau):
        """Soft update of target network parameters"""
        for target_param, param in zip(target_network.parameters(), source_network.parameters()):
            target_param.data.copy_(tau * param.data + (1 - tau) * target_param.data)

class TD3Agent:
    """
    Twin Delayed Deep Deterministic Policy Gradient
    Uses two critics to reduce overestimation bias
    """
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        # Actor
        self.actor = ActorNetwork(state_dim, action_dim, hidden_dim)
        self.actor_target = ActorNetwork(state_dim, action_dim, hidden_dim)
        
        # Two critics (to reduce overestimation)
        self.critic1 = CriticNetwork(state_dim, action_dim, hidden_dim)
        self.critic2 = CriticNetwork(state_dim, action_dim, hidden_dim)
        self.critic1_target = CriticNetwork(state_dim, action_dim, hidden_dim)
        self.critic2_target = CriticNetwork(state_dim, action_dim, hidden_dim)
        
        # Target networks start with same weights as main networks
        self.hard_update(self.actor_target, self.actor)
        self.hard_update(self.critic1_target, self.critic1)
        self.hard_update(self.critic2_target, self.critic2)
    
    def update_critic(self, states, actions, rewards, next_states, dones):
        """Update both critic networks"""
        with torch.no_grad():
            # Target policy smoothing (add noise to target actions)
            noise = torch.randn_like(actions) * 0.2
            noise = torch.clamp(noise, -0.5, 0.5)  # Clipping
            next_actions = torch.clamp(
                self.actor_target(next_states) + noise,
                -1, 1
            )
            
            # Target Q-values from both critics (use minimum to reduce overestimation)
            next_q1 = self.critic1_target(next_states, next_actions)
            next_q2 = self.critic2_target(next_states, next_actions)
            next_q = torch.min(next_q1, next_q2)
            target_q = rewards + (1 - dones) * self.gamma * next_q
        
        # Critic losses
        current_q1 = self.critic1(states, actions)
        current_q2 = self.critic2(states, actions)
        critic1_loss = F.mse_loss(current_q1, target_q)
        critic2_loss = F.mse_loss(current_q2, target_q)
        
        # Update both critics
        self.critic1_optimizer.zero_grad()
        critic1_loss.backward()
        self.critic1_optimizer.step()
        
        self.critic2_optimizer.zero_grad()
        critic2_loss.backward()
        self.critic2_optimizer.step()
    
    def delayed_actor_update(self, states):
        """
        Update actor network with delay (update every 2 critic updates)
        Also applies policy delay to reduce variance
        """
        if self.total_update_steps % 2 == 0:  # Delayed update
            # Use critic 1 for actor update (both critics would give similar gradients)
            predicted_actions = self.actor(states)
            actor_loss = -self.critic1(states, predicted_actions).mean()
            
            self.actor_optimizer.zero_grad()
            actor_loss.backward()
            self.actor_optimizer.step()
            
            # Soft update targets
            self.soft_update(self.actor_target, self.actor, self.tau)
            self.soft_update(self.critic1_target, self.critic1, self.tau)
            self.soft_update(self.critic2_target, self.critic2, self.tau)
```

### Soft Actor-Critic (SAC) - Maximum Entropy RL

SAC maximizes both expected reward and policy entropy:

**Objective Function**::
J(π) = E[Σ_t r(s_t, a_t) + αH(π(·|s_t))]

Where H(π(·|s_t)) = -∫ π(a|s_t) log π(a|s_t) da is the entropy of the policy.

```python
class SoftActorCriticAgent:
    def __init__(self, state_dim, action_dim, hidden_dim=256, alpha=0.2):
        # Actor (Gaussian policy)
        self.actor = GaussianPolicy(state_dim, action_dim, hidden_dim)
        
        # Two Q-networks for double Q-learning
        self.q_net1 = QNetwork(state_dim, action_dim, hidden_dim)
        self.q_net2 = QNetwork(state_dim, action_dim, hidden_dim)
        
        # Value network (for entropy computation)
        self.v_net = VNetwork(state_dim, hidden_dim)
        
        # Target networks for stability
        self.q_net1_target = copy.deepcopy(self.q_net1)
        self.q_net2_target = copy.deepcopy(self.q_net2)
        
        # Optimizers
        self.actor_optimizer = torch.optim.Adam(self.actor.parameters(), lr=3e-4)
        self.q1_optimizer = torch.optim.Adam(self.q_net1.parameters(), lr=3e-4)
        self.q2_optimizer = torch.optim.Adam(self.q_net2.parameters(), lr=3e-4)
        self.v_optimizer = torch.optim.Adam(self.v_net.parameters(), lr=3e-4)
        
        # Adaptive entropy coefficient
        self.target_entropy = -torch.prod(torch.Tensor(action_dim)).item()
        self.log_alpha = torch.zeros(1, requires_grad=True)
        self.alpha_optimizer = torch.optim.Adam([self.log_alpha], lr=3e-4)
        self.alpha = alpha
    
    def update(self, replay_buffer, batch_size=256):
        state, action, reward, next_state, done = replay_buffer.sample(batch_size)
        
        # Update Q functions
        with torch.no_grad():
            next_action, next_log_prob, _ = self.actor.sample(next_state)
            next_v = self.v_net(next_state)
            next_q = torch.min(
                self.q_net1_target(next_state, next_action),
                self.q_net2_target(next_state, next_action)
            ) - self.alpha * next_log_prob
            target_q = reward + (1 - done) * self.gamma * next_q
        
        # Q-function losses
        current_q1 = self.q_net1(state, action)
        current_q2 = self.q_net2(state, action)
        q1_loss = F.mse_loss(current_q1, target_q)
        q2_loss = F.mse_loss(current_q2, target_q)
        
        # Update Q networks
        self.q1_optimizer.zero_grad()
        q1_loss.backward()
        self.q1_optimizer.step()
        
        self.q2_optimizer.zero_grad()
        q2_loss.backward()
        self.q2_optimizer.step()
        
        # Update policy
        pi, log_pi, _ = self.actor.sample(state)
        
        with torch.no_grad():
            q1_pi = self.q_net1(state, pi)
            q2_pi = self.q_net2(state, pi)
            min_q_pi = torch.min(q1_pi, q2_pi)
        
        # Policy loss (maximize expected Q-value while maintaining entropy)
        policy_loss = ((self.alpha * log_pi) - min_q_pi).mean()
        
        self.actor_optimizer.zero_grad()
        policy_loss.backward()
        self.actor_optimizer.step()
        
        # Update entropy coefficient
        alpha_loss = -(self.log_alpha * (log_pi + self.target_entropy).detach()).mean()
        self.alpha_optimizer.zero_grad()
        alpha_loss.backward()
        self.alpha_optimizer.step()
        self.alpha = self.log_alpha.exp()
```

## ملٹی-ایجنٹ سسٹم

### ملٹی-ایجنٹ ری انفورسمنٹ لرننگ (MARL)

MARL deals with multiple agents learning simultaneously in the same environment:

**Mathematical Formulation (Markov Game)**::
G = (N, S, {A_i}, P, {R_i}, γ)

Where:
- N: Set of agents
- S: Joint state space
- A_i: Action space for agent i
- P: Joint transition probabilities
- R_i: Reward function for agent i
- γ: Discount factor

**Types of MARL Scenarios**:
- **Cooperative**: All agents maximize joint reward
- **Competitive**: Agents have opposing interests
- **Mixed**: Combination of cooperation and competition

```python
class MultiAgentDDPG:
    def __init__(self, n_agents, state_dim, action_dim, hidden_dim=256):
        self.n_agents = n_agents
        
        # Each agent has its own actor-critic pair
        self.actors = [DDPGActor(state_dim, action_dim, hidden_dim) 
                      for _ in range(n_agents)]
        self.critics = [DDPGCritic(state_dim * n_agents, action_dim * n_agents, hidden_dim) 
                       for _ in range(n_agents)]
        
        # Target networks for stability
        self.target_actors = [copy.deepcopy(actor) for actor in self.actors]
        self.target_critics = [copy.deepcopy(critic) for critic in self.critics]
        
        # Optimizers
        self.actor_optimizers = [
            torch.optim.Adam(actor.parameters(), lr=1e-4) for actor in self.actors
        ]
        self.critic_optimizers = [
            torch.optim.Adam(critic.parameters(), lr=1e-3) for critic in self.critics
        ]
    
    def get_actions(self, states, add_noise=True):
        """Get actions for all agents"""
        actions = []
        for i in range(self.n_agents):
            action = self.actors[i](states[i])
            if add_noise:
                noise = torch.randn_like(action) * 0.1
                action += noise
            actions.append(torch.clamp(action, -1, 1))
        return actions
    
    def centralized_training_update(self, states, actions, rewards, next_states, dones):
        """
        Centralized training with decentralized execution (CTDE)
        Critics have access to global information during training
        """
        for i in range(self.n_agents):
            # Combine all states and actions for centralized critic
            all_states = torch.cat(states, dim=1)  # Concatenate all agent states
            all_next_actions = []
            for j in range(self.n_agents):
                if j == i:  # Use current agent's action
                    all_next_actions.append(actions[j])
                else:  # Use target policy's action
                    with torch.no_grad():
                        next_action = self.target_actors[j](next_states[j])
                        all_next_actions.append(next_action)
            
            all_next_actions_cat = torch.cat(all_next_actions, dim=1)
            
            # Compute target Q-value using target networks
            with torch.no_grad():
                next_q1 = self.target_critics[i](all_states, all_next_actions_cat)
                target_q = rewards[i] + (1 - dones[i]) * self.gamma * next_q1
            
            # Current Q-value
            all_actions_cat = torch.cat(actions, dim=1)
            current_q = self.critics[i](all_states, all_actions_cat)
            
            # Critic loss
            critic_loss = F.mse_loss(current_q, target_q)
            
            # Update critic
            self.critic_optimizers[i].zero_grad()
            critic_loss.backward()
            self.critic_optimizers[i].step()
            
            # Actor update (use current centralized critic)
            current_action = self.actors[i](states[i])
            
            # Replace this agent's action in the action list
            all_actions_for_actor = []
            for j in range(self.n_agents):
                if j == i:
                    all_actions_for_actor.append(current_action)
                else:
                    all_actions_for_actor.append(actions[j].detach())  # Detach others
            all_actions_actor_cat = torch.cat(all_actions_for_actor, dim=1)
            
            # Actor loss (maximize own Q-value)
            actor_loss = -self.critics[i](all_states, all_actions_actor_cat).mean()
            
            # Update actor
            self.actor_optimizers[i].zero_grad()
            actor_loss.backward()
            self.actor_optimizers[i].step()
        
        # Soft update target networks
        for i in range(self.n_agents):
            self.soft_update(self.target_actors[i], self.actors[i], self.tau)
            self.soft_update(self.target_critics[i], self.critics[i], self.tau)
```

```python
            adapt_traj = self.sample_trajectory(task.adaptation_env)
            eval_traj = self.sample_trajectory(task.evaluation_env)
            
            # Adapt to specific task

        meta_loss.backward()
        self.meta_optimizer.step()

        adaptation_traj = self.sample_trajectory(new_task_env)

            loss = self.compute_task_loss(adapted_network, adaptation_traj)

            gradients = torch.autograd.grad(loss, adapted_network.parameters())
            for param, grad in zip(adapted_network.parameters(), gradients):
                param.data = param.data - 0.01 * grad  # Fast adaptation LR
        
        return adapted_network
```
