# AI Agents for Robotics (English Advanced Version)

## Chapter Overview

AI agents for robotics represent sophisticated autonomous systems that integrate advanced AI methodologies including probabilistic reasoning, optimal control theory, multi-agent systems theory, and cognitive architectures. These agents implement complex mathematical frameworks grounded in decision theory, utility optimization, and information theory to enable autonomous robots to operate effectively in uncertain, dynamic environments. The theoretical foundations encompass Markov decision processes, Bayesian inference, game theory, and reinforcement learning theory.

### Mathematical Framework

AI agents are fundamentally modeled as rational decision-makers that maximize expected utility:

U(π) = E[∑_{t=0}^∞ γ^t R(s_t, a_t) | π]

Where:
- π: Policy function mapping states to actions
- U(π): Expected utility under policy π
- R(s_t, a_t): Reward function at time t
- γ: Discount factor for future rewards
- s_t, a_t: State and action at time t

### Formal Agent Definition

A complete AI agent can be defined as a tuple:
A = ⟨S, A, T, R, Ω, O, P, U, G⟩

Where:
- S: State space
- A: Action space
- T: Transition function T(s, a, s') = P(s_{t+1} = s' | s_t = s, a_t = a)
- R: Reward function R(s, a)
- Ω: Observation space
- O: Observation function O(o, s) = P(o_t = o | s_t = s)
- P: Prior beliefs about environment
- U: Utility function
- G: Goals and objectives

### Partially Observable Markov Decision Process (POMDP)

POMDP agents handle uncertainty and partial observability:

```python
class POMDPAgent:
    def __init__(self, states, actions, observations, 
                 transition_function, observation_function, reward_function):
        self.states = states
        self.actions = actions
        self.observations = observations
        self.T = transition_function  # T(s, a, s')
        self.O = observation_function  # O(o, s)
        self.R = reward_function  # R(s, a)
        
        # Belief state: probability distribution over states
        self.belief_state = np.ones(len(states)) / len(states)
        
        # Value function for each belief point
        self.value_function = {}
    
    def update_belief(self, action, observation):
        """Update belief state using Bayes' rule"""
        new_belief = np.zeros(len(self.states))
        
        for si, s in enumerate(self.states):
            # P(o|s) * sum_s' P(s|s',a) * b(s')
            obs_prob = self.O(observation, s)
            state_prob_sum = 0
            
            for sj, prev_s in enumerate(self.states):
                state_prob_sum += self.T(prev_s, action, s) * self.belief_state[sj]
            
            new_belief[si] = obs_prob * state_prob_sum
        
        # Normalize
        if np.sum(new_belief) > 0:
            self.belief_state = new_belief / np.sum(new_belief)
        else:
            # If all probabilities sum to zero, maintain previous belief
            pass
    
    def expected_utility(self, action):
        """Compute expected utility of an action"""
        utility = 0
        for si, s in enumerate(self.states):
            reward = self.R(s, action)
            utility += self.belief_state[si] * reward
        return utility
    
    def select_action(self):
        """Select action with highest expected utility"""
        best_action = None
        best_utility = float('-inf')
        
        for action in self.actions:
            utility = self.expected_utility(action)
            if utility > best_utility:
                best_utility = utility
                best_action = action
        
        return best_action
```

```python
class ProbabilisticRoadmap:
    def __init__(self, environment_bounds, collision_checker):
        self.environment_bounds = environment_bounds
        self.collision_checker = collision_checker
        self.roadmap = Graph()
        self.k_neighbors = 10
    
    def build_roadmap(self, num_samples=1000):
        """Build roadmap by sampling collision-free points"""
        free_configs = []
        
        for _ in range(num_samples):
            config = self.sample_configuration()
            if self.collision_checker.is_valid(config):
                free_configs.append(config)
        
        # Connect nearby configurations
        for i, config1 in enumerate(free_configs):
            neighbors = self.find_k_nearest(config1, free_configs, self.k_neighbors)
            for config2 in neighbors:
                if self.is_directly_connectable(config1, config2):
                    self.roadmap.add_edge(config1, config2)
    
    def sample_configuration(self):
        """Sample random configuration in environment"""
        return np.random.uniform(self.environment_bounds[0], 
                                self.environment_bounds[1])
    
    def find_k_nearest(self, config, configs, k):
        """Find k nearest configurations"""
        distances = [np.linalg.norm(np.array(config) - np.array(c)) 
                     for c in configs]
        indices = np.argpartition(distances, k)[:k]
        return [configs[i] for i in indices if i != configs.index(config)]
    
    def query_path(self, start, goal):
        """Find path between start and goal using roadmap"""
        # Add start and goal to roadmap temporarily
        self.roadmap.add_node(start)
        self.roadmap.add_node(goal)
        
        # Connect to nearby roadmap nodes
        start_neighbors = self.find_k_nearest(start, 
                                            list(self.roadmap.nodes), 
                                            self.k_neighbors)
        goal_neighbors = self.find_k_nearest(goal, 
                                           list(self.roadmap.nodes), 
                                           self.k_neighbors)
        
        for neighbor in start_neighbors:
            if self.is_directly_connectable(start, neighbor):
                self.roadmap.add_edge(start, neighbor)
        
        for neighbor in goal_neighbors:
            if self.is_directly_connectable(goal, neighbor):
                self.roadmap.add_edge(goal, neighbor)
        
        # Use path planning algorithm to find route
        path = self.find_shortest_path(start, goal)
        
        # Remove temporary connections
        for neighbor in start_neighbors:
            self.roadmap.remove_edge_if_exists(start, neighbor)
        for neighbor in goal_neighbors:
            self.roadmap.remove_edge_if_exists(goal, neighbor)
        
        return path
```

#### Rapidly-exploring Random Trees (RRT)

```python
class RRTPlanner:
    def __init__(self, start_configuration, configuration_space, collision_checker):
        self.start_config = start_configuration
        self.configuration_space = configuration_space
        self.collision_checker = collision_checker
        self.tree = nx.Graph()
        self.tree.add_node(start_configuration)
        self.step_size = 0.1
        self.goal_bias = 0.05
    
    def plan(self, goal_configuration, max_iterations=10000):
        """Plan path using RRT algorithm"""
        for i in range(max_iterations):
            # Sample random configuration
            if np.random.random() < self.goal_bias:
                random_config = goal_configuration
            else:
                random_config = self.sample_free_space()
            
            # Find nearest vertex in tree
            nearest_vertex = self.find_nearest_vertex(random_config)
            
            # Extend towards random configuration
            new_config = self.extend_towards(nearest_vertex, random_config)
            
            if new_config is not None:
                # Add new configuration to tree
                self.tree.add_edge(nearest_vertex, new_config)
                
                # Check if goal reached
                if self.is_near_goal(new_config, goal_configuration):
                    return self.extract_path(goal_configuration)
        
        return None  # No path found
    
    def sample_free_space(self):
        """Sample configuration in free space"""
        while True:
            config = self.configuration_space.sample()
            if self.collision_checker.is_valid(config):
                return config
    
    def find_nearest_vertex(self, config):
        """Find nearest vertex in tree to given configuration"""
        min_distance = float('inf')
        nearest = None
        
        for vertex in self.tree.nodes:
            distance = self.configuration_distance(vertex, config)
            if distance < min_distance:
                min_distance = distance
                nearest = vertex
        
        return nearest
    
    def extend_towards(self, from_config, to_config):
        """Extend tree towards configuration"""
        direction = np.array(to_config) - np.array(from_config)
        distance = np.linalg.norm(direction)
        
        if distance <= self.step_size:
            candidate = to_config
        else:
            direction = direction / distance
            candidate = tuple(np.array(from_config) + self.step_size * direction)
        
        # Check if path is collision-free
        if self.is_path_collision_free(from_config, candidate):
            return candidate
        else:
            return None
    
    def is_path_collision_free(self, start, end):
        """Check if path between configurations is collision-free"""
        num_steps = max(int(np.linalg.norm(np.array(end) - np.array(start)) / 0.01), 10)
        
        for i in range(1, num_steps):
            t = i / num_steps
            intermediate = tuple(np.array(start) * (1 - t) + np.array(end) * t)
            if not self.collision_checker.is_valid(intermediate):
                return False
        
        return True
```

```python
class MultiModalFusionAgent:
    """
    Agent that fuses Kalman filtering for sensor fusion
    """
    def __init__(self, state_dim, sensors):
        self.state_dim = state_dim
        self.sensors = sensors  # List of sensor objects
        self.state_vector = np.zeros(state_dim)
        self.covariance_matrix = np.eye(state_dim)
        
        # Sensor models
        self.sensor_models = {
            sensor.name: self.build_sensor_model(sensor)
            for sensor in sensors
        }
    
    def build_sensor_model(self, sensor):
        """Build observation model for sensor"""
        return {
            'H': sensor.observation_matrix,  # Observation matrix
            'R': sensor.noise_covariance,    # Sensor noise
            'transform': sensor.observation_transform
        }
    
    def predict(self, control_input, dt):
        """Prediction step of Kalman filter"""
        # State transition model (simplified as identity for now)
        F = np.eye(self.state_dim)  # Could be more complex
        
        # Process noise covariance
        Q = self.estimate_process_noise(dt)
        
        # Predict state
        self.state_vector = F @ self.state_vector + control_input
        
        # Predict covariance
        self.covariance_matrix = F @ self.covariance_matrix @ F.T + Q
    
    def update(self, sensor_readings):
        """Update step of Kalman filter with multiple sensors"""
        for sensor_name, reading in sensor_readings.items():
            if sensor_name in self.sensor_models:
                model = self.sensor_models[sensor_name]
                
                # Compute innovation
                predicted_observation = model['H'] @ self.state_vector
                innovation = reading - predicted_observation
                
                # Innovation covariance
                S = model['H'] @ self.covariance_matrix @ model['H'].T + model['R']
                
                # Kalman gain
                K = self.covariance_matrix @ model['H'].T @ np.linalg.inv(S)
                
                # Update state
                self.state_vector = self.state_vector + K @ innovation
                
                # Update covariance
                I = np.eye(len(self.state_vector))
                self.covariance_matrix = (I - K @ model['H']) @ self.covariance_matrix

                                (self.num_particles, self.state_dim))
        self.particles += control_input + noise
    
    def update_weights(self, sensor_readings, sensor_models):
        """Update particle weights based on sensor observations"""
        for i, particle in enumerate(self.particles):
            total_weight = 1.0
            for sensor_name, reading in sensor_readings.items():
                if sensor_name in sensor_models:
                    model = sensor_models[sensor_name]
                    predicted_reading = model['transform'](particle)

                        error, model['R']

    def gaussian_likelihood(self, error, covariance):
        """Compute likelihood under Gaussian assumption"""
        norm_constant = 1.0 / np.sqrt((2 * np.pi) ** len(error) * np.linalg.det(covariance))
        exponent = -0.5 * error.T @ np.linalg.inv(covariance) @ error
        return norm_constant * np.exp(exponent)
    
    def resample(self):
        """Resample particles based on weights"""
        indices = np.random.choice(
            self.num_particles, 
            size=self.num_particles, 
            p=self.weights
        )
        self.particles = self.particles[indices]
        self.weights = np.ones(self.num_particles) / self.num_particles
    
    def estimate_state(self):
        """Estimate state as weighted average of particles"""
        return np.average(self.particles, axis=0, weights=self.weights)
```

```python
        # Estimate safety of proposed action
        safety_score = self.unsafe_estimator.estimate(
            current_state, proposed_action
        )
        
        if safety_score > self.safety_threshold:
            # Action is unsafe, repair it
            safe_action = self.repair(
                proposed_action, current_state, safety_score
            )
            return safe_action
        else:
            # Action is safe, return as-is
            return proposed_action
```

```python
def compute_safety_score(agent, scenarios):
    """
    Compute safety score based on various safety metrics
    """
    safety_metrics = {
        'collision_frequency': 0,
        'safe_zone_violations': 0,
        'emergency_stops': 0,
        'human_safety_incidents': 0
    }
    
    total_episodes = 0
    for scenario in scenarios:
        env = scenario.initialize_environment()
        agent.reset()
        
        for episode in scenario.episodes:
            done = False
            while not done:
                action = agent.act(env.get_observation())
                _, _, done, info = env.step(action)
                
                # Check for safety violations
                if info.get('collision'):
                    safety_metrics['collision_frequency'] += 1
                if info.get('unsafe_zone_entered'):
                    safety_metrics['safe_zone_violations'] += 1
                if info.get('emergency_stop_triggered'):
                    safety_metrics['emergency_stops'] += 1
                if info.get('human_safety_violation'):
                    safety_metrics['human_safety_incidents'] += 1
            
            total_episodes += 1
    
    # Normalize and compute composite safety score
    collision_rate = safety_metrics['collision_frequency'] / max(total_episodes, 1)
    zone_violations_rate = safety_metrics['safe_zone_violations'] / max(total_episodes, 1)
    
    # Invert rates (higher is worse) to get scores (higher is better)
    safety_score = (1 - collision_rate) * (1 - zone_violations_rate) * 100
    
    return max(0, min(100, safety_score))  # Scale to [0, 100]

def compute_adaptability_score(agent, scenarios):
    """
    Compute adaptability based on agent's ability to adjust to new conditions
    """
    adaptation_metrics = {
        'task_switching_time': [],
        'performance_recovery_rate': [],
        'novel_situation_handling': []
    }
    
    for scenario in scenarios:
        # Measure how quickly agent adapts to new tasks/environments
        baseline_performance = measure_baseline_performance(agent, scenario.baseline)
        adapted_performance = measure_performance_after_change(agent, scenario.change)
        
        adaptation_time = measure_time_to_restore_performance(
            agent, baseline_performance, adapted_performance
        )
        adaptation_metrics['task_switching_time'].append(adaptation_time)
        
        recovery_rate = compute_performance_recovery_rate(
            agent, adapted_performance, baseline_performance
        )
        adaptation_metrics['performance_recovery_rate'].append(recovery_rate)
        
        # Novel situation handling
        novel_score = evaluate_novel_situation_response(agent, scenario.novel_elements)
        adaptation_metrics['novel_situation_handling'].append(novel_score)
    
    # Average across all scenarios
    avg_switching_time = np.mean(adaptation_metrics['task_switching_time'])
    avg_recovery_rate = np.mean(adaptation_metrics['performance_recovery_rate'])
    avg_novel_handling = np.mean(adaptation_metrics['novel_situation_handling'])
    
    # Combine into composite adaptability score
    adaptability_score = (avg_recovery_rate * 0.5 + 
                         (1 / (avg_switching_time + 1)) * 0.3 + 
                         avg_novel_handling * 0.2) * 100
    
    return adaptability_score
```
