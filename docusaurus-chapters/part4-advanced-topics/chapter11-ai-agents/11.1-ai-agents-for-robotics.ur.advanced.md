# روبوٹکس کے لیے AI ایجنٹ (اردو ایڈوانسڈ ورژن)

## باب کا جائزہ

روبوٹکس کے لیے AI ایجنٹ جدید مصنوعی ذہانت کے نظام کی نمائندگی کرتے ہیں جو احتمال کی منطق، بہترین کنٹرول تھیوری، ملٹی-ایجنٹ سسٹم تھیوری، اور کوگنیٹو آرکیٹیکچر کو ضم کرتے ہیں تاکہ غیر یقینی، متحرک ماحول میں خودکار فیصلے کیے جا سکیں۔ یہ ایجنٹ اعلیٰ درجے کے ریاضیاتی فریم ورکس کا استعمال کرتے ہیں جو فیصلہ سازی کے نظریہ، مفادات کے کام کے تصور، اور معلوماتی نظریات پر مبنی ہیں تاکہ تسلسل میں کارروائیوں کو منظم کیا جا سکے۔ نظریاتی بنیادوں میں یونیورسل ایپروکسیمیشن تھیورم، ریپریزنٹیشن لرننگ تھیوری، اور آپٹیمائزیشن جیومیٹری شامل ہیں۔

### ریاضیاتی فریم ورک

AI ایجنٹ کو اس طرح مظہر کیا جا سکتا ہے:
A = ⟨S, A, T, R, Ω, O, π, γ, μ, H⟩

جہاں:
- S: حالت کی جگہ (ممکنہ طور پر جاری اور لامحدود)
- A: کارروائی کی جگہ (ڈسکریٹ یا کنٹی نیوئس)
- T: منتقلی کی ترجیحات T(s,a,s') = P(s'|s,a)
- R: انعام فنکشن R(s,a,s')
- Ω: مشاہدہ جگہ
- O: مشاہدہ فنکشن O(o,s)
- π: پالیسی π(s) → a
- γ: ڈسکاؤنٹ فیکٹر
- μ: ابتدائی حالت کی تقسیم
- H: میموری کی حد (اگر ہیورسٹک ہے)

### اسٹوکاسٹک کنٹرول فارمولیشن

ایجنٹ کا مقصد سیکھنے کے ذریعے یہ تلاش کرنا ہے:
π* = argmax_π E[∑_{t=0}^∞ γ^t R(s_t, a_t) | π]

جہاں R(s_t, a_t) = E[r_t | S_t=s_t, A_t=a_t]

```python
class HierarchicalRLAgent:
    def __init__(self, state_dim, action_dim, options_dim, 
                 high_level_policy, low_level_policies):
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.options_dim = options_dim
        
        # ہائی لیول پالیسی: آپشنس کا انتخاب کرتا ہے
        self.high_level_policy = high_level_policy
        self.low_level_policies = low_level_policies  # فی آپشن لو لیول پالیسی
        self.option_terminations = [TerminationNet() for _ in range(options_dim)]
    
    def sample_option(self, state, state_history):
        """ہائی لیول پالیسی کا استعمال کرتے ہوئے آپشن منتخب کریں"""
        option_logits = self.high_level_policy(state, state_history)
        option = torch.multinomial(torch.softmax(option_logits, dim=-1), 1).item()
        return option
    
    def should_terminate(self, option, state):
        """چیک کریں کہ آیا موجودہ آپشن کو ختم کرنا چاہیے"""
        termination_prob = self.option_terminations[option](state)
        return np.random.rand() < termination_prob.item()
    
    def execute_option(self, option, state, max_steps=100):
        """موجودہ آپشن کو زیادہ سے زیادہ اسٹیپس تک چلائیں"""
        trajectory = []
        current_state = state
        steps = 0
        
        while steps < max_steps and not self.should_terminate(option, current_state):
            # آپشن کی لو لیول پالیسی کا استعمال کریں
            action = self.low_level_policies[option](current_state)
            
            # کارروائی انجام دیں (ماحول میں)
            next_state, reward, done, info = self.environment.step(action)
            
            trajectory.append({
                'state': current_state,
                'action': action,
                'reward': reward,
                'done': done,
                'option': option
            })
            
            current_state = next_state
            steps += 1
            
            if done:
                break
        
        return trajectory, current_state
```

### Option-Critic الگورتھم

Option-Critic الگورتھم آپشنس اور ان کی مدت کے مطابق سیکھتا ہے:

```python
class OptionCritic(nn.Module):
    def __init__(self, state_dim, action_dim, n_options, hidden_dim=256):
        super(OptionCritic, self).__init__()
        
        # Q-ویلیو فنکشن: Q(s,o) - آپشن کا ویلیو حالت میں
        self.q_network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_options)  # ہر آپشن کے لیے ایک Q-ویلیو
        )
        
        # Intr-option پالیسی: π(a|s,o)
        self.intra_option_policy = nn.Sequential(
            nn.Linear(state_dim + n_options, hidden_dim),
            nn.ReLU(),  # + آپشن کا ایمبیڈنگ
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
            nn.Softmax(dim=-1)
        )
        
        # آپشن ٹرمینیشن فنکشن: β(s,o)
        self.termination_network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, n_options),
            nn.Sigmoid()  # ہر آپشن کے لیے ٹرمینیشن امکان
        )
    
    def forward(self, state, option_embedding=None):
        q_values = self.q_network(state)
        
        if option_embedding is not None:
            # Intra-option پالیسی کے لیے حالت + آپشن کو جوڑیں
            state_option = torch.cat([state, option_embedding], dim=-1)
            action_probs = self.intra_option_policy(state_option)
        else:
            action_probs = None
        
        termination_probs = self.termination_network(state)
        
        return q_values, action_probs, termination_probs

def update_option_critic(agent, state, option, action, reward, 
                        next_state, next_option, done, gamma):
    """Option-Critic الگورتھم کے لیے اپ ڈیٹ قواعد"""
    current_q, _, _ = agent(state, option_onehot(option))
    next_q, _, next_termination = agent(next_state, option_onehot(next_option))
    
    # اندرونی اختیار کے لیے پالیسی نقصان
    action_probs, _ = agent.intra_option_policy(
        torch.cat([state, option_onehot(option)], dim=-1)
    )
    log_prob = torch.log(action_probs[action] + 1e-8)
    
    # ایڈوانٹیج حساب کریں
    with torch.no_grad():
        next_q_max = torch.max(next_q)
        target_q = reward + gamma * (
            (1 - next_termination[next_option]) * next_q_max + 
            next_termination[next_option] * torch.max(next_q)
        )
    
    advantage = target_q - current_q[option]
    
    # پالیسی گریڈیئنٹ
    policy_loss = -log_prob * advantage
    
    # Q-ویلیو نقصان
    q_loss = F.mse_loss(current_q[option], target_q)
    
    # ٹرمینیشن نقصان
    termination_loss = termination_probs[option] * advantage
    
    return policy_loss + q_loss + termination_loss
```

### کو وارینس-بیسڈ سینسر فیوژن

```python
class CovarianceBasedFusion:
    """
    کو وارینس کی بنیاد پر سینسر فیوژن
    کم کو وارینس والے سینسرز کو زیادہ وزن دیں
    """
    def __init__(self, sensor_specs):
        """
        sensor_specs: [{name, noise_variance, measurement_function}, ...]
        """
        self.sensors = sensor_specs
        self.last_measurements = {}
        self.covariances = {s['name']: s['noise_variance'] for s in sensor_specs}
    
    def fuse_measurements(self, measurements):
        """
        متعدد سینسرز سے ماپ کو ضم کریں
        """
        # امکان کا وزن حساب کریں: w_i = 1 / σ²_i
        weights = {}
        weighted_sum = 0
        total_weight = 0
        
        for sensor_name, measurement in measurements.items():
            if sensor_name in self.covariances:
                variance = self.covariances[sensor_name]
                weight = 1.0 / (variance + 1e-8)  # 0 وارینس سے بچیں
                weights[sensor_name] = weight
                
                weighted_sum += weight * measurement
                total_weight += weight
        
        if total_weight > 0:
            fused_estimate = weighted_sum / total_weight
            fused_variance = 1.0 / total_weight
        else:
            # کوئی جائز وزن نہیں، پہلا سینسر کا استعمال کریں
            fused_estimate = next(iter(measurements.values()))
            fused_variance = max(self.covariances.values())
        
        return fused_estimate, fused_variance

class ExtendedKalmanFilterFusion:
    """
    گھریلو ماحول کے مطابق EKF کا استعمال کرتے ہوئے سینسر فیوژن
    """
    def __init__(self, state_dim, process_noise_covariance):
        self.state_dim = state_dim
        self.state = np.zeros(state_dim)  # حالت ویکٹر
        self.covariance = np.eye(state_dim)  # کو وارینس میٹرکس
        self.Q = process_noise_covariance  # عمل نوائز کوواریئنس
    
    def predict(self, control_input=None):
        """حالت اور کوواریئنس کی پیش گوئی کریں"""
        # یہ حالت کا تصور کریں کہ X_{t+1} = F X_t + B u_t + w_t
        F = self.compute_state_transition_matrix()  # دنیا کا ماڈل
        B = self.compute_control_matrix() if control_input is not None else 0
        
        # حالت کی پیش گوئی
        self.state = F @ self.state
        if control_input is not None:
            self.state += B @ control_input
        
        # کوواریئنس کی پیش گوئی
        self.covariance = F @ self.covariance @ F.T + self.Q
    
    def update(self, measurements):
        """
        متعدد سینسرز کے مشاہدات کے مطابق حالت کو اپ ڈیٹ کریں
        measurements: {'sensor1': (z1, R1), 'sensor2': (z2, R2), ...}
        """
        for sensor_name, (measurement, sensor_covariance) in measurements.items():
            # مشاہدہ ماڈل: z = H x + v
            H = self.compute_observation_matrix(sensor_name)
            
            # ابھار ( innovation )
            predicted_measurement = H @ self.state
            innovation = measurement - predicted_measurement
            
            # ابھار کوواریئنس
            innovation_cov = H @ self.covariance @ H.T + sensor_covariance
            
            # کیلمین گین
            K = self.covariance @ H.T @ np.linalg.inv(innovation_cov)
            
            # حالت کو اپ ڈیٹ کریں
            self.state = self.state + K @ innovation
            
            # کوواریئنس کو اپ ڈیٹ کریں
            I = np.eye(len(self.state))
            self.covariance = (I - K @ H) @ self.covariance
    
    def compute_state_transition_matrix(self):
        """حالت منتقلی میٹرکس F کمپیوٹ کریں"""
        # یہ مسئلہ کے مطابق ہوگا
        # مثال کے طور پر، مستقل رفتار ماڈل
        F = np.eye(self.state_dim)
        dt = self.time_step  # وقت کا قدم
        
        # اگر حالت [x, y, vx, vy] ہے، تو:
        if self.state_dim >= 4:
            F[0, 2] = dt  # x = x + vx*dt
            F[1, 3] = dt  # y = y + vy*dt
        
        return F
    
    def compute_observation_matrix(self, sensor_name):
        """
        سینسر کے لیے مشاہدہ میٹرکس H کمپیوٹ کریں
        """
        if sensor_name == 'position':
            # اگر حالت [x, y, vx, vy] ہے اور ہم x,y حاصل کر رہے ہیں:
            H = np.zeros((2, self.state_dim))
            H[0, 0] = 1  # مشاہدہ x
            H[1, 1] = 1  # مشاہدہ y
        elif sensor_name == 'velocity':
            # مشاہدہ vx, vy
            H = np.zeros((2, self.state_dim))
            H[0, 2] = 1  # مشاہدہ vx
            H[1, 3] = 1  # مشاہدہ vy
        else:
            raise ValueError(f"Unknown sensor: {sensor_name}")
        
        return H
```

### فوکسڈ جنرل فیوژن نیٹ ورک

```python
class FocusedGeneralFusionNetwork(nn.Module):
    """
    توجہ والے میکنزمز کا استعمال کرتے ہوئے متعدد سینسر ڈیٹا کو ضم کریں
    """
    def __init__(self, sensor_dimensions, fusion_output_dim):
        super(FocusedGeneralFusionNetwork, self).__init__()
        
        self.num_sensors = len(sensor_dimensions)
        self.sensor_dimensions = sensor_dimensions
        
        # ہر سینسر کے لیے الگ انکوڈنگ
        self.sensor_encoders = nn.ModuleList([
            nn.Sequential(
                nn.Linear(dim, 128),
                nn.ReLU(),
                nn.Linear(128, 256),
                nn.ReLU()
            ) for dim in sensor_dimensions
        ])
        
        # متعدد سر اٹینشن: متعلقہ سینسرز پر توجہ مرکوز کریں
        self.multihead_attention = nn.MultiheadAttention(
            embed_dim=256, num_heads=8, dropout=0.1
        )
        
        # فیوژن کریں
        self.fusion_network = nn.Sequential(
            nn.Linear(256 * len(sensor_dimensions), 512),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(512, fusion_output_dim),
            nn.ReLU()
        )
    
    def forward(self, sensor_inputs):
        """
        sensor_inputs: [sensor1_data, sensor2_data, ...]
        """
        encoded_features = []
        
        # ہر سینسر کو الگ الگ انکوڈ کریں
        for i, (sensor_data, encoder) in enumerate(zip(sensor_inputs, self.sensor_encoders)):
            encoded = encoder(sensor_data)
            encoded_features.append(encoded)
        
        # سینسرز کو ایک ساتھ ملائیں
        stacked_features = torch.stack(encoded_features, dim=1)  # [Batch, Num_Sensors, Features]
        
        # متعدد سر توجہ: متعلقہ سینسرز پر توجہ مرکوز کریں
        attended_features, attention_weights = self.multihead_attention(
            stacked_features, stacked_features, stacked_features
        )
        
        # فلیٹن اور حتمی فیوژن
        batch_size = attended_features.size(0)
        flattened = attended_features.view(batch_size, -1)
        fused_output = self.fusion_network(flattened)
        
        return fused_output, attention_weights

def create_multimodal_agent():
    """
    کئی ماڈیولز والے سینسرز کو ضم کرنے کے لیے ایجنٹ تخلیق کریں
    """
    # سینسر ڈیمنشنس: [ویژن، لائڈر، امیو، ٹیکٹائل]
    sensor_dims = [3, 256, 6, 128]  # ویژن، لائڈر، امیو، ٹیکٹائل کے لیے
    fusion_agent = FocusedGeneralFusionNetwork(sensor_dims, fusion_output_dim=512)
    
    return fusion_agent
```

### ماحول کے ماڈل کی تعمیر

```python
class WorldModel(nn.Module):
    """
    سیکھا ہوا ماحول کا ماڈل جس میں ڈائینمکس اور انعام شامل ہیں
    """
    def __init__(self, state_dim, action_dim, hidden_dim=256):
        super(WorldModel, self).__init__()
        
        # حالت-ایکشن کو ایمبیڈ کریں
        self.state_action_encoder = nn.Sequential(
            nn.Linear(state_dim + action_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim)
        )
        
        # منتقلی ماڈل: اگلی حالت کی پیش گوئی کریں
        self.transition_model = nn.Sequential(
            nn.Linear(hidden_dim + hidden_dim, hidden_dim),  # [prev_state_encoded, state_action_embed]
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, state_dim)  # Delta update to current state
        )
        
        # انعام ماڈل: انعام کی پیش گوئی کریں
        self.reward_model = nn.Sequential(
            nn.Linear(hidden_dim + hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)  # واحد انعام سکیلر
        )
        
        # مسل اور کو واریئنس کے لیے وی بیریئنکل ماڈل
        self.state_var_model = nn.Sequential(
            nn.Linear(hidden_dim + hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, state_dim)
        )
    
    def forward(self, current_state, action, prev_encoded_state=None):
        # حالت-ایکشن کو ایمبیڈ کریں
        state_action = torch.cat([current_state, action], dim=-1)
        sa_embed = self.state_action_encoder(state_action)
        
        # پچھلی حالت کو بھی ہدف میں شامل کریں اگر دی گئی ہو
        if prev_encoded_state is not None:
            transition_input = torch.cat([prev_encoded_state, sa_embed], dim=-1)
        else:
            transition_input = sa_embed
        
        # اگلی حالت کی پیش گوئی کریں
        state_delta = self.transition_model(transition_input)
        next_state_pred = current_state + state_delta  # Residual update
        
        # انعام کی پیش گوئی کریں
        reward_input = torch.cat([prev_encoded_state, sa_embed], dim=-1) if prev_encoded_state is not None else sa_embed
        reward_pred = self.reward_model(reward_input)
        
        # واریئنس (اچنبیت کی جانچ) کی پیش گوئی کریں
        var_pred = torch.exp(self.state_var_model(transition_input))
        
        return next_state_pred, reward_pred, var_pred

class ModelBasedAgent:
    """
    سیکھے ہوئے ماحول کے ماڈل کا استعمال کرکے منصوبہ بندی کریں
    """
    def __init__(self, world_model, policy_network, planning_horizon=10):
        self.world_model = world_model
        self.policy_network = policy_network
        self.planning_horizon = planning_horizon
        self.optimizer = torch.optim.Adam(
            list(world_model.parameters()) + list(policy_network.parameters()),
            lr=3e-4
        )
    
    def plan_with_model(self, initial_state, goal_state):
        """سیکھے ہوئے ماڈل کا استعمال کرکے منصوبہ بندی کریں"""
        best_sequence = None
        best_value = float('-inf')
        
        # بے ترتیب ایکشن سیکوئنز نمونہ لیں
        for _ in range(1000):  # مثال کے طور پر 1000 نمونے
            state = initial_state.clone()
            total_value = 0
            
            action_sequence = []
            for t in range(self.planning_horizon):
                # ایکشن سیکھنے کے لیے پالیسی نیٹ ورک استعمال کریں
                action_probs = self.policy_network(state)
                action = torch.multinomial(action_probs, 1)  # یا اس کے بجائے دیگر حکم دیں
                
                # ماڈل کا استعمال کرکے اگلی حالت اور انعام کی پیش گوئی کریں
                next_state_pred, reward_pred, _ = self.world_model(state, action)
                
                total_value += (0.99 ** t) * reward_pred.item()
                state = next_state_pred
                action_sequence.append(action)
            
            if total_value > best_value:
                best_value = total_value
                best_sequence = action_sequence[:]
        
        return best_sequence[0] if best_sequence else None  # پہلا ایکشن واپس کریں
    
    def update_world_model(self, real_experience_batch):
        """حقیقی تجربے کے ساتھ ماحول کے ماڈل کو اپ ڈیٹ کریں"""
        states, actions, rewards, next_states = real_experience_batch
        
        loss = 0
        for i in range(len(states)):
            pred_next_state, pred_reward, pred_var = self.world_model(
                states[i], actions[i], states[i-1] if i > 0 else None
            )
            
            # حالت اور انعام کے نقصان کو یکجا کریں
            state_loss = F.mse_loss(pred_next_state, next_states[i])
            reward_loss = F.mse_loss(pred_reward, rewards[i].unsqueeze(0))
            
            loss += state_loss + reward_loss
        
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
```

```python
class CommNet(nn.Module):
    """
    روبوٹ کمیونیکیشن نیٹ ورک
    """
    def __init__(self, n_agents, msg_dim, hidden_dim=128):
        super(CommNet, self).__init__()
        self.n_agents = n_agents
        self.msg_dim = msg_dim
        
        # ایجنٹ کی حالت کو پیغام میں تبدیل کریں
        self.state_to_msg = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, msg_dim)
        )
        
        # موصولہ پیغامات کو عمل میں لائیں
        self.comm_processor = nn.GRU(
            input_size=msg_dim,
            hidden_size=hidden_dim,
            batch_first=True
        )
        
        # متوازی پیغامات کے لیے
        self.message_sharing = nn.MultiheadAttention(
            embed_dim=msg_dim, num_heads=4
        )
    
    def forward(self, agent_states, prev_messages=None):
        """
        ایجنٹس کے درمیان مواصلات کو ماڈل کریں
        """
        batch_size = agent_states.size(0)
        
        # ہر ایجنٹ کی حالت سے پیغام تیار کریں
        messages = []
        for i in range(self.n_agents):
            agent_state = agent_states[:, i, :]  # [batch, state_dim]
            msg = self.state_to_msg(agent_state)  # [batch, msg_dim]
            messages.append(msg)
        
        # سب کو ایک ساتھ جوڑیں
        all_messages = torch.stack(messages, dim=1)  # [batch, n_agents, msg_dim]
        
        # مواصلات کی تازہ کاری کریں (GRU یا اٹینشن کے ذریعے)
        if prev_messages is not None:
            combined_msgs = torch.cat([prev_messages, all_messages], dim=-1)
            updated_messages, _ = self.comm_processor(combined_msgs)
        else:
            updated_messages, _ = self.comm_processor(all_messages)
        
        # پیغامات کو ایجنٹس کے لیے دستیاب کرائیں
        return updated_messages
```

```python
    """

        """
```

```python
class ParticleFilterFusion:
    """
    پارٹیکل فلٹر کا استعمال کرتے ہوئے غیر لکیری سینسر فیوژن
    """
    def __init__(self, state_dim, n_particles=1000):
        self.state_dim = state_dim
        self.n_particles = n_particles
        self.particles = np.random.randn(n_particles, state_dim)
        self.weights = np.ones(n_particles) / n_particles
    
    def predict_particles(self, control_input, process_noise_cov):
        """اقدامات اور عمل نوائز کی بنیاد پر پارٹیکلز کو آگے بڑھائیں"""
        for i in range(self.n_particles):
            # عمل ماڈل کا استعمال کریں: x_{t+1} = f(x_t, u_t) + w_t
            self.particles[i] = self.dynamics_model(
                self.particles[i], control_input
            ) + np.random.multivariate_normal(
                np.zeros(self.state_dim), process_noise_cov
            )
    
    def update_weights(self, measurements):
        """نئے مشاہدات کی بنیاد پر پارٹیکل ویٹس کو اپ ڈیٹ کریں"""
        for i in range(self.n_particles):
            total_likelihood = 1.0
            
            for sensor_name, (z, R) in measurements.items():
                # پارٹیکل کے لیے متوقع مشاہدہ حاصل کریں
                expected_z = self.observation_model(sensor_name, self.particles[i])
                
                # مشاہدہ لائک لی ہوڈ کمپیوٹ کریں
                innovation = z - expected_z
                likelihood = multivariate_normal.pdf(
                    innovation, 
                    mean=np.zeros_like(innovation), 
                    cov=R
                )
                
                total_likelihood *= likelihood
            
            # کل لائکلی ہوڈ کو موجودہ وزن سے ضرب دیں
            self.weights[i] *= total_likelihood
        
        # نارملائز کریں
        self.weights += 1e-10  # نیومریکل استحکام کے لیے
        self.weights /= np.sum(self.weights)
    
    def resample(self):
        """کم ویٹس والے پارٹیکلز کو چھوٹے، زیادہ ویٹس والے کو زیادہ دوبارہ نمونہ بندی کریں"""
        indices = np.random.choice(
            self.n_particles, 
            size=self.n_particles, 
            p=self.weights
        )
        self.particles = self.particles[indices]
        self.weights = np.ones(self.n_particles) / self.n_particles
    
    def estimate_state(self):
        """وزن دی گئی اوسط کے ذریعے حالت کا تخمینہ لگائیں"""
        return np.average(self.particles, axis=0, weights=self.weights)
```

```python
class DistributionalRLAgent:
    """
    Q-ویلیو کی بجائے واپسی کے پورے تقسیم کو سیکھنا
    """
    def __init__(self, state_dim, action_dim, n_atoms=51, v_min=-10.0, v_max=10.0):
        self.n_atoms = n_atoms
        self.v_min = v_min
        self.v_max = v_max
        self.action_dim = action_dim
        
        # سپورٹ کے لیے
        self.support = torch.linspace(v_min, v_max, n_atoms)
        self.delta_z = (v_max - v_min) / (n_atoms - 1)
        
        # تقسیم کو آؤٹ پٹ کرنے والے نیٹ ورک
        self.network = nn.Sequential(
            nn.Linear(state_dim, 512),
            nn.ReLU(),
            nn.Linear(512, 512),
            nn.ReLU(),
            nn.Linear(512, action_dim * n_atoms),  # ہر ایکشن کے لیے n_atoms پروبابلیٹیز
            nn.Softmax(dim=-1)  # پچھلے محور پر نارملائز کریں
        )
    
    def forward(self, state):
        logits = self.network(state)
        # بعد میں ایکشن اور ایٹمس کے لیے میپ کریں
        return logits.view(-1, self.action_dim, self.n_atoms)
    
    def compute_q_values(self, state):
        """تقسیم سے متوقع Q-ویلیوز حساب کریں"""
        probs = self.forward(state)
        # Q(s,a) = Σ_i Z_i * p(Z_i | s,a)
        q_values = torch.sum(probs * self.support, dim=2)
        return q_values

class CategoricalDQN(DistributionalRLAgent):
    """
    منفی کیٹاگوریکل بیلمن ایکویشن کا استعمال کرتے ہوئے
    """
    def update_categorical_targets(self, states, actions, rewards, next_states, dones, gamma=0.99):
        with torch.no_grad():
            # اگلی حالت کے لیے تقسیم حاصل کریں
            next_logits = self.forward(next_states)
            next_probs = F.softmax(next_logits, dim=-1)
            
            # قطعی میزان کے ساتھ اگلی تقسیم کا امتزاج حاصل کریں
            next_q_values = torch.sum(next_probs * self.support, dim=2)
            next_actions = torch.argmax(next_q_values, dim=1)
            
            # مطلوبہ تقسیم کو استحکام دیں
            projected_probs = self.project_distribution(
                rewards, next_probs, next_actions, dones
            )
        
        # موجودہ ریاست کے لیے تقسیم حاصل کریں
        current_logits = self.forward(states)
        current_probs = F.softmax(current_logits, dim=-1)
        
        # نفاذ کا نقصان (کروس اینٹروپی)
        criterion = nn.CrossEntropyLoss(reduction='none')
        loss = criterion(
            current_probs[range(len(actions)), actions], 
            projected_probs
        )
        
        return loss.mean()
    
    def project_distribution(self, rewards, next_probs, next_actions, dones):
        """next_probs کو مطلوبہ تقسیم میں منسلک کریں"""
        batch_size = rewards.size(0)
        
        # تکلیف کے مساوات کو لاگو کریں: Tz = R + γ * z
        support = self.support.to(rewards.device)
        Tz = rewards.unsqueeze(1) + (1 - dones.float()).unsqueeze(1) * gamma * support.unsqueeze(0)
        Tz = torch.clamp(Tz, self.v_min, self.v_max)
        
        # Tz کے ذریعے نکٹرز کا حساب لگائیں
        b = (Tz - self.v_min) / self.delta_z
        l = b.floor().long()
        u = b.ceil().long()
        
        # ہدف تقسیم کو تعمیر کریں
        proj_dist = torch.zeros_like(next_probs)
        for i in range(batch_size):
            for j in range(self.n_atoms):
                pj = next_probs[i, next_actions[i], j]
                proj_dist[i, l[i, j]] += pj * (u[i, j] - b[i, j])
                proj_dist[i, u[i, j]] += pj * (b[i, j] - l[i, j])
        
        return proj_dist
```
